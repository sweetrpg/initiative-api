func main() {
        _ = godotenv.Load(".env")

        cfg, err := config.FromEnv()
        if err != nil {
                log.Fatalf("invalid configuration:\n%v", err)
        }

        // common.go/logging reads LOG_LEVEL itself; pass the validated, defaulted value.
        _ = os.Setenv(constants.LOG_LEVEL, cfg.LogLevel)
        logging.Init()

        tracing.SetupTracing(constants.ServiceName)
        defer tracing.TeardownTracing()

        database.SetupDatabase() // reads the MongoDB connection settings from the environment
        defer database.TeardownDatabase()
        db := store.MongoDatabase{DB: database.Db}

        idxCtx, idxCancel := context.WithTimeout(context.Background(), time.Minute)
        if err := data.EnsureIndexes(idxCtx, db); err != nil {
                log.Fatalf("indexes: %v", err)
        }
        idxCancel()

        themes, err := domain.NewThemes(db)
        if err != nil {
                log.Fatalf("themes: %v", err)
        }
        creds := domain.NewCredentials(db)
        // Credentials stay off (503) until the auth-event consumer is bound and caught up, so a deny
        // published while the service was down is applied before any secret authenticates (PADR-0039).
        if natsURL := os.Getenv("NATS_URL"); natsURL == "" {
                logging.Logger.Warn("NATS_URL is not set; integration credentials are disabled")
        } else {
                consumer := events.New(creds, creds)
                if err := consumer.Start(context.Background(), natsURL, os.Getenv("NATS_DURABLE_NAME")); err != nil {
                        log.Fatalf("auth events: %v", err)
                }
                defer consumer.Stop()
        }
        pool := util.RedisPool() // non-nil: config requires REDIS_HOST
        rosters, encounters, sessions := domain.NewRosters(db), domain.NewEncounters(db), domain.NewSessions(db)
        svc := server.Services{
                Locks: encounters.Locks(),
                Limits: domain.NewLimits(
                        domain.Kind{Name: "themes", Key: domain.KeyThemeSlots, Repo: themes.Items()},
                        domain.Kind{Name: "rosters", Key: domain.KeySavedRosters, Repo: rosters.Items()},
                        domain.Kind{Name: "encounters", Key: domain.KeySavedEncounters, Repo: encounters.Items()},
                        domain.Kind{Name: "sessions", Key: domain.KeySessionsPerGM, Repo: sessions.Items()},
                        domain.Kind{Name: "credentials", Key: domain.KeyIntegrationCredentials, Repo: creds.Items()},
                ),
                Authz:        authz.NewClient(cfg.AuthAPIURL, cfg.UsersAPIURL),
                Entitlements: entitlements.New(cfg.EntitlementsAPIURL),
                Themes:       themes,
                Credentials:  creds,
                Rosters:      rosters,
                Encounters:   encounters,
                Sessions:     sessions,
                Relations:    relations.New(cfg.AuthAPIURL),
        }
        srv := &http.Server{Addr: cfg.BindAddress, Handler: newRouter(cfg, svc, pool), ReadHeaderTimeout: 10 * time.Second}

        go func() {
                if err := srv.ListenAndServe(); err != nil && !errors.Is(err, http.ErrServerClosed) {
                        logging.Logger.Error("server failed", "error", err.Error())
                        os.Exit(1)
                }
        }()

        stop := make(chan os.Signal, 1)
        signal.Notify(stop, syscall.SIGINT, syscall.SIGTERM)
        <-stop

        ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
        defer cancel()
        if err := srv.Shutdown(ctx); err != nil {
                logging.Logger.Error("shutdown failed", "error", err.Error())
        }
}

func newRouter(cfg config.Config, svc server.Services, pool *redis.Pool) *gin.Engine {
        httpLogger := slog.New(slog.NewJSONHandler(os.Stdout, nil))

        r := gin.New()
        // Correlation runs first so the access log and every handler see the ID.
        r.Use(server.Correlation())
        r.Use(sloggin.New(httpLogger))
        r.Use(gin.Recovery())
        r.Use(otelgin.Middleware(constants.ServiceName))

        if len(cfg.AllowedOrigins) > 0 {
                r.Use(cors.New(cors.Config{
                        AllowOrigins:     cfg.AllowedOrigins,
                        AllowMethods:     []string{"GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"},
                        AllowHeaders:     []string{"Origin", "Content-Type", "Accept", "Authorization", constants.CorrelationHeader},
                        ExposeHeaders:    []string{constants.CorrelationHeader},
                        AllowCredentials: true,
                        MaxAge:           12 * time.Hour,
                }))
        }

        r.Use(server.RateLimit(pool, ratelimit.DefaultOptions(), svc.Credentials)...)

        server.SetupHandlers(r, svc)
        return r
}

// Command reconciler runs one OpenFGA tuple reconciliation pass and exits (PADR-0035, PADR-0038).
// It exits non-zero when the pass fails or the reconciler token is rejected or expired, so the
// CronJob shows the failure and the alert fires.
package main

import (
        "context"
        "errors"
        "log"
        "os"
        "strings"
        "time"

        "github.com/sweetrpg/common.go/logging"
        "github.com/sweetrpg/initiative-api/data"
        "github.com/sweetrpg/initiative-api/reconcile"
        "github.com/sweetrpg/initiative-api/store"
        "github.com/sweetrpg/mongodb.go/database"
)

func main() { os.Exit(run()) }

func run() int {
        logging.Init()
        authURL, tokenFile := os.Getenv("AUTH_API_URL"), os.Getenv("RECONCILER_TOKEN_FILE")
        if authURL == "" || tokenFile == "" {
                log.Print("AUTH_API_URL and RECONCILER_TOKEN_FILE are required")
                return 2
        }
        raw, err := os.ReadFile(tokenFile)
        token := strings.TrimSpace(string(raw))
        if err != nil || token == "" {
                log.Print("reconciler token file is missing or empty; mint a new token (PADR-0038)")
                return 2
        }

        database.SetupDatabase()
        defer database.TeardownDatabase()

        // Under the CronJob's activeDeadlineSeconds (270); the last minute is reserved for mint
        // garbage collection (reconcile.Reconciler.GCReserve).
        ctx, cancel := context.WithTimeout(context.Background(), 4*time.Minute)
        defer cancel()
        sessions := data.NewSessions(store.MongoDatabase{DB: database.Db})
        api := reconcile.NewAuthAPI(authURL, token)
        r := reconcile.Reconciler{Members: sessions, Tuples: api, Mints: api, Sessions: sessions}
        res, err := r.Run(ctx)
        logging.Logger.Info("reconcile finished", "sessions", res.Sessions, "written", res.Written, "deleted", res.Deleted, "failed", res.Failed,
                "mints_closed", res.MintsClosed, "mints_skipped", res.MintsSkipped)
        switch {
        case errors.Is(err, reconcile.ErrTokenRejected):
                logging.Logger.Error("reconciler token rejected or expired; an operator must mint a new one")
                return 3
        case err != nil:
                logging.Logger.Error("reconcile failed", "error", err.Error())
                return 1
        }
        return 0
}

// Config is the validated runtime configuration.
type Config struct {
        BindAddress string
        Env         string
        Version     string
        LogLevel    string
        // OTLPEndpoint is the trace collector base URL; empty leaves the exporter default.
        OTLPEndpoint string
        // AllowedOrigins is the CORS allowlist; empty disables CORS (fail closed).
        AllowedOrigins []string
        // AuthAPIURL is auth-api's base URL. Bearer tokens are verified there via /authz/check
        // (signature, expiry, iss, aud, roles); this service holds no signing keys or credentials.
        AuthAPIURL string
        // UsersAPIURL is users-api's base URL, used to resolve the verified caller to users._id.
        UsersAPIURL string
        // EntitlementsAPIURL is entitlements-api's base URL for gating checks.
        EntitlementsAPIURL string
        // RedisHost is the rate limiter's counter store; the limiter fails closed without it
        // (PADR-0036), so it is required. Port and password are read by the Redis pool itself.
        RedisHost string
}

var logLevels = map[string]bool{"DEBUG": true, "INFO": true, "WARN": true, "ERROR": true}

// Load reads the environment through lookup (os.LookupEnv in production) and returns every
// validation problem at once, so a misconfigured deploy reports all of them in one start attempt.
func Load(lookup func(string) (string, bool)) (Config, error) {
        get := func(key, def string) string {
                if v, ok := lookup(key); ok && strings.TrimSpace(v) != "" {
                        return strings.TrimSpace(v)
                }
                return def
        }

        c := Config{
                BindAddress:        get("BIND_ADDRESS", ":8000"),
                Env:                get("ENV", "dev"),
                Version:            get("VERSION", "v0.0.0"),
                LogLevel:           strings.ToUpper(get(constants.LOG_LEVEL, "WARN")),
                OTLPEndpoint:       get(constants.OTEL_EXPORTER_OTLP_ENDPOINT, ""),
                AuthAPIURL:         strings.TrimRight(get(constants.AUTH_API_URL, ""), "/"),
                UsersAPIURL:        strings.TrimRight(get(constants.USERS_API_URL, ""), "/"),
                EntitlementsAPIURL: strings.TrimRight(get(constants.ENTITLEMENTS_API_URL, ""), "/"),
                RedisHost:          get("REDIS_HOST", ""),
        }
        if origins := get(constants.ALLOWED_ORIGINS, ""); origins != "" {
                c.AllowedOrigins = strings.Split(origins, ",")
        }

        var errs []error
        if _, _, err := net.SplitHostPort(c.BindAddress); err != nil {
                errs = append(errs, fmt.Errorf("BIND_ADDRESS %q: %w", c.BindAddress, err))
        }
        if !logLevels[c.LogLevel] {
                errs = append(errs, fmt.Errorf("%s %q: must be one of DEBUG, INFO, WARN, ERROR", constants.LOG_LEVEL, c.LogLevel))
        }
        errs = append(errs, checkURL(constants.OTEL_EXPORTER_OTLP_ENDPOINT, c.OTLPEndpoint, false)...)
        errs = append(errs, checkURL(constants.AUTH_API_URL, c.AuthAPIURL, true)...)
        errs = append(errs, checkURL(constants.USERS_API_URL, c.UsersAPIURL, true)...)
        errs = append(errs, checkURL(constants.ENTITLEMENTS_API_URL, c.EntitlementsAPIURL, true)...)
        if c.RedisHost == "" {
                errs = append(errs, errors.New("REDIS_HOST: required"))
        }
        for _, o := range c.AllowedOrigins {
                if strings.TrimSpace(o) == "*" {
                        errs = append(errs, fmt.Errorf("%s: wildcard origin is not allowed", constants.ALLOWED_ORIGINS))
                }
        }

        return c, errors.Join(errs...)
}

// FromEnv is Load against the process environment.
func FromEnv() (Config, error) { return Load(os.LookupEnv) }

// checkURL validates an absolute http(s) URL; an empty value is an error only when required.
func checkURL(key, value string, required bool) []error {
        if value == "" {
                if required {
                        return []error{fmt.Errorf("%s: required", key)}
                }
                return nil
        }
        if u, err := url.Parse(value); err != nil || (u.Scheme != "http" && u.Scheme != "https") || u.Host == "" {
                return []error{fmt.Errorf("%s %q: must be an absolute http(s) URL", key, value)}
        }
        return nil
}

// CredentialDeletion is the append-only record of an administrator's permanent deletion of an
// integration credential. Its _id is the deleted credential's id, so a retried deletion cannot
// write a second entry. Nothing in the service updates or removes these (PADR-0027).
type CredentialDeletion struct {
        CredentialID string    `bson:"_id" json:"credential_id"`
        OwnerUserID  string    `bson:"owner_user_id" json:"owner_user_id"`
        Label        string    `bson:"label" json:"label"`
        Scopes       []string  `bson:"scopes" json:"scopes"`
        Reason       string    `bson:"reason" json:"reason"`
        CreatedBy    string    `bson:"created_by" json:"deleted_by"`
        CreatedAt    time.Time `bson:"created_at" json:"deleted_at"`
}

// ownerPattern is the shape of a user id (Auth0 subject or users-api id). An owner filter taken
// from a request must match it, so only a plain id can reach a query.
var ownerPattern = regexp.MustCompile(`^[A-Za-z0-9_.|:-]{1,128}$`)

func checkOwner(owner string) error {
        if owner != "" && !ownerPattern.MatchString(owner) {
                return invalid("owner is not a valid user id")
        }
        return nil
}

// ListAll returns one page of credentials across owners, newest first. Administrator use only:
// callers gate it on the admin role. Its filter has no request input; one owner's credentials go
// through the owner-scoped List instead.
func (r *Credentials) ListAll(ctx context.Context, q Query) (Page[IntegrationCredential], error) {
        if len(q.Sort) == 0 {
                q.Sort = []SortKey{{Field: "created_at", Desc: true}}
        }
        return runQuery[IntegrationCredential](ctx, r.c, r.spec, bson.D{{Key: "deleted_at", Value: nil}}, q)
}

// ListOwned is the owner-scoped List for an owner id taken from an administrator's request.
func (r *Credentials) ListOwned(ctx context.Context, owner string, q Query) (Page[IntegrationCredential], error) {
        if err := checkOwner(owner); err != nil {
                return Page[IntegrationCredential]{}, err
        }
        if len(q.Sort) == 0 {
                q.Sort = []SortKey{{Field: "created_at", Desc: true}}
        }
        return r.List(ctx, owner, q)
}

// Live reports whether the credential can still authenticate: present, unrevoked, not over the
// owner's limit. Open streams re-check it so a revoke or deletion ends them.
func (r *Credentials) Live(ctx context.Context, id string) (bool, error) {
        id, ok := cleanID(id)
        if !ok {
                return false, nil
        }
        var c IntegrationCredential
        filter := bson.D{{Key: "_id", Value: id}, {Key: "revoked_at", Value: nil}, {Key: "deleted_at", Value: nil}, {Key: "inactive_at", Value: nil}}
        err := r.c.FindOne(ctx, filter, &c)
        if errors.Is(err, store.ErrNotFound) {
                return false, nil
        }
        return err == nil, err
}

// DeletePermanently removes any owner's credential and records who did it and why. The audit
// entry is written first, so a failure after it leaves the credential in place and a retry
// completes the deletion without a second entry. Returns the entry.
func (r *Credentials) DeletePermanently(ctx context.Context, actor, id, reason string) (CredentialDeletion, error) {
        var out CredentialDeletion
        if strings.TrimSpace(actor) == "" || strings.TrimSpace(reason) == "" {
                return out, fmt.Errorf("%w: administrator and reason are required", ErrInvalid)
        }
        id, ok := cleanID(id)
        if !ok {
                return out, ErrNotFound
        }
        var c IntegrationCredential
        err := r.c.FindOne(ctx, bson.D{{Key: "_id", Value: id}}, &c)
        if errors.Is(err, store.ErrNotFound) {
                return out, ErrNotFound
        }
        if err != nil {
                return out, err
        }
        out = CredentialDeletion{
                CredentialID: c.ID, OwnerUserID: c.OwnerUserID, Label: c.Label, Scopes: c.Scopes,
                Reason: strings.TrimSpace(reason), CreatedBy: actor, CreatedAt: r.now(),
        }
        // A duplicate means an earlier attempt wrote the entry: return that stored row, so the
        // response and log match what the audit collection holds.
        if err := r.deletions.InsertOne(ctx, out); errors.Is(err, store.ErrDuplicate) {
                if err := r.deletions.FindOne(ctx, bson.D{{Key: "_id", Value: id}}, &out); err != nil {
                        return CredentialDeletion{}, err
                }
        } else if err != nil {
                return CredentialDeletion{}, err
        }
        if err := r.c.DeleteOne(ctx, bson.D{{Key: "_id", Value: id}}); err != nil && !errors.Is(err, store.ErrNotFound) {
                return CredentialDeletion{}, err
        }
        return out, nil
}

// DeletionSpec pages the audit entries, newest first.
var DeletionSpec = Spec{
        Name:     CollectionCredentialDeletions,
        Sortable: []string{"created_at"},
}

// Deletions returns one page of audit entries, newest first; a non-empty NextCursor means more.
// owner narrows it to one owner's deleted credentials.
func (r *Credentials) Deletions(ctx context.Context, owner string, q Query) (Page[CredentialDeletion], error) {
        if err := checkOwner(owner); err != nil {
                return Page[CredentialDeletion]{}, err
        }
        if len(q.Sort) == 0 {
                q.Sort = []SortKey{{Field: "created_at", Desc: true}}
        }
        if owner == "" {
                return runQuery[CredentialDeletion](ctx, r.deletions, DeletionSpec, bson.D{}, q)
        }
        return runQuery[CredentialDeletion](ctx, r.deletions, DeletionSpec, bson.D{{Key: "owner_user_id", Value: owner}}, q)
}

// Credentials is the repository for integration credentials: the owner-scoped operations come
// from Owned, plus the state changes (revoke, use) and the one lookup that is not owner-scoped.
type Credentials struct {
        *Owned[IntegrationCredential, *IntegrationCredential]
        gate, deletions store.Collection
}

// NewCredentials builds the credential repository for db.
func NewCredentials(db store.Database) *Credentials {
        return &Credentials{NewOwned[IntegrationCredential](db, CredentialSpec), db.Collection(CollectionCredentialGate), db.Collection(CollectionCredentialDeletions)}
}

// activeScope is owner's credentials that are neither revoked nor deleted.
func activeScope(owner string) bson.D {
        return append(scope(owner), bson.E{Key: "revoked_at", Value: nil})
}

// ItemStates lists owner's unrevoked credentials; a revoked one no longer counts toward the limit.
func (r *Credentials) ItemStates(ctx context.Context, owner string) ([]ItemState, error) {
        if err := checkUserID(owner); err != nil {
                return nil, err
        }
        return r.itemStates(ctx, activeScope(owner))
}

// CountActive counts owner's credentials that can still authenticate.
func (r *Credentials) CountActive(ctx context.Context, owner string) (int64, error) {
        if err := checkUserID(owner); err != nil {
                return 0, err
        }
        return r.c.CountDocuments(ctx, activeScope(owner))
}

// Revoke ends owner's credential at once. Revoking an already revoked credential succeeds and
// returns it unchanged. Another owner's credential is ErrForbidden, an unknown one ErrNotFound.
func (r *Credentials) Revoke(ctx context.Context, owner, actor, id string) (IntegrationCredential, error) {
        var out IntegrationCredential
        if err := requireOwnerActor(owner, actor); err != nil {
                return out, err
        }
        id, ok := cleanID(id)
        if !ok {
                return out, ErrNotFound
        }
        now := r.now()
        update := bson.D{
                {Key: "$set", Value: bson.D{{Key: "revoked_at", Value: now}, {Key: "updated_at", Value: now}, {Key: "updated_by", Value: actor}}},
                {Key: "$inc", Value: bson.D{{Key: "revision", Value: 1}}},
        }
        filter := append(scopeID(owner, id), bson.E{Key: "revoked_at", Value: nil})
        err := r.c.FindOneAndUpdate(ctx, filter, update, &out)
        if !errors.Is(err, store.ErrNotFound) {
                return out, err
        }
        // Nothing matched: already revoked (fine), someone else's, or missing.
        cur, gerr := r.Get(ctx, owner, id)
        if gerr == nil {
                return cur, nil
        }
        if errors.Is(gerr, ErrNotFound) {
                _, cerr := r.classify(ctx, owner, id)
                if cerr != nil && !errors.Is(cerr, ErrRevisionConflict) {
                        return out, cerr
                }
        }
        return out, gerr
}

// Authenticate resolves a presented secret's hash to its live, unrevoked credential and stamps
// last_used_at in the same write, so a revocation is seen by the very next request. This is the
// one lookup that is not owner-scoped: the secret is what identifies the owner. secretHash must
// be the hash the service computed, never a client-supplied value.
func (r *Credentials) Authenticate(ctx context.Context, secretHash string) (IntegrationCredential, error) {
        var out IntegrationCredential
        if secretHash == "" {
                return out, ErrNotFound
        }
        filter := bson.D{{Key: "secret_hash", Value: secretHash}, {Key: "revoked_at", Value: nil}, {Key: "deleted_at", Value: nil}, {Key: "inactive_at", Value: nil}}
        update := bson.D{{Key: "$set", Value: bson.D{{Key: "last_used_at", Value: r.now()}}}}
        err := r.c.FindOneAndUpdate(ctx, filter, update, &out)
        if errors.Is(err, store.ErrNotFound) {
                return out, ErrNotFound
        }
        return out, err
}

// Revoke reasons the platform sets (PADR-0039).
const (
        ReasonOwnerDenied      = "owner_denied"
        ReasonOwnerRoleRemoved = "owner_role_removed"
)

// RevokeAllByOwner ends every active credential of owner, recording reason, and returns how many
// it ended. Each pass takes one credential, so a retry after a failure finishes the rest. This is
// the auth-event path, so it sets updated_by to the reason rather than a user.
func (r *Credentials) RevokeAllByOwner(ctx context.Context, owner, reason string) (int, error) {
        if strings.TrimSpace(owner) == "" || strings.TrimSpace(reason) == "" {
                return 0, fmt.Errorf("%w: owner and reason are required", ErrInvalid)
        }
        n := 0
        for {
                now := r.now()
                update := bson.D{
                        {Key: "$set", Value: bson.D{{Key: "revoked_at", Value: now}, {Key: "revoke_reason", Value: reason}, {Key: "updated_at", Value: now}, {Key: "updated_by", Value: reason}}},
                        {Key: "$inc", Value: bson.D{{Key: "revision", Value: 1}}},
                }
                var out IntegrationCredential
                err := r.c.FindOneAndUpdate(ctx, activeScope(owner), update, &out)
                if errors.Is(err, store.ErrNotFound) {
                        return n, nil
                }
                if err != nil {
                        return n, err
                }
                n++
        }
}

const gateID = "gate"

// GateClosed reports whether the persisted gate is closed. No document means open.
func (r *Credentials) GateClosed(ctx context.Context) (bool, error) {
        var doc struct {
                Closed bool `bson:"closed"`
        }
        err := r.gate.FindOne(ctx, bson.D{{Key: "_id", Value: gateID}}, &doc)
        if errors.Is(err, store.ErrNotFound) {
                return false, nil
        }
        return doc.Closed, err
}

// SetGate persists the gate state so a restart cannot reopen it before a reconcile.
func (r *Credentials) SetGate(ctx context.Context, closed bool, actor string) error {
        col := r.gate
        filter := bson.D{{Key: "_id", Value: gateID}}
        update := bson.D{{Key: "$set", Value: bson.D{{Key: "closed", Value: closed}, {Key: "updated_at", Value: r.now()}, {Key: "updated_by", Value: actor}}}}
        var out bson.M
        err := col.FindOneAndUpdate(ctx, filter, update, &out)
        if !errors.Is(err, store.ErrNotFound) {
                return err
        }
        err = col.InsertOne(ctx, bson.D{{Key: "_id", Value: gateID}, {Key: "closed", Value: closed}, {Key: "updated_at", Value: r.now()}, {Key: "updated_by", Value: actor}})
        if errors.Is(err, store.ErrDuplicate) {
                return col.FindOneAndUpdate(ctx, filter, update, &out)
        }
        return err
}

// Decision is a granted authorization for one user on one session (PADR-0035). Every session data
// function takes one and has no unchecked form: the only way to get a granted Decision is
// Authorize, which asks auth-api (OpenFGA) with the caller's own bearer token. The zero value
// grants nothing.
type Decision struct {
        actor, session, relation string
}

// Actor returns the user the decision was made for.
func (d Decision) Actor() string { return d.actor }

// SessionID returns the session the decision covers.
func (d Decision) SessionID() string { return d.session }

// Relations on a session, weakest first. The model (auth-api fga/model.fga) makes can_edit and
// can_approve the same set (gm or co_gm), gm a subset of that, and can_view everyone.
const (
        relView    = "can_view"
        relEdit    = "can_edit"
        relApprove = "can_approve"
        relGM      = "gm"
)

var relationRank = map[string]int{relView: 1, relEdit: 2, relApprove: 2, relGM: 3}

// require refuses the zero Decision and any decision made for a weaker relation than min, so a
// can_view grant (any observer) cannot be spent on a write that needs can_edit.
func (d Decision) require(min string) error {
        if d.actor == "" || d.session == "" || relationRank[d.relation] == 0 || relationRank[d.relation] < relationRank[min] {
                return ErrForbidden
        }
        return nil
}

// RelationChecker answers whether the holder of token has relation on object.
// *relations.Client implements it.
type RelationChecker interface {
        Check(ctx context.Context, user, token, object, relation string) (bool, error)
}

type gate struct {
        checker RelationChecker
        token   string
}

type gateKey struct{}

// WithGate puts the checker and the caller's bearer token on ctx for Authorize. Per PADR-0037 the
// token is the caller's own; there is no service token.
func WithGate(ctx context.Context, c RelationChecker, token string) context.Context {
        return context.WithValue(ctx, gateKey{}, gate{c, token})
}

// Authorize asks whether actor holds relation on the session. A malformed session id is
// ErrNotFound; a denial is ErrForbidden and carries no session data. When no decision can be
// obtained (no gate on ctx, auth-api or OpenFGA down) the error is relations.ErrUnavailable, or
// relations.ErrUnauthorized for a rejected token: session data fails closed.
func Authorize(ctx context.Context, actor, sessionID, relation string) (Decision, error) {
        if err := checkUserID(actor); err != nil {
                return Decision{}, err
        }
        id, ok := cleanID(sessionID)
        if !ok {
                return Decision{}, ErrNotFound
        }
        g, ok := ctx.Value(gateKey{}).(gate)
        if !ok || g.checker == nil {
                return Decision{}, relations.ErrUnavailable
        }
        allowed, err := g.checker.Check(ctx, actor, g.token, "session:"+id, relation)
        if err != nil {
                return Decision{}, err
        }
        if !allowed {
                return Decision{}, ErrForbidden
        }
        return Decision{actor: actor, session: id, relation: relation}, nil
}

var (

        tieBreakers = []string{TieRandom, TieAlphabetical, TieQuery}
        statuses    = []string{StatusDraft, StatusActive, StatusEnded}

// Encounters is the owner-scoped encounter repository. It validates enumerated values and
// queue references before any write reaches the collection.
type Encounters struct {
        *Owned[Encounter, *Encounter]
        systems *SystemRepo
}

// NewEncounters builds the encounter repository for db.
func NewEncounters(db store.Database) *Encounters {
        return &Encounters{Owned: NewOwned[Encounter](db, EncounterSpec), systems: NewSystemRepo(db)}
}

// ValidateEncounter checks a whole encounter document: ordering, tie breaker, and status come
// from their constants, and every turn_queue entry names one of the encounter's combatants.
func ValidateEncounter(e *Encounter) error {
        if !slices.Contains(orderings, e.Ordering) {
                return fmt.Errorf("%w: ordering %q", ErrInvalid, e.Ordering)
        }
        if !slices.Contains(tieBreakers, e.TieBreaker) {
                return fmt.Errorf("%w: tie_breaker %q", ErrInvalid, e.TieBreaker)
        }
        return ValidateState(EncounterState{Status: e.Status, Round: e.Round,
                CurrentCombatantID: e.CurrentCombatantID, TurnQueue: e.TurnQueue, Combatants: e.Combatants})
}

// EncounterState is the live combat state of an encounter, written as one unit.
type EncounterState struct {
        Status             string
        Round              int
        CurrentCombatantID string
        TurnQueue          []string
        Combatants         []Combatant
}

// ValidateState checks status, a non-negative round, unique combatant IDs, and that the queue and
// the current combatant refer to combatants of this state.
func ValidateState(st EncounterState) error {
        if !slices.Contains(statuses, st.Status) {
                return fmt.Errorf("%w: status %q", ErrInvalid, st.Status)
        }
        if st.Round < 0 {
                return fmt.Errorf("%w: round %d", ErrInvalid, st.Round)
        }
        ids := make(map[string]bool, len(st.Combatants))
        for _, c := range st.Combatants {
                if err := validateCombatant(c); err != nil {
                        return err
                }
                if ids[c.ID] {
                        return fmt.Errorf("%w: duplicate combatant id %q", ErrInvalid, c.ID)
                }
                ids[c.ID] = true
        }
        if st.CurrentCombatantID != "" && !ids[st.CurrentCombatantID] {
                return fmt.Errorf("%w: current combatant %q is not a combatant", ErrInvalid, st.CurrentCombatantID)
        }
        for _, id := range st.TurnQueue {
                if !ids[id] {
                        return fmt.Errorf("%w: turn_queue entry %q is not a combatant", ErrInvalid, id)
                }
        }
        return nil
}

func validateEncounterSet(set map[string]any) error {
        if v, ok := set["game_system_id"]; ok {
                if s, isStr := v.(string); !isStr || strings.TrimSpace(s) == "" {
                        return fmt.Errorf("%w: game_system_id must be a non-empty string", ErrInvalid)
                }
        }
        if v, ok := set["ordering"]; ok {
                if s, isStr := v.(string); !isStr || !slices.Contains(orderings, s) {
                        return fmt.Errorf("%w: ordering %v", ErrInvalid, v)
                }
        }
        if v, ok := set["tie_breaker"]; ok {
                if s, isStr := v.(string); !isStr || !slices.Contains(tieBreakers, s) {
                        return fmt.Errorf("%w: tie_breaker %v", ErrInvalid, v)
                }
        }
        return nil
}

// Create resets e's combat state to a fresh draft, validates it, then stores it as owner's new
// encounter. Callers cannot create an encounter that is already running.
func (r *Encounters) Create(ctx context.Context, ownerUserID, actingUserID string, e *Encounter) error {
        e.Status, e.Round, e.CurrentCombatantID, e.TurnQueue = StatusDraft, 0, "", nil
        if err := ValidateEncounter(e); err != nil {
                return err
        }
        if err := r.checkHealth(ctx, e.GameSystemID, e.Combatants, nil); err != nil {
                return err
        }
        e.RefKeys = refKeys(e.Combatants)
        return r.Owned.Create(ctx, ownerUserID, actingUserID, e)
}

// Update validates the enumerated values in set, then applies it with the revision check. A change
// of game system also checks every combatant's health against the new system.
func (r *Encounters) Update(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, set map[string]any) (Encounter, error) {
        if err := validateEncounterSet(set); err != nil {
                return Encounter{}, err
        }
        if sysID, ok := set["game_system_id"].(string); ok {
                if err := r.requireSystem(ctx, sysID); err != nil {
                        return Encounter{}, err
                }
                // Every combatant's health must fit the new system before the switch is stored.
                cur, err := r.Load(ctx, ownerUserID, id)
                if err != nil {
                        return Encounter{}, err
                }
                if err := r.checkHealth(ctx, sysID, cur.Combatants, nil); err != nil {
                        return Encounter{}, err
                }
        }
        return r.Owned.Update(ctx, ownerUserID, actingUserID, id, expectedRevision, set)
}

func validateCombatant(c Combatant) error {
        if strings.TrimSpace(c.Name) == "" {
                return fmt.Errorf("%w: combatant %q has no name", ErrInvalid, c.ID)
        }
        if !slices.Contains(Kinds, c.Kind) {
                return fmt.Errorf("%w: combatant %q kind %q", ErrInvalid, c.ID, c.Kind)
        }
        for _, cond := range c.Conditions {
                if strings.TrimSpace(cond.Name) == "" || (cond.RoundsRemaining != nil && *cond.RoundsRemaining < 1) {
                        return fmt.Errorf("%w: combatant %q has an invalid condition", ErrInvalid, c.ID)
                }
        }
        return nil
}

// requireSystem rejects a game system that has no definition. The No System fallback applies only
// to an encounter whose existing system lost its definition, never to choosing one: creating an
// encounter and switching its system both require the definition to exist.
func (r *Encounters) requireSystem(ctx context.Context, sysID string) error {
        _, err := r.systems.Get(ctx, sysID)
        if errors.Is(err, ErrNotFound) {
                return invalid("the game system is not defined")
        }
        return err
}

// checkHealth validates the health of each combatant in cs against the encounter's game system
// definition (the default system when the encounter's has none). Only combatants that are new or
// whose health differs from existing are checked, so a stored value that a later definition change
// made invalid does not block unrelated edits. The system is looked up only when something needs
// checking.
func (r *Encounters) checkHealth(ctx context.Context, sysID string, cs, existing []Combatant) error {
        before := make(map[string]map[string]int, len(existing))
        for _, c := range existing {
                before[c.ID] = c.Health
        }
        var changed []Combatant
        for _, c := range cs {
                if prev, ok := before[c.ID]; len(c.Health) > 0 && (!ok || !maps.Equal(prev, c.Health)) {
                        changed = append(changed, c)
                }
        }
        if len(changed) == 0 {
                return nil
        }
        sys, err := r.systems.GetOrDefault(ctx, sysID)
        if errors.Is(err, ErrNotFound) {
                return invalid("the encounter's game system has no definition and no default system is defined")
        }
        if err != nil {
                return err
        }
        for _, c := range changed {
                if err := CheckHealth(sys, c.Health); err != nil {
                        return err
                }
        }
        return nil
}

// SetState validates st and writes it as the encounter's combat state with the revision check.
func (r *Encounters) SetState(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, st EncounterState) (Encounter, error) {
        return r.Write(ctx, ownerUserID, actingUserID, id, expectedRevision, nil, st)
}

// Write validates settings (Updatable fields) and st (structure, and combatant health against the
// encounter's game system), then writes both in one atomic write with the revision check. Use it
// when a setting change, such as the ordering, comes with recomputed state.
func (r *Encounters) Write(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, settings map[string]any, st EncounterState) (Encounter, error) {
        if err := validateEncounterSet(settings); err != nil {
                return Encounter{}, err
        }
        if err := ValidateState(st); err != nil {
                return Encounter{}, err
        }
        if sysID, ok := settings["game_system_id"].(string); ok {
                if err := r.requireSystem(ctx, sysID); err != nil {
                        return Encounter{}, err
                }
        }
        if slices.ContainsFunc(st.Combatants, func(c Combatant) bool { return len(c.Health) > 0 }) {
                cur, err := r.Load(ctx, ownerUserID, id)
                if err != nil {
                        return Encounter{}, err
                }
                sysID := cur.GameSystemID
                if v, ok := settings["game_system_id"].(string); ok {
                        sysID = v
                }
                existing := cur.Combatants
                if _, switched := settings["game_system_id"]; switched {
                        existing = nil // a new system: every combatant's health is checked afresh
                }
                if err := r.checkHealth(ctx, sysID, st.Combatants, existing); err != nil {
                        return Encounter{}, err
                }
        }
        state := map[string]any{
                "status": st.Status, "round": st.Round, "current_combatant_id": st.CurrentCombatantID,
                "turn_queue": nonNil(st.TurnQueue), "combatants": nonNilC(st.Combatants),
                "ref_keys": nonNil(refKeys(st.Combatants)),
        }
        return r.updateBoth(ctx, ownerUserID, actingUserID, id, expectedRevision, settings, state)
}

func nonNil(s []string) []string {
        if s == nil {
                return []string{}
        }
        return s
}

func nonNilC(s []Combatant) []Combatant {
        if s == nil {
                return []Combatant{}
        }
        return s
}

// Load reads owner's encounter; a miss is classified 403 (another owner's) or 404 (absent).
func (r *Encounters) Load(ctx context.Context, owner, id string) (Encounter, error) {
        cur, err := r.Get(ctx, owner, id)
        if errors.Is(err, ErrNotFound) {
                if cerr := r.Missing(ctx, owner, id); cerr != nil {
                        return Encounter{}, cerr
                }
        }
        return cur, err
}

const (
        maxRefLen     = 200
        maxRefMatches = 100
)

// RefKey is the fixed-shape lookup key for one external reference: the hex SHA-256 of the
// length-prefixed key and value. Lookups filter on this alone, so request text never becomes part
// of a query filter.
func RefKey(key, value string) string {
        sum := sha256.Sum256([]byte(fmt.Sprintf("%d:%s%s", len(key), key, value)))
        return hex.EncodeToString(sum[:])
}

// refKeys derives Encounter.RefKeys from the live (not removed) combatants.
func refKeys(cs []Combatant) []string {
        var out []string
        for _, c := range cs {
                if c.Removed {
                        continue
                }
                for k, v := range c.ExternalRefs {
                        out = append(out, RefKey(k, v))
                }
        }
        slices.Sort(out)
        return slices.Compact(out)
}

// RefMatch is a combatant whose external_refs hold a looked-up pair, with its encounter's ID.
type RefMatch struct {
        EncounterID string    `json:"encounter_id"`
        Combatant   Combatant `json:"combatant"`
}

// FindByExternalRef returns owner's live combatants whose external_refs map key to value. The
// database matches on the derived RefKey; the stored values are then compared exactly, so a hash
// collision cannot return a wrong combatant.
func (r *Encounters) FindByExternalRef(ctx context.Context, owner, key, value string) ([]RefMatch, error) {
        if err := checkUserID(owner); err != nil {
                return nil, err
        }
        if key == "" || value == "" || len(key) > maxRefLen || len(value) > maxRefLen {
                return nil, invalid("ref_key and ref_value are required, at most %d bytes", maxRefLen)
        }
        filter := append(scope(owner), bson.E{Key: "ref_keys", Value: RefKey(key, value)})
        var encs []Encounter
        if err := r.c.Find(ctx, filter, store.FindOptions{Limit: maxRefMatches}, &encs); err != nil {
                return nil, err
        }
        var out []RefMatch
        for _, e := range encs {
                for _, c := range e.Combatants {
                        if !c.Removed && c.ExternalRefs[key] == value {
                                out = append(out, RefMatch{EncounterID: e.ID, Combatant: c})
                        }
                }
        }
        return out, nil
}

// LockGated locks owner's encounters that use a game system other than No System, and returns how
// many it locked. A locked encounter keeps all its data; only writes stop (spec
// initiative-game-systems, "Losing the entitlement locks gated-system items"). Each pass takes one
// encounter, so a retry after a failure finishes the rest.
func (r *Encounters) LockGated(ctx context.Context, owner string) (int, error) {
        filter := append(scope(owner),
                bson.E{Key: "game_system_id", Value: bson.D{{Key: "$ne", Value: DefaultSystemID}}},
                bson.E{Key: "locked_at", Value: nil})
        return r.flip(ctx, filter, bson.D{{Key: "$set", Value: bson.D{{Key: "locked_at", Value: r.now()}}}})
}

// UnlockAll restores owner's locked encounters exactly as they were.
func (r *Encounters) UnlockAll(ctx context.Context, owner string) (int, error) {
        filter := append(scope(owner), bson.E{Key: "locked_at", Value: bson.D{{Key: "$ne", Value: nil}}})
        return r.flip(ctx, filter, bson.D{{Key: "$unset", Value: bson.D{{Key: "locked_at", Value: ""}}}})
}

func (r *Encounters) flip(ctx context.Context, filter, update bson.D) (int, error) {
        n := 0
        for {
                var out Encounter
                err := r.c.FindOneAndUpdate(ctx, filter, update, &out)
                if errors.Is(err, store.ErrNotFound) {
                        return n, nil
                }
                if err != nil {
                        return n, err
                }
                n++
        }
}

// HealthError reports the health field that failed validation. It wraps ErrInvalid.
type HealthError struct {
        Field  string
        Reason string
}

func (e *HealthError) Error() string { return fmt.Sprintf("health field %q: %s", e.Field, e.Reason) }

// Unwrap lets errors.Is match ErrInvalid.
func (e *HealthError) Unwrap() error { return ErrInvalid }

// CheckHealth validates h against sys's tracker definition: every key must be a defined field, a
// flag must be 0 or 1, and each value must be at least the field's Min and at most its Max (or,
// for MaxRef, the value of the referenced field in h when present). Keys are checked in sorted
// order so the reported field is stable.
func CheckHealth(sys GameSystem, h map[string]int) error {
        fields := make(map[string]TrackerField, len(sys.Fields))
        for _, f := range sys.Fields {
                fields[f.Key] = f
        }
        for _, k := range slices.Sorted(maps.Keys(h)) {
                v := h[k]
                f, ok := fields[k]
                if !ok {
                        return &HealthError{k, "not defined by the game system"}
                }
                if f.Kind == FieldFlag && v != 0 && v != 1 {
                        return &HealthError{k, fmt.Sprintf("%d is not 0 or 1", v)}
                }
                if f.Min != nil && v < *f.Min {
                        return &HealthError{k, fmt.Sprintf("%d is below the minimum %d", v, *f.Min)}
                }
                if f.Max != nil && v > *f.Max {
                        return &HealthError{k, fmt.Sprintf("%d is above the maximum %d", v, *f.Max)}
                }
                if ref, ok := h[f.MaxRef]; f.MaxRef != "" && ok && v > ref {
                        return &HealthError{k, fmt.Sprintf("%d is above %s (%d)", v, f.MaxRef, ref)}
                }
        }
        return nil
}

// ItemState is what limit enforcement needs to know about one stored item.
type ItemState struct {
        ID        string
        CreatedAt time.Time
        KeepRank  int
        Inactive  bool
}

// ItemStates lists owner's live items oldest first, for deciding which stay active.
func (r *Owned[T, P]) ItemStates(ctx context.Context, owner string) ([]ItemState, error) {
        if err := checkUserID(owner); err != nil {
                return nil, err
        }
        return r.itemStates(ctx, scope(owner))
}

func (r *Owned[T, P]) itemStates(ctx context.Context, filter bson.D) ([]ItemState, error) {
        var docs []T
        opts := store.FindOptions{
                Sort:       bson.D{{Key: "created_at", Value: 1}, {Key: "_id", Value: 1}},
                Projection: bson.D{{Key: "created_at", Value: 1}, {Key: "keep_rank", Value: 1}, {Key: "inactive_at", Value: 1}},
        }
        if err := r.c.Find(ctx, filter, opts, &docs); err != nil {
                return nil, err
        }
        out := make([]ItemState, 0, len(docs))
        for i := range docs {
                m := P(&docs[i]).GetMeta()
                out = append(out, ItemState{ID: m.ID, CreatedAt: m.CreatedAt, KeepRank: m.KeepRank, Inactive: m.InactiveAt != nil})
        }
        return out, nil
}

// SetInactive marks owner's item inactive or active. It changes neither the revision nor the
// audit stamp: the owner did not edit the item.
func (r *Owned[T, P]) SetInactive(ctx context.Context, owner, id string, inactive bool) error {
        if err := checkUserID(owner); err != nil {
                return err
        }
        id, ok := cleanID(id)
        if !ok {
                return ErrNotFound
        }
        var update bson.D
        if inactive {
                update = bson.D{{Key: "$set", Value: bson.D{{Key: "inactive_at", Value: r.now()}}}}
        } else {
                update = bson.D{{Key: "$unset", Value: bson.D{{Key: "inactive_at", Value: ""}}}}
        }
        var discard T
        err := r.c.FindOneAndUpdate(ctx, scopeID(owner, id), update, &discard)
        if errors.Is(err, store.ErrNotFound) {
                return ErrNotFound
        }
        return err
}

// SetKeepRanks records the owner's choice: ids get ranks 1..n in order, every other item loses
// its rank. Items already in that state are not written.
func (r *Owned[T, P]) SetKeepRanks(ctx context.Context, owner string, ids []string) error {
        states, err := r.ItemStates(ctx, owner)
        if err != nil {
                return err
        }
        want := make(map[string]int, len(ids))
        for i, id := range ids {
                want[id] = i + 1
        }
        for _, st := range states {
                if st.KeepRank == want[st.ID] {
                        continue
                }
                update := bson.D{{Key: "$unset", Value: bson.D{{Key: "keep_rank", Value: ""}}}}
                if n := want[st.ID]; n > 0 {
                        update = bson.D{{Key: "$set", Value: bson.D{{Key: "keep_rank", Value: n}}}}
                }
                var discard T
                if err := r.c.FindOneAndUpdate(ctx, scopeID(owner, st.ID), update, &discard); err != nil && !errors.Is(err, store.ErrNotFound) {
                        return fmt.Errorf("set keep rank: %w", err)
                }
        }
        return nil
}

}

type Doc[T any] interface {
        *T
        GetMeta() *Meta
}

// Owned is the owner-scoped repository for one collection of user-owned documents. Every method
// requires the owner's user ID and folds it, plus the soft-delete filter, into the query itself.
// There is deliberately no method that reads or writes by ID alone.
type Owned[T any, P Doc[T]] struct {
        c     store.Collection
        spec  Spec
        now   func() time.Time
        newID func() string
}

// NewOwned builds the repository for spec's collection in db.
func NewOwned[T any, P Doc[T]](db store.Database, spec Spec) *Owned[T, P] {
        return &Owned[T, P]{
                c:     db.Collection(spec.Name),
                spec:  spec,
                now:   func() time.Time { return time.Now().UTC() },
                newID: func() string { return primitive.NewObjectID().Hex() },
        }
}

// scope is the filter every owner-scoped query starts from.
func scope(owner string) bson.D {
        return bson.D{{Key: "owner_user_id", Value: owner}, {Key: "deleted_at", Value: nil}}
}

// scopeID is the owner-scoped filter for one record. id must already have passed cleanID.
func scopeID(owner, id string) bson.D {
        return append(bson.D{{Key: "_id", Value: id}}, scope(owner)...)
}

// cleanID returns the canonical form of a client-supplied record ID, or false when it is not a
// 24-character hex ObjectID. Every ID that reaches a query filter goes through here, so a request
// value can only ever be a plain ObjectID string, never an operator document or arbitrary text.
func cleanID(id string) (string, bool) {
        oid, err := primitive.ObjectIDFromHex(id)
        if err != nil {
                return "", false
        }
        return oid.Hex(), true
}

// requireOwnerActor refuses a write with no valid owner or acting user id.
func requireOwnerActor(owner, actor string) error {
        if err := checkUserID(owner); err != nil {
                return err
        }
        return checkUserID(actor)
}

// Create stores doc as owner's new record: it assigns the ID, owner, revision 1, and the audit
// stamp (created_by and updated_by are actor). Any ID or owner already on doc is overwritten.
func (r *Owned[T, P]) Create(ctx context.Context, ownerUserID, actingUserID string, doc P) error {
        return r.createWithID(ctx, ownerUserID, actingUserID, "", doc)
}

// createWithID is Create with a caller-chosen ID; an empty id means generate one.
func (r *Owned[T, P]) createWithID(ctx context.Context, ownerUserID, actingUserID, id string, doc P) error {
        if err := requireOwnerActor(ownerUserID, actingUserID); err != nil {
                return err
        }
        m := doc.GetMeta()
        m.ID = id
        if id == "" {
                m.ID = r.newID()
        }
        m.OwnerUserID = ownerUserID
        m.Revision = 1
        m.DeletedAt, m.DeletedBy = nil, nil
        models.StampCreate(&m.Auditable, actingUserID, r.now())
        return r.mapDuplicate(r.c.InsertOne(ctx, doc))
}

// mapDuplicate turns a unique-index rejection into ErrConflict for specs that opt in.
func (r *Owned[T, P]) mapDuplicate(err error) error {
        if r.spec.ConflictOnDuplicate && errors.Is(err, store.ErrDuplicate) {
                return ErrConflict
        }
        return err
}

// Get returns owner's live record. Another owner's record is indistinguishable from a missing
// one on reads.
func (r *Owned[T, P]) Get(ctx context.Context, ownerUserID, id string) (T, error) {
        var out T
        if err := checkUserID(ownerUserID); err != nil {
                return out, err
        }
        id, ok := cleanID(id)
        if !ok {
                return out, ErrNotFound
        }
        err := r.c.FindOne(ctx, scopeID(ownerUserID, id), &out)
        if errors.Is(err, store.ErrNotFound) {
                return out, ErrNotFound
        }
        return out, err
}

// reservedFields are managed by the data layer and rejected in Update's set.
var reservedFields = map[string]bool{
        "_id": true, "owner_user_id": true, "revision": true,
        "created_at": true, "created_by": true, "updated_at": true, "updated_by": true,
        "deleted_at": true, "deleted_by": true,
}

func validateSet(spec Spec, set map[string]any) error {
        if len(set) == 0 {
                return fmt.Errorf("%w: nothing to update", ErrInvalid)
        }
        for k := range set {
                if reservedFields[k] || strings.HasPrefix(k, "$") || strings.Contains(k, ".") || k == "" {
                        return fmt.Errorf("%w: field %q cannot be set", ErrInvalid, k)
                }
                if !in(spec.Updatable, k) {
                        return fmt.Errorf("%w: field %q is not updatable on %s", ErrInvalid, k, spec.Name)
                }
        }
        return nil
}

func setDoc(set map[string]any, extra ...bson.E) bson.D {
        keys := make([]string, 0, len(set))
        for k := range set {
                keys = append(keys, k)
        }
        sort.Strings(keys)
        d := make(bson.D, 0, len(keys)+len(extra))
        for _, k := range keys {
                d = append(d, bson.E{Key: k, Value: set[k]})
        }
        return append(d, extra...)
}

// Update sets fields on owner's record if its revision is still expectedRevision, bumping the
// revision and stamping updated_*. The match is one atomic write on {_id, owner, live, revision},
// so a stale revision or another owner's record modifies nothing. When nothing matched, the
// cause is classified afterwards: ErrNotFound, ErrForbidden, or ErrRevisionConflict (with the
// current record returned).
func (r *Owned[T, P]) Update(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, set map[string]any) (T, error) {
        var out T
        if err := requireOwnerActor(ownerUserID, actingUserID); err != nil {
                return out, err
        }
        if err := validateSet(r.spec, set); err != nil {
                return out, err
        }
        return r.apply(ctx, ownerUserID, actingUserID, id, expectedRevision, set)
}

// updateState is Update for the spec's StateFields: values computed by domain rules, never taken
// from client input. Everything else about the write is identical (owner-scoped, compare-and-set,
// classified failures).
func (r *Owned[T, P]) updateState(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, set map[string]any) (T, error) {
        var out T
        if err := requireOwnerActor(ownerUserID, actingUserID); err != nil {
                return out, err
        }
        if len(set) == 0 {
                return out, fmt.Errorf("%w: nothing to update", ErrInvalid)
        }
        for k := range set {
                if !in(r.spec.StateFields, k) {
                        return out, fmt.Errorf("%w: field %q is not a state field of %s", ErrInvalid, k, r.spec.Name)
                }
        }
        return r.apply(ctx, ownerUserID, actingUserID, id, expectedRevision, set)
}

// updateBoth writes settings (Updatable fields) and state (StateFields) in one atomic,
// compare-and-set write, for changes where a setting and the state derived from it must land
// together. Either map may be empty, not both.
func (r *Owned[T, P]) updateBoth(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, settings, state map[string]any) (T, error) {
        var out T
        if err := requireOwnerActor(ownerUserID, actingUserID); err != nil {
                return out, err
        }
        merged := make(map[string]any, len(settings)+len(state))
        if len(settings) > 0 {
                if err := validateSet(r.spec, settings); err != nil {
                        return out, err
                }
                maps.Copy(merged, settings)
        }
        for k, v := range state {
                if !in(r.spec.StateFields, k) {
                        return out, fmt.Errorf("%w: field %q is not a state field of %s", ErrInvalid, k, r.spec.Name)
                }
                merged[k] = v
        }
        if len(merged) == 0 {
                return out, fmt.Errorf("%w: nothing to update", ErrInvalid)
        }
        return r.apply(ctx, ownerUserID, actingUserID, id, expectedRevision, merged)
}

func (r *Owned[T, P]) apply(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64, set map[string]any) (T, error) {
        var out T
        id, ok := cleanID(id)
        if !ok {
                return out, ErrNotFound
        }
        filter := append(scopeID(ownerUserID, id), bson.E{Key: "revision", Value: expectedRevision}, bson.E{Key: "inactive_at", Value: nil}, bson.E{Key: "locked_at", Value: nil})
        update := bson.D{
                {Key: "$set", Value: setDoc(set,
                        bson.E{Key: "updated_at", Value: r.now()}, bson.E{Key: "updated_by", Value: actingUserID})},
                {Key: "$inc", Value: bson.D{{Key: "revision", Value: 1}}},
        }
        err := r.c.FindOneAndUpdate(ctx, filter, update, &out)
        if errors.Is(err, store.ErrNotFound) {
                cur, cerr := r.classify(ctx, ownerUserID, id)
                if errors.Is(cerr, ErrRevisionConflict) {
                        var zero T
                        switch m := P(&cur).GetMeta(); {
                        case m.LockedAt != nil:
                                return zero, ErrLocked
                        case m.InactiveAt != nil:
                                return zero, ErrInactive
                        }
                }
                return cur, cerr
        }
        return out, r.mapDuplicate(err)
}

// SoftDelete tombstones owner's record if its revision is still expectedRevision (deleted_at/
// deleted_by, plus the spec's UnsetOnDelete fields), per PADR-0027 for user-owned domain data.
// Failures are classified like Update's, including ErrRevisionConflict.
func (r *Owned[T, P]) SoftDelete(ctx context.Context, ownerUserID, actingUserID, id string, expectedRevision int64) error {
        if err := requireOwnerActor(ownerUserID, actingUserID); err != nil {
                return err
        }
        now := r.now()
        update := bson.D{
                {Key: "$set", Value: bson.D{
                        {Key: "deleted_at", Value: now}, {Key: "deleted_by", Value: actingUserID},
                        {Key: "updated_at", Value: now}, {Key: "updated_by", Value: actingUserID}}},
                {Key: "$inc", Value: bson.D{{Key: "revision", Value: 1}}},
        }
        if len(r.spec.UnsetOnDelete) > 0 {
                unset := make(bson.D, 0, len(r.spec.UnsetOnDelete))
                for _, f := range r.spec.UnsetOnDelete {
                        unset = append(unset, bson.E{Key: f, Value: ""})
                }
                update = append(update, bson.E{Key: "$unset", Value: unset})
        }
        id, ok := cleanID(id)
        if !ok {
                return ErrNotFound
        }
        var discard T
        filter := append(scopeID(ownerUserID, id), bson.E{Key: "revision", Value: expectedRevision})
        err := r.c.FindOneAndUpdate(ctx, filter, update, &discard)
        if errors.Is(err, store.ErrNotFound) {
                _, err = r.classify(ctx, ownerUserID, id)
        }
        return err
}

// Missing explains why owner's scoped read of id found nothing: ErrNotFound when no live record
// exists, ErrForbidden when it belongs to another owner. Call it only after a scoped read or write
// has already failed; it never grants access. It returns nil if the record is visible to owner.
func (r *Owned[T, P]) Missing(ctx context.Context, owner, id string) error {
        if err := checkUserID(owner); err != nil {
                return err
        }
        if _, err := r.classify(ctx, owner, id); !errors.Is(err, ErrRevisionConflict) {
                return err
        }
        return nil
}

// classify runs only after an owner-scoped write matched nothing. It does an owner-agnostic
// lookup to tell 404 from 403 from 409; it never grants access.
func (r *Owned[T, P]) classify(ctx context.Context, owner, id string) (T, error) {
        var cur T
        id, ok := cleanID(id)
        if !ok {
                return cur, ErrNotFound
        }
        err := r.c.FindOne(ctx, bson.D{{Key: "_id", Value: id}}, &cur)
        if errors.Is(err, store.ErrNotFound) {
                return cur, ErrNotFound
        }
        if err != nil {
                return cur, err
        }
        m := P(&cur).GetMeta()
        switch {
        case m.DeletedAt != nil:
                var zero T
                return zero, ErrNotFound
        case m.OwnerUserID != owner:
                var zero T
                return zero, ErrForbidden
        }
        return cur, ErrRevisionConflict
}

package data

import (
        "context"
        "encoding/base64"
        "fmt"
        "regexp"

        "github.com/sweetrpg/initiative-api/store"
        "go.mongodb.org/mongo-driver/bson"
        "go.mongodb.org/mongo-driver/bson/primitive"
)

// List bounds.
const (
        DefaultLimit = 50
        MaxLimit     = 200
        maxInValues  = 100
        maxValueLen  = 128
)

// Op is a filter operator. Clients choose from this closed set; a client string never becomes an
// operator key or a live regular expression (PADR-0030).
type Op string

// Filter operators.
const (
        OpEq       Op = "eq"
        OpIn       Op = "in"
        OpContains Op = "contains"
)

// Filter restricts a list to records whose Field matches. eq takes one value, in takes up to
// maxInValues, contains takes one non-empty string and matches case-insensitively as a literal.
type Filter struct {
        Field  string
        Op     Op
        Values []any
}

// SortKey orders a list. _id is always appended as the final tie-break so paging is stable.
type SortKey struct {
        Field string
        Desc  bool
}

// Query is the query-layer shape of a list call: everything here is applied by the database.
// Limit 0 means DefaultLimit; larger than MaxLimit is clamped. Fields is an optional projection
// (empty returns whole documents). Cursor is the opaque NextCursor of the previous page.
type Query struct {
        Filters []Filter
        Sort    []SortKey
        Fields  []string
        Limit   int
        Cursor  string
}

// Page is one page of results; NextCursor is empty on the last page.
type Page[T any] struct {
        Items      []T
        NextCursor string
}

func invalid(format string, a ...any) error {
        return fmt.Errorf("%w: %s", ErrInvalid, fmt.Sprintf(format, a...))
}

func in(list []string, s string) bool {
        for _, x := range list {
                if x == s {
                        return true
                }
        }
        return false
}

// scalar admits only plain values, so a nested document can never smuggle in an operator.
func scalar(v any) (any, bool) {
        switch x := v.(type) {
        case string, bool, int64:
                return x, true
        case int:
                return int64(x), true
        case float64:
                if x == float64(int64(x)) {
                        return int64(x), true
                }
        }
        return nil, false
}

func filterClause(spec Spec, f Filter) (bson.D, error) {
        if !in(spec.Filterable, f.Field) {
                return nil, invalid("cannot filter by %q", f.Field)
        }
        vals := make(bson.A, 0, len(f.Values))
        for _, v := range f.Values {
                s, ok := scalar(v)
                if !ok {
                        return nil, invalid("filter %q has a non-scalar value", f.Field)
                }
                vals = append(vals, s)
        }
        switch f.Op {
        case OpEq:
                if len(vals) != 1 {
                        return nil, invalid("eq on %q needs exactly one value", f.Field)
                }
                return bson.D{{Key: f.Field, Value: vals[0]}}, nil
        case OpIn:
                if len(vals) == 0 || len(vals) > maxInValues {
                        return nil, invalid("in on %q needs 1 to %d values", f.Field, maxInValues)
                }
                return bson.D{{Key: f.Field, Value: bson.D{{Key: "$in", Value: vals}}}}, nil
        case OpContains:
                s, ok := firstString(vals)
                if len(vals) != 1 || !ok || s == "" || len(s) > maxValueLen {
                        return nil, invalid("contains on %q needs one non-empty string up to %d bytes", f.Field, maxValueLen)
                }
                return bson.D{{Key: f.Field, Value: bson.D{{Key: "$regex", Value: regexp.QuoteMeta(s)}, {Key: "$options", Value: "i"}}}}, nil
        }
        return nil, invalid("unknown operator %q", f.Op)
}

func firstString(v bson.A) (string, bool) {
        if len(v) == 0 {
                return "", false
        }
        s, ok := v[0].(string)
        return s, ok
}

func sortKeys(spec Spec, in2 []SortKey) ([]SortKey, error) {
        keys := append([]SortKey(nil), in2...)
        seen := map[string]bool{}
        for _, k := range keys {
                if k.Field != "_id" && !in(spec.Sortable, k.Field) {
                        return nil, invalid("cannot sort by %q", k.Field)
                }
                if seen[k.Field] {
                        return nil, invalid("sort field %q repeated", k.Field)
                }
                seen[k.Field] = true
        }
        if !seen["_id"] {
                keys = append(keys, SortKey{Field: "_id"})
        }
        return keys, nil
}

func projection(spec Spec, fields []string, keys []SortKey) (bson.D, error) {
        if len(fields) == 0 {
                return nil, nil
        }
        p := bson.D{}
        added := map[string]bool{"_id": true}
        for _, f := range fields {
                if !in(spec.Projectable, f) {
                        return nil, invalid("cannot project %q", f)
                }
                if !added[f] {
                        added[f] = true
                        p = append(p, bson.E{Key: f, Value: 1})
                }
        }
        for _, k := range keys { // the cursor needs the sort values of the last row
                if !added[k.Field] {
                        added[k.Field] = true
                        p = append(p, bson.E{Key: k.Field, Value: 1})
                }
        }
        return p, nil
}

type plan struct {
        filter bson.D
        opts   store.FindOptions
        keys   []SortKey
        limit  int
}

func buildPlan(spec Spec, base bson.D, q Query) (plan, error) {
        if q.Limit < 0 {
                return plan{}, invalid("limit must not be negative")
        }
        limit := q.Limit
        if limit == 0 {
                limit = DefaultLimit
        }
        limit = min(limit, MaxLimit)

        keys, err := sortKeys(spec, q.Sort)
        if err != nil {
                return plan{}, err
        }
        proj, err := projection(spec, q.Fields, keys)
        if err != nil {
                return plan{}, err
        }
        clauses, err := clausesFor(spec, base, q)
        if err != nil {
                return plan{}, err
        }
        if q.Cursor != "" {
                vals, err := decodeCursor(q.Cursor, len(keys))
                if err != nil {
                        return plan{}, err
                }
                clauses = append(clauses, after(keys, vals))
        }
        sortD := make(bson.D, 0, len(keys))
        for _, k := range keys {
                dir := 1
                if k.Desc {
                        dir = -1
                }
                sortD = append(sortD, bson.E{Key: k.Field, Value: dir})
        }
        // Fetch one extra row to learn whether another page exists.
        return plan{
                filter: filterOf(clauses),
                opts:   store.FindOptions{Sort: sortD, Projection: proj, Limit: int64(limit + 1)},
                keys:   keys,
                limit:  limit,
        }, nil
}

// filterOf ANDs clauses. MongoDB rejects an empty $and, so no clauses means an empty filter.
func filterOf(clauses bson.A) bson.D {
        if len(clauses) == 0 {
                return bson.D{}
        }
        return bson.D{{Key: "$and", Value: clauses}}
}

func clausesFor(spec Spec, base bson.D, q Query) (bson.A, error) {
        clauses := bson.A{}
        if len(base) > 0 {
                clauses = append(clauses, base)
        }
        for _, f := range q.Filters {
                c, err := filterClause(spec, f)
                if err != nil {
                        return nil, err
                }
                clauses = append(clauses, c)
        }
        return clauses, nil
}

// after builds the keyset condition for rows strictly after vals in sort order.
func after(keys []SortKey, vals []any) bson.D {
        ors := make(bson.A, 0, len(keys))
        for i := range keys {
                and := bson.D{}
                for j := 0; j < i; j++ {
                        and = append(and, bson.E{Key: keys[j].Field, Value: vals[j]})
                }
                op := "$gt"
                if keys[i].Desc {
                        op = "$lt"
                }
                and = append(and, bson.E{Key: keys[i].Field, Value: bson.D{{Key: op, Value: vals[i]}}})
                ors = append(ors, and)
        }
        return bson.D{{Key: "$or", Value: ors}}
}

func encodeCursor(vals []any) (string, error) {
        b, err := bson.MarshalExtJSON(bson.M{"v": bson.A(vals)}, true, false)
        return base64.RawURLEncoding.EncodeToString(b), err
}

func decodeCursor(c string, want int) ([]any, error) {
        b, err := base64.RawURLEncoding.DecodeString(c)
        if err != nil {
                return nil, invalid("malformed cursor")
        }
        var m bson.M
        if err := bson.UnmarshalExtJSON(b, true, &m); err != nil {
                return nil, invalid("malformed cursor")
        }
        arr, ok := m["v"].(primitive.A)
        if !ok || len(arr) != want {
                return nil, invalid("cursor does not match this query's sort")
        }
        for _, v := range arr {
                switch v.(type) {
                case string, int32, int64, float64, bool, primitive.DateTime:
                default:
                        return nil, invalid("malformed cursor")
                }
        }
        return arr, nil
}

// cursorFor encodes the sort values of the last row on a page.
func cursorFor(item any, keys []SortKey) (string, error) {
        raw, err := bson.Marshal(item)
        if err != nil {
                return "", err
        }
        var m bson.M
        if err := bson.Unmarshal(raw, &m); err != nil {
                return "", err
        }
        vals := make([]any, len(keys))
        for i, k := range keys {
                v, ok := m[k.Field]
                if !ok || v == nil {
                        return "", invalid("cannot page: sort field %q is missing on a record", k.Field)
                }
                vals[i] = v
        }
        return encodeCursor(vals)
}

// runQuery applies q entirely in the database: filter, sort, projection, limit, and keyset
// cursor. base is the mandatory scope (owner and live-record filters) every caller supplies.
func runQuery[T any](ctx context.Context, c store.Collection, spec Spec, base bson.D, q Query) (Page[T], error) {
        p, err := buildPlan(spec, base, q)
        if err != nil {
                return Page[T]{}, err
        }
        var items []T
        if err := c.Find(ctx, p.filter, p.opts, &items); err != nil {
                return Page[T]{}, err
        }
        if len(items) <= p.limit {
                return Page[T]{Items: items}, nil
        }
        items = items[:p.limit]
        next, err := cursorFor(items[p.limit-1], p.keys)
        if err != nil {
                return Page[T]{}, err
        }
        return Page[T]{Items: items, NextCursor: next}, nil
}

// List returns one page of owner's live records.
func (r *Owned[T, P]) List(ctx context.Context, ownerUserID string, q Query) (Page[T], error) {
        if err := checkUserID(ownerUserID); err != nil {
                return Page[T]{}, err
        }
        return runQuery[T](ctx, r.c, r.spec, scope(ownerUserID), q)
}

// Count returns how many of owner's live records match filters, counted by the database.
func (r *Owned[T, P]) Count(ctx context.Context, ownerUserID string, filters []Filter) (int64, error) {
        if err := checkUserID(ownerUserID); err != nil {
                return 0, err
        }
        clauses, err := clausesFor(r.spec, scope(ownerUserID), Query{Filters: filters})
        if err != nil {
                return 0, err
        }
        return r.c.CountDocuments(ctx, filterOf(clauses))
}

// Session roles.
const (
        RoleGM       = "gm"
        RoleCoGM     = "co_gm"
        RolePlayer   = "player"
        RoleObserver = "observer"
)

// Join request states.
const (
        RequestPending  = "pending"
        RequestApproved = "approved"
        RequestRejected = "rejected"
)

// Session is a named group owned by one GM (the owner).
type Session struct {
        Meta `bson:",inline"`
        Name string `bson:"name" json:"name"`
        // OwnerLimits holds the owner's entitled values for the session-related keys as last read with
        // the owner's own token, and OwnerLimitsAt when. Members' checks use these because a member's
        // token cannot read the owner's entitlements (initiative-entitlement-gating).
        OwnerLimits   map[string]int `bson:"owner_limits,omitempty" json:"-"`
        OwnerLimitsAt *time.Time     `bson:"owner_limits_at,omitempty" json:"-"`
}

// Member is one user's place in a session. This collection is the source of truth for who holds
// which role (initiative-session-authorization).
type Member struct {
        ID        string    `bson:"_id" json:"-"`
        SessionID string    `bson:"session_id" json:"session_id"`
        UserID    string    `bson:"user_id" json:"user_id"`
        Role      string    `bson:"role" json:"role"`

// Sessions is the repository for sessions and their membership records. The session document is
// owner-scoped (its owner is the GM); membership records are looked up by session, and the domain
// layer decides from them who may do what.
//
// The owner-scoped repository is not embedded: its Get, Update, List and the rest would be session
// reads and writes with no Decision (PADR-0035). Only Open, Count, SoftDelete and Limits expose it.
type Sessions struct {
        owned                              *Owned[Session, *Session]
        c                                  store.Collection
        members, invites, requests, blocks store.Collection
        now                                func() time.Time
}

// NewSessions builds the repository on db.
func NewSessions(db store.Database) *Sessions {
        o := NewOwned[Session](db, SessionSpec)
        return &Sessions{
                owned: o, c: o.c,
                members: db.Collection(CollectionSessionMembers), invites: db.Collection(CollectionSessionInvites),
                requests: db.Collection(CollectionSessionRequests), blocks: db.Collection(CollectionSessionBlocks),
                now: func() time.Time { return time.Now().UTC() },
        }
}

func newID() string { return primitive.NewObjectID().Hex() }

// Every read of a membership collection filters deleted_at (PADR-0027: these records are soft
// deleted), so the two scope helpers carry it.
func sessionScope(sessionID string) bson.D {
        return bson.D{{Key: "session_id", Value: sessionID}, {Key: "deleted_at", Value: nil}}
}

func sessionUser(sessionID, userID string) bson.D {
        return bson.D{{Key: "session_id", Value: sessionID}, {Key: "user_id", Value: userID}, {Key: "deleted_at", Value: nil}}
}

// softDelete marks the first live record matching filter deleted by actor and returns ErrNotFound
// when there is none.
func (r *Sessions) softDelete(ctx context.Context, c store.Collection, filter bson.D, actor string) error {
        var gone bson.M
        err := c.FindOneAndUpdate(ctx, filter, bson.D{
                {Key: "$set", Value: bson.D{{Key: "deleted_at", Value: r.now()}, {Key: "deleted_by", Value: actor}}},
                {Key: "$unset", Value: bson.D{{Key: "live", Value: ""}}},
        }, &gone)
        if errors.Is(err, store.ErrNotFound) {
                return ErrNotFound
        }
        return err
}

// pairOK validates the user id a call acts on.
func pairOK(d Decision, min, userID string) error {
        if err := d.require(min); err != nil {
                return err
        }
        return checkUserID(userID)
}

// checkUserID requires a plain user id, so an id from a request cannot reach a query as anything
// else.
func checkUserID(u string) error {
        if u == "" || !ownerPattern.MatchString(u) {
                return invalid("user id is not valid")
        }
        return nil
}

// Open creates the session owned by actor with the ID auth-api minted for them and records actor
// as its GM. Creating is not a session read: it is gated by the sessions_per_gm entitlement, which
// the caller checks first, and by the mint that put the GM tuple on id.
func (r *Sessions) Open(ctx context.Context, actor, id string, s *Session) error {
        id, ok := cleanID(id)
        if !ok {
                return invalid("session id is not valid")
        }
        if err := r.owned.createWithID(ctx, actor, actor, id, s); err != nil {
                return err
        }
        m := Member{ID: newID(), SessionID: s.ID, UserID: actor, Role: RoleGM, CreatedAt: r.now(), CreatedBy: actor, Live: true}
        if err := r.members.InsertOne(ctx, m); err != nil {
                // Without its GM the session could never be deleted, and it would count against
                // sessions_per_gm. Remove it; if even that fails, say so, since the caller ends the mint.
                if derr := r.owned.SoftDelete(ctx, actor, actor, s.ID, s.Revision); derr != nil {
                        return fmt.Errorf("%w (and removing the session failed: %v)", err, derr)
                }
                return err
        }
        return nil
}

// Count returns how many sessions owner has, including inactive ones.
func (r *Sessions) Count(ctx context.Context, owner string) (int64, error) {
        return r.owned.Count(ctx, owner, nil)
}

// SoftDelete deletes the session at expectedRevision. Only the GM may.
func (r *Sessions) SoftDelete(ctx context.Context, d Decision, expectedRevision int64) error {
        if err := d.require(relGM); err != nil {
                return err
        }
        return r.owned.SoftDelete(ctx, d.actor, d.actor, d.session, expectedRevision)
}

// JoinTicket is an invite the data layer has resolved and validated (known, live, not revoked, not
// expired) for one user. It is the gate for joining (initiative-sessions, "Join by code") and can
// only come from ResolveInvite, so the join calls cannot be aimed at an arbitrary session. The zero
// value is refused.
type JoinTicket struct{ actor, session string }

// ResolveInvite turns a presented code's hash into a ticket for actor, judging expiry at now. An unknown, deleted,
// revoked, or expired invite is ErrNotFound, whatever the reason.
func (r *Sessions) ResolveInvite(ctx context.Context, actor, codeHash string, now time.Time) (JoinTicket, error) {
        if err := checkUserID(actor); err != nil {
                return JoinTicket{}, err
        }
        if codeHash == "" {
                return JoinTicket{}, ErrNotFound
        }
        var in Invite
        err := r.invites.FindOne(ctx, bson.D{{Key: "code_hash", Value: codeHash}, {Key: "deleted_at", Value: nil}}, &in)
        if errors.Is(err, store.ErrNotFound) {
                return JoinTicket{}, ErrNotFound
        }
        if err != nil {
                return JoinTicket{}, err
        }
        if in.RevokedAt != nil || !now.Before(in.ExpiresAt) {
                return JoinTicket{}, ErrNotFound
        }
        return JoinTicket{actor: actor, session: in.SessionID}, nil
}

func (t JoinTicket) check() error {
        if t.actor == "" || t.session == "" {
                return ErrForbidden
        }
        return nil
}

}

// RevokeInvite ends an invite; revoking twice is harmless.
func (r *Sessions) RevokeInvite(ctx context.Context, d Decision, inviteID string) (Invite, error) {
        var in Invite
        if err := d.require(relEdit); err != nil {
                return in, err
        }
        id, ok := cleanID(inviteID)
        if !ok {
                return in, ErrNotFound
        }
        filter := append(bson.D{{Key: "_id", Value: id}}, sessionScope(d.session)...)
        err := r.invites.FindOne(ctx, filter, &in)
        if errors.Is(err, store.ErrNotFound) {
                return in, ErrNotFound
        }
        if err != nil || in.RevokedAt != nil {
                return in, err
        }
        err = r.invites.FindOneAndUpdate(ctx, append(filter, bson.E{Key: "revoked_at", Value: nil}),
                bson.D{{Key: "$set", Value: bson.D{{Key: "revoked_at", Value: r.now()}, {Key: "revoked_by", Value: d.actor}}}}, &in)
        if errors.Is(err, store.ErrNotFound) { // revoked concurrently
                return in, r.invites.FindOne(ctx, filter, &in)
        }
        return in, err
}

// JoinState is what a presented invite lets its holder learn: whether the session still exists
// and whether they are blocked or already a member. It carries no session data.
type JoinState struct {
        Exists, Active, Blocked, Member bool
}

// ForJoin reads the join state for the ticket holder. The ticket is the gate (initiative-sessions,
// "Join by code"), and the pending request created from it grants nothing.
func (r *Sessions) ForJoin(ctx context.Context, t JoinTicket) (JoinState, error) {
        var st JoinState
        if err := t.check(); err != nil {
                return st, err
        }
        actor := t.actor
        var sess Session
        err := r.c.FindOne(ctx, bson.D{{Key: "_id", Value: t.session}, {Key: "deleted_at", Value: nil}}, &sess)
        if errors.Is(err, store.ErrNotFound) {
                return st, nil
        }
        if err != nil {
                return st, err
        }
        st.Exists, st.Active = true, sess.InactiveAt == nil
        var b Block
        switch err := r.blocks.FindOne(ctx, sessionUser(t.session, actor), &b); {
        case err == nil:
                st.Blocked = true
        case !errors.Is(err, store.ErrNotFound):
                return st, err
        }
        var m Member
        switch err := r.members.FindOne(ctx, sessionUser(t.session, actor), &m); {
        case err == nil:
                st.Member = true
        case !errors.Is(err, store.ErrNotFound):
                return st, err
        }
        return st, nil
}

// RequestJoin records a pending request from the ticket holder for the ticket's session. An earlier settled
// request becomes pending again (the caller has checked the user is not a member, so an approved
// one means the approval never finished); an existing pending one is returned unchanged.
func (r *Sessions) RequestJoin(ctx context.Context, t JoinTicket) (JoinRequest, error) {
        var jr JoinRequest
        if err := t.check(); err != nil {
                return jr, err
        }
        actor, sessionID := t.actor, t.session
        err := r.requests.FindOne(ctx, sessionUser(sessionID, actor), &jr)
        switch {
        case errors.Is(err, store.ErrNotFound):
                jr = JoinRequest{ID: newID(), SessionID: sessionID, UserID: actor, Status: RequestPending, CreatedAt: r.now(), Live: true}
                return jr, r.requests.InsertOne(ctx, jr)
        case err != nil:
                return jr, err
        case jr.Status == RequestPending:
                return jr, nil
        }
        // Decode into a fresh value: decoding over jr would keep the fields the update removed.
        var reset JoinRequest
        err = r.requests.FindOneAndUpdate(ctx, sessionUser(sessionID, actor), bson.D{
                {Key: "$set", Value: bson.D{{Key: "status", Value: RequestPending}, {Key: "created_at", Value: r.now()}}},
                {Key: "$unset", Value: bson.D{{Key: "reason", Value: ""}, {Key: "role", Value: ""}, {Key: "decided_by", Value: ""}, {Key: "decided_at", Value: ""}}},
        }, &reset)
        return reset, err
}

}

// Decide settles a pending request as approved (with role) or rejected (with reason). A request
// already settled is ErrConflict.
func (r *Sessions) Decide(ctx context.Context, d Decision, requestID, status, role, reason string) (JoinRequest, error) {
        var jr JoinRequest
        if err := d.require(relApprove); err != nil {
                return jr, err
        }
        id, ok := cleanID(requestID)
        if !ok {
                return jr, ErrNotFound
        }
        filter := append(bson.D{{Key: "_id", Value: id}}, sessionScope(d.session)...)
        set := bson.D{{Key: "status", Value: status}, {Key: "decided_by", Value: d.actor}, {Key: "decided_at", Value: r.now()}}
        if role != "" {
                set = append(set, bson.E{Key: "role", Value: role})
        }
        if reason != "" {
                set = append(set, bson.E{Key: "reason", Value: reason})
        }
        err := r.requests.FindOneAndUpdate(ctx, append(filter, bson.E{Key: "status", Value: RequestPending}), bson.D{{Key: "$set", Value: set}}, &jr)
        if !errors.Is(err, store.ErrNotFound) {
                return jr, err
        }
        if ferr := r.requests.FindOne(ctx, filter, &jr); ferr == nil {
                return jr, ErrConflict
        }
        return jr, ErrNotFound
}

// DropSession soft-deletes every membership record of the session: members, invites, requests,
// and blocks. The session document itself is soft-deleted by SoftDelete.
func (r *Sessions) DropSession(ctx context.Context, d Decision) error {
        if err := d.require(relGM); err != nil {
                return err
        }
        return r.dropRecords(ctx, d.session, d.actor)
}

// dropRecords soft-deletes every membership record of the session as actor.
func (r *Sessions) dropRecords(ctx context.Context, sessionID, actor string) error {
        for _, c := range []store.Collection{r.members, r.invites, r.requests, r.blocks} {
                for {
                        err := r.softDelete(ctx, c, sessionScope(sessionID), actor)
                        if errors.Is(err, ErrNotFound) {
                                break
                        }
                        if err != nil {
                                return err
                        }
                }
        }
        return nil
}

// ScanMembers pages through every membership record, live and soft-deleted, across all sessions.

                q.Sort = []SortKey{{Field: "created_at"}}
        }
        return runQuery[Member](ctx, r.members, SessionMemberSpec, bson.D{}, q)
}

// SessionMembers returns every membership record of one session, live and soft-deleted, read
// fresh. Reconciler only: it has no acting user (PADR-0035, PADR-0038).
func (r *Sessions) SessionMembers(ctx context.Context, sessionID string) ([]Member, error) {
        id, ok := cleanID(sessionID)
        if !ok {
                return nil, ErrNotFound
        }
        var out []Member
        cursor := ""
        for {
                page, err := runQuery[Member](ctx, r.members, SessionMemberSpec, bson.D{{Key: "session_id", Value: id}},
                        Query{Sort: []SortKey{{Field: "created_at"}}, Limit: MaxLimit, Cursor: cursor})
                if err != nil {
                        return nil, err
                }
                out = append(out, page.Items...)
                if page.NextCursor == "" {
                        return out, nil
                }
                cursor = page.NextCursor
        }
}

// LookupSessions maps each of ids that has a session record to whether it is soft-deleted; an ID
// with no record is absent. It reads no session content. Reconciler only: it decides which
// minted session IDs never became sessions (PADR-0040).
func (r *Sessions) LookupSessions(ctx context.Context, ids []string) (map[string]bool, error) {
        var found []Session
        in := make(bson.A, 0, len(ids))
        for _, id := range ids {
                in = append(in, id)
        }
        if err := r.c.Find(ctx, bson.D{{Key: "_id", Value: bson.D{{Key: "$in", Value: in}}}}, store.FindOptions{}, &found); err != nil {
                return nil, err
        }
        out := make(map[string]bool, len(found))
        for _, s := range found {
                out[s.ID] = s.DeletedAt != nil
        }
        return out, nil
}

// liveGMFilter matches live GM member records of the sessions in session (an id or an $in). The
// live predicate lets Mongo use the partial session_user index; deleted_at alone would scan.
func liveGMFilter(session any) bson.D {
        return bson.D{{Key: "session_id", Value: session}, {Key: "role", Value: RoleGM}, {Key: "live", Value: bson.D{{Key: "$exists", Value: true}}}}
}

// SessionsWithGM returns the IDs among ids that have a live GM member record. Reconciler only.
func (r *Sessions) SessionsWithGM(ctx context.Context, ids []string) (map[string]bool, error) {
        in := make(bson.A, 0, len(ids))
        for _, id := range ids {
                in = append(in, id)
        }
        var gms []Member
        if err := r.members.Find(ctx, liveGMFilter(bson.D{{Key: "$in", Value: in}}), store.FindOptions{}, &gms); err != nil {
                return nil, err
        }
        out := make(map[string]bool, len(gms))
        for _, m := range gms {
                out[m.SessionID] = true
        }
        return out, nil
}

// gmLessMinAge matches the 15-minute min_age the reconciler lists mints with, and is re-checked
// here against the session itself.
const gmLessMinAge = 15 * time.Minute

// RemoveGMLess soft-deletes a live session that has no live GM record, and every membership record
// of it, as the system. Such a session is left by a creation whose GM record could not be stored
// and whose removal also failed (Open); nobody can delete it, and it counts toward the owner's
// sessions_per_gm. It refuses (ErrConflict) a session that does have a GM or is younger than gmLessMinAge, so a lookup that raced
// a creation cannot remove a real session. Reconciler only.
func (r *Sessions) RemoveGMLess(ctx context.Context, sessionID string) error {
        id, ok := cleanID(sessionID)
        if !ok {
                return ErrNotFound
        }
        var s Session
        err := r.c.FindOne(ctx, bson.D{{Key: "_id", Value: id}, {Key: "deleted_at", Value: nil}}, &s)
        if errors.Is(err, store.ErrNotFound) {
                return nil // already gone
        }
        if err != nil {
                return err
        }
        if s.CreatedAt.After(r.now().Add(-gmLessMinAge)) {
                return ErrConflict // too young: Open may still be between its two inserts
        }
        var gm Member
        switch err := r.members.FindOne(ctx, liveGMFilter(id), &gm); {
        case err == nil:
                return ErrConflict
        case !errors.Is(err, store.ErrNotFound):
                return err
        }
        // Records first: a failure here leaves the session live, so the next pass finds it again.
        if err := r.dropRecords(ctx, id, models.SystemActor); err != nil {
                return err
        }
        return r.owned.SoftDelete(ctx, s.OwnerUserID, models.SystemActor, id, s.Revision)
}

// Spec declares one collection: its name, the fields a client may filter, sort, and project by
// (allow-lists; anything else is rejected, PADR-0030), fields to unset on soft delete, and its
// indexes.
type Spec struct {
        Name        string
        Filterable  []string
        Sortable    []string
        Projectable []string
        // Updatable lists the fields Owned.Update may set. State changes with their own rules
        // (credential revocation, last use, scopes, secret) are not here; they get dedicated functions.
        Updatable []string
        // StateFields lists the fields only Encounters.SetState may set: results of domain rules.
        StateFields   []string
        UnsetOnDelete []string
        // ConflictOnDuplicate maps a unique-index rejection to ErrConflict. Set it only where the
        // collection's one unique index is a user-facing constraint (theme slot); elsewhere the raw
        // store.ErrDuplicate is returned so it is not mislabeled.
        ConflictOnDuplicate bool
        Indexes             []store.IndexSpec
}

func asc(fields ...string) bson.D {
        d := make(bson.D, 0, len(fields))
        for _, f := range fields {
                d = append(d, bson.E{Key: f, Value: 1})
        }
        return d
}

// Outcome is what Upsert did with a definition.
type Outcome string

// Upsert outcomes.
const (
        Created   Outcome = "created"
        Updated   Outcome = "updated"
        Unchanged Outcome = "unchanged"
)

var slug = regexp.MustCompile(`^[a-z0-9][a-z0-9_-]{0,63}$`)

// SystemRepo reads and seeds game-system definitions. They are shared reference data with no
// owner, so there is no owner scoping; the only writer is the seed loader, as the system actor.
type SystemRepo struct {
        c   store.Collection
        now func() time.Time
}

// NewSystemRepo builds the repository for the game_systems collection in db.
func NewSystemRepo(db store.Database) *SystemRepo {
        return &SystemRepo{c: db.Collection(SystemSpec.Name), now: func() time.Time { return time.Now().UTC() }}
}

// Get returns the system whose ID or alias equals idOrAlias.
func (s *SystemRepo) Get(ctx context.Context, idOrAlias string) (GameSystem, error) {
        var out GameSystem
        filter := bson.D{{Key: "$or", Value: bson.A{
                bson.D{{Key: "_id", Value: idOrAlias}}, bson.D{{Key: "aliases", Value: idOrAlias}},
        }}}
        err := s.c.FindOne(ctx, filter, &out)
        if errors.Is(err, store.ErrNotFound) {
                return out, ErrNotFound
        }
        return out, err
}

// DefaultSystemID is "No System": the simple tracker every user can use, and the definition an
// encounter falls back to when its game system has no definition.
const DefaultSystemID = "none"

// GetOrDefault returns the system for idOrAlias, or the default system when none is defined for it.
func (s *SystemRepo) GetOrDefault(ctx context.Context, idOrAlias string) (GameSystem, error) {
        sys, err := s.Get(ctx, idOrAlias)
        if errors.Is(err, ErrNotFound) {
                return s.Get(ctx, DefaultSystemID)
        }
        return sys, err
}

// List returns one page of systems; filtering (for example gated=false for users without the
// entitlement) goes through q.
func (s *SystemRepo) List(ctx context.Context, q Query) (Page[GameSystem], error) {
        return runQuery[GameSystem](ctx, s.c, SystemSpec, nil, q)
}

// Upsert creates or updates sys by ID. A definition identical to the stored one (same content
// hash) is left untouched, so re-seeding keeps IDs, revisions, and audit fields stable.
func (s *SystemRepo) Upsert(ctx context.Context, sys GameSystem) (Outcome, error) {
        if err := ValidateSystem(sys); err != nil {
                return "", err
        }
        hash, err := definitionHash(sys)
        if err != nil {
                return "", err
        }
        sys.DefinitionHash = hash

        // A concurrent loader can change the row between our read and write; re-read and retry once
        // before giving up, so a lost race is never reported as "unchanged".
        for attempt := 0; attempt < 2; attempt++ {
                var (
                        cur GameSystem
                        out Outcome
                )
                err = s.c.FindOne(ctx, bson.D{{Key: "_id", Value: sys.ID}}, &cur)
                switch {
                case errors.Is(err, store.ErrNotFound):
                        out, err = s.create(ctx, sys)
                case err != nil:
                        return "", err
                case cur.DefinitionHash == hash:
                        return Unchanged, nil
                default:
                        out, err = s.update(ctx, sys, cur.DefinitionHash)
                }
                if !errors.Is(err, errLostRace) {
                        return out, err
                }
        }
        return "", fmt.Errorf("system %q: %w", sys.ID, errLostRace)
}

// errLostRace means another writer changed the definition between the read and the write.
var errLostRace = errors.New("data: concurrent change to system definition")

func (s *SystemRepo) create(ctx context.Context, sys GameSystem) (Outcome, error) {
        sys.Revision = 1
        models.StampCreate(&sys.Auditable, models.SystemActor, s.now())
        err := s.c.InsertOne(ctx, sys)
        if errors.Is(err, store.ErrDuplicate) {
                // Another replica may have inserted this same ID first; if so, retry as an update/no-op.
                var existing GameSystem
                switch rerr := s.c.FindOne(ctx, bson.D{{Key: "_id", Value: sys.ID}}, &existing); {
                case rerr == nil:
                        return "", errLostRace
                case errors.Is(rerr, store.ErrNotFound):
                        return "", fmt.Errorf("system %q alias conflicts with another system: %w", sys.ID, err)
                default:
                        return "", rerr
                }
        }
        if err != nil {
                return "", err
        }
        return Created, nil
}

func (s *SystemRepo) update(ctx context.Context, sys GameSystem, prevHash string) (Outcome, error) {
        filter := bson.D{{Key: "_id", Value: sys.ID}, {Key: "definition_hash", Value: prevHash}}
        update := bson.D{
                {Key: "$set", Value: bson.D{
                        {Key: "name", Value: sys.Name}, {Key: "family", Value: sys.Family},
                        {Key: "aliases", Value: sys.Aliases}, {Key: "gated", Value: sys.Gated},
                        {Key: "fields", Value: sys.Fields}, {Key: "definition_hash", Value: sys.DefinitionHash},
                        {Key: "updated_at", Value: s.now()}, {Key: "updated_by", Value: models.SystemActor}}},
                {Key: "$inc", Value: bson.D{{Key: "revision", Value: 1}}},
        }
        var out GameSystem
        err := s.c.FindOneAndUpdate(ctx, filter, update, &out)
        if errors.Is(err, store.ErrDuplicate) {
                return "", fmt.Errorf("system %q alias conflicts with another system: %w", sys.ID, err)
        }
        if errors.Is(err, store.ErrNotFound) {
                return "", errLostRace
        }
        if err != nil {
                return "", err
        }
        return Updated, nil
}

func definitionHash(sys GameSystem) (string, error) {
        b, err := json.Marshal(struct {
                Name    string         `json:"name"`
                Family  string         `json:"family"`
                Aliases []string       `json:"aliases"`
                Gated   bool           `json:"gated"`
                Fields  []TrackerField `json:"fields"`
        }{sys.Name, sys.Family, sys.Aliases, sys.Gated, sys.Fields})
        if err != nil {
                return "", err
        }
        sum := sha256.Sum256(b)
        return hex.EncodeToString(sum[:]), nil
}

// ValidateSystem checks a definition is self-consistent: slug IDs and aliases, known field
// kinds, unique keys, bounds that make sense, and max references that resolve.
func ValidateSystem(sys GameSystem) error {
        bad := func(format string, a ...any) error {
                return fmt.Errorf("%w: system %q: %s", ErrInvalid, sys.ID, fmt.Sprintf(format, a...))
        }
        if !slug.MatchString(sys.ID) {
                return bad("id must be a lowercase slug")
        }
        if sys.Name == "" || sys.Family == "" || len(sys.Fields) == 0 {
                return bad("name, family and at least one field are required")
        }
        for _, a := range sys.Aliases {
                if !slug.MatchString(a) {
                        return bad("alias %q must be a lowercase slug", a)
                }
        }
        byKey := map[string]TrackerField{}
        for _, f := range sys.Fields {
                if f.Key == "" || byKey[f.Key].Key != "" {
                        return bad("field keys must be non-empty and unique (%q)", f.Key)
                }
                byKey[f.Key] = f
        }
        for _, f := range sys.Fields {
                if err := validateField(f, byKey); err != nil {
                        return bad("%v", err)
                }
        }
        return nil
}

func validateField(f TrackerField, byKey map[string]TrackerField) error {
        switch f.Kind {
        case FieldCounter, FieldPool, FieldFlag, FieldTrack:
        default:
                return fmt.Errorf("field %q has unknown kind %q", f.Key, f.Kind)
        }
        if f.Max != nil && f.MaxRef != "" {
                return fmt.Errorf("field %q sets both max and max_ref", f.Key)
        }
        if f.Kind == FieldFlag && (f.Def < 0 || f.Def > 1 || f.MaxRef != "" ||
                (f.Min != nil && *f.Min < 0) || (f.Max != nil && *f.Max > 1)) {
                return fmt.Errorf("flag field %q must stay within 0 and 1", f.Key)
        }
        if f.MaxRef != "" {
                if ref, ok := byKey[f.MaxRef]; !ok || f.MaxRef == f.Key {
                        return fmt.Errorf("field %q max_ref %q does not name another field", f.Key, f.MaxRef)
                } else if f.Def > ref.Def {
                        return fmt.Errorf("field %q default exceeds its max_ref %q default", f.Key, f.MaxRef)
                }
        }
        if f.Min != nil && f.Max != nil && *f.Min > *f.Max {
                return fmt.Errorf("field %q min exceeds max", f.Key)
        }
        if (f.Min != nil && f.Def < *f.Min) || (f.Max != nil && f.Def > *f.Max) {
                return fmt.Errorf("field %q default is out of bounds", f.Key)
        }
        return nil
}

// CombatantPatch is a partial combatant edit: nil fields are left alone. Health values are merged
// over the stored map (keys not in the patch keep their values); the merged map is what gets
// validated.
type CombatantPatch struct {
        Name       *string
        Kind       *string
        Initiative *float64
        Notes      *string
        Health     map[string]int
}

func newCombatant(c data.Combatant, id string) (data.Combatant, error) {
        c.Name = strings.TrimSpace(c.Name)
        if c.Name == "" {
                return c, invalid("combatant name is required")
        }
        if !slices.Contains(kinds, c.Kind) {
                return c, invalid("combatant kind %q", c.Kind)
        }
        c.ID, c.Removed = id, false
        c.Health, c.ExternalRefs = maps.Clone(c.Health), maps.Clone(c.ExternalRefs)
        c.Conditions = slices.Clone(c.Conditions)
        if c.Conditions == nil {
                c.Conditions = []data.Condition{}
        }
        return c, nil
}

func findCombatant(st *data.EncounterState, id string) (*data.Combatant, error) {
        i := slices.IndexFunc(st.Combatants, func(c data.Combatant) bool { return c.ID == id })
        if i < 0 {
                return nil, data.ErrNotFound
        }
        return &st.Combatants[i], nil
}

// AddCombatant appends c to the encounter with a new ID.
func (s *Encounters) AddCombatant(ctx context.Context, owner, actor, id string, rev int64, c data.Combatant) (data.Encounter, error) {
        n, err := newCombatant(c, s.newID())
        if err != nil {
                return data.Encounter{}, err
        }
        return s.editQueue(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                st.Combatants = append(st.Combatants, n)
                return nil
        })
}

// UpdateCombatant applies patch to the combatant with ID combatantID.
func (s *Encounters) UpdateCombatant(ctx context.Context, owner, actor, id string, rev int64, combatantID string, patch CombatantPatch) (data.Encounter, error) {
        return s.editQueue(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                c, err := findCombatant(st, combatantID)
                if err != nil {
                        return err
                }
                return applyPatch(c, patch)
        })
}

func applyPatch(c *data.Combatant, p CombatantPatch) error {
        if p.Name != nil {
                name := strings.TrimSpace(*p.Name)
                if name == "" {
                        return invalid("combatant name is required")
                }
                c.Name = name
        }
        if p.Kind != nil {
                if !slices.Contains(kinds, *p.Kind) {
                        return invalid("combatant kind %q", *p.Kind)
                }
                c.Kind = *p.Kind
        }
        if p.Initiative != nil {
                v := *p.Initiative
                c.Initiative = &v
        }
        if p.Notes != nil {
                c.Notes = *p.Notes
        }
        if p.Health != nil {
                merged := maps.Clone(c.Health)
                if merged == nil {
                        merged = map[string]int{}
                }
                maps.Copy(merged, p.Health)
                c.Health = merged
        }
        return nil
}

// RemoveCombatant marks the combatant removed. It stays in the record and leaves the turn queue;
// if it was current, the turn passes to the combatant that takes its place in the queue.
func (s *Encounters) RemoveCombatant(ctx context.Context, owner, actor, id string, rev int64, combatantID string) (data.Encounter, error) {
        return s.editQueue(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                c, err := findCombatant(st, combatantID)
                if err != nil {
                        return err
                }
                c.Removed = true
                return nil
        })
}

// RestoreCombatant clears the removed mark and puts the combatant back in the turn queue.
func (s *Encounters) RestoreCombatant(ctx context.Context, owner, actor, id string, rev int64, combatantID string) (data.Encounter, error) {
        return s.editQueue(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                c, err := findCombatant(st, combatantID)
                if err != nil {
                        return err
                }
                c.Removed = false
                return nil
        })
}

// KeyIntegrationCredentials is the entitlement limit on active integration credentials.
const KeyIntegrationCredentials = "initiative.integration_credentials"

// SecretPrefix starts every integration secret, so the auth middleware can tell one from a user
// token without a lookup.
const SecretPrefix = "ic_"

const maxLabel = 100

// Credentials issues, lists, revokes, and authenticates integration credentials.
type Credentials struct {
        repo *data.Credentials
        open atomic.Bool
}

// ErrCredentialsClosed means integration credentials are off: the auth-event consumer is not
// caught up, or it fell behind the stream's retention and awaits a reconcile (PADR-0039). It is
// fail closed, so no credential authenticates and none is issued.
var ErrCredentialsClosed = errors.New("integration credentials are closed")

// NewCredentials builds the service on db.
func NewCredentials(db store.Database) *Credentials {
        return &Credentials{repo: data.NewCredentials(db)}
}

// hashSecret is the stored form of a secret. The secret is 256 random bits, so a plain SHA-256 is
// enough; a slow hash would only add cost to every request.
func hashSecret(secret string) string {
        sum := sha256.Sum256([]byte(secret))
        return hex.EncodeToString(sum[:])
}

func newSecret() (string, error) {
        b := make([]byte, 32)
        if _, err := rand.Read(b); err != nil {
                return "", err
        }
        return SecretPrefix + base64.RawURLEncoding.EncodeToString(b), nil
}

func validateScopes(scopes []string) error {
        if len(scopes) == 0 {
                return fmt.Errorf("%w: at least one scope is required", data.ErrInvalid)
        }
        for _, s := range scopes {
                if s != data.ScopeRead && s != data.ScopeWrite {
                        return fmt.Errorf("%w: unknown scope %q", data.ErrInvalid, s)
                }
        }
        return nil
}

// Open lets credentials work, unless a persisted close from an earlier run is still in force.
// Call it once the auth-event consumer has caught up. It reports whether the gate is now open.
func (s *Credentials) Open(ctx context.Context) (bool, error) {
        closed, err := s.repo.GateClosed(ctx)
        if err != nil || closed {
                return false, err
        }
        s.open.Store(true)
        return true, nil
}

// Close turns credentials off and persists that, so a restart stays closed until Reopen.
func (s *Credentials) Close(ctx context.Context) error {
        s.open.Store(false)
        return s.repo.SetGate(ctx, true, "retention-gap")
}

// Reopen clears a persisted close on an administrator's say-so; actor is recorded on the gate.
func (s *Credentials) Reopen(ctx context.Context, actor string) error {
        if err := s.repo.SetGate(ctx, false, actor); err != nil {
                return err
        }
        s.open.Store(true)
        return nil
}

// RevokeOwner ends every active credential of an owner the auth service has denied or stripped of
// a role, and returns how many it ended. Credentials stay for audit.
func (s *Credentials) RevokeOwner(ctx context.Context, owner, reason string) (int, error) {
        if reason != data.ReasonOwnerDenied && reason != data.ReasonOwnerRoleRemoved {
                return 0, fmt.Errorf("%w: unknown revoke reason %q", data.ErrInvalid, reason)
        }
        return s.repo.RevokeAllByOwner(ctx, owner, reason)
}

// Create issues a credential for owner and returns it with its secret, which is never available
// again. At or over the entitled limit of active credentials it returns *EntitlementError.
func (s *Credentials) Create(ctx context.Context, owner, label string, scopes []string, chk Checker) (data.IntegrationCredential, string, error) {
        if !s.open.Load() {
                return data.IntegrationCredential{}, "", ErrCredentialsClosed
        }
        label = strings.TrimSpace(label)
        if label == "" || len([]rune(label)) > maxLabel {
                return data.IntegrationCredential{}, "", fmt.Errorf("%w: label is required and at most %d characters", data.ErrInvalid, maxLabel)
        }
        if err := validateScopes(scopes); err != nil {
                return data.IntegrationCredential{}, "", err
        }
        res, err := chk.Check(ctx, KeyIntegrationCredentials, entitlements.FailClosed)
        if err != nil {
                return data.IntegrationCredential{}, "", err
        }
        // ponytail: count-then-insert can overshoot the limit by the number of concurrent creates.
        // Tighten with a per-owner slot index if that ever matters.
        n, err := s.repo.CountActive(ctx, owner)
        if err != nil {
                return data.IntegrationCredential{}, "", err
        }
        if n >= int64(res.Limit) {
                return data.IntegrationCredential{}, "", &EntitlementError{Key: KeyIntegrationCredentials, Limit: res.Limit}
        }
        secret, err := newSecret()
        if err != nil {
                return data.IntegrationCredential{}, "", err
        }
        c := data.IntegrationCredential{Label: label, Scopes: slices.Compact(slices.Sorted(slices.Values(scopes))), SecretHash: hashSecret(secret)}
        if err := s.repo.Create(ctx, owner, owner, &c); err != nil {
                return data.IntegrationCredential{}, "", err
        }
        return c, secret, nil
}

// List returns owner's credentials, newest first. Revoked ones stay listed.
func (s *Credentials) List(ctx context.Context, owner string, q data.Query) (data.Page[data.IntegrationCredential], error) {
        return s.repo.List(ctx, owner, q)
}

// Revoke ends owner's credential at once; repeating it is harmless.
func (s *Credentials) Revoke(ctx context.Context, owner, id string) (data.IntegrationCredential, error) {
        return s.repo.Revoke(ctx, owner, owner, id)
}

// Authenticate resolves a presented secret to its credential and records the use. A revoked,
// deleted, or unknown secret is data.ErrNotFound; the next request after a revoke fails here.
func (s *Credentials) Authenticate(ctx context.Context, secret string) (data.IntegrationCredential, error) {
        if !strings.HasPrefix(secret, SecretPrefix) {
                return data.IntegrationCredential{}, data.ErrNotFound
        }
        if !s.open.Load() {
                return data.IntegrationCredential{}, ErrCredentialsClosed
        }
        return s.repo.Authenticate(ctx, hashSecret(secret))
}

// ListAll returns credentials across owners for an administrator, optionally for one owner.
func (s *Credentials) ListAll(ctx context.Context, owner string, q data.Query) (data.Page[data.IntegrationCredential], error) {
        if owner != "" {
                return s.repo.ListOwned(ctx, owner, q)
        }
        return s.repo.ListAll(ctx, q)
}

// DeletePermanently removes a credential for good at an administrator's say-so and records the

func (s *Credentials) DeletePermanently(ctx context.Context, admin, id, reason string) (data.CredentialDeletion, error) {
        return s.repo.DeletePermanently(ctx, admin, id, reason)
}

func (s *Credentials) Deletions(ctx context.Context, owner string, q data.Query) (data.Page[data.CredentialDeletion], error) {
        return s.repo.Deletions(ctx, owner, q)
}

// credential calls it on each heartbeat or event and closes when it turns false.
func (s *Credentials) Live(ctx context.Context, id string) (bool, error) {
        if !s.open.Load() {
                return false, nil
        }
        return s.repo.Live(ctx, id)
}

// Create stores enc as owner's new draft encounter. The name is required, the game system must
// exist, and any combatants supplied get server-assigned IDs.
func (s *Encounters) Create(ctx context.Context, owner, actor string, enc data.Encounter) (data.Encounter, error) {
        enc.Name = strings.TrimSpace(enc.Name)
        if enc.Name == "" {
                return data.Encounter{}, invalid("encounter name is required")
        }
        _, err := s.systems.Get(ctx, enc.GameSystemID)
        if errors.Is(err, data.ErrNotFound) {
                return data.Encounter{}, invalid("the encounter's game system is not defined")
        }
        if err != nil {
                return data.Encounter{}, err
        }
        if err := requireSystem(ctx, enc.GameSystemID); err != nil {
                return data.Encounter{}, err
        }
        combatants := make([]data.Combatant, 0, len(enc.Combatants))
        for _, c := range enc.Combatants {
                n, err := newCombatant(c, s.newID())
                if err != nil {
                        return data.Encounter{}, err
                }
                combatants = append(combatants, n)
        }
        enc.Combatants = combatants
        if err := s.repo.Create(ctx, owner, actor, &enc); err != nil {
                return data.Encounter{}, err
        }
        return enc, nil
}

// UpdateSettings sets the encounter's settings (name, game_system_id, ordering, tie_breaker,
// display flags, theme_id, position). Combat state cannot be set here, but a new ordering, tie
// breaker, or game system recomputes the turn queue in the same write.
func (s *Encounters) UpdateSettings(ctx context.Context, owner, actor, id string, rev int64, set map[string]any) (data.Encounter, error) {
        if sysID, ok := set["game_system_id"].(string); ok {
                if err := requireSystem(ctx, sysID); err != nil {
                        return data.Encounter{}, err
                }
        }
        _, ord := set["ordering"]
        _, tie := set["tie_breaker"]
        _, sys := set["game_system_id"]
        if !ord && !tie && !sys {
                return s.repo.Update(ctx, owner, actor, id, rev, set)
        }
        cur, err := s.repo.Load(ctx, owner, id)
        if err != nil {
                return data.Encounter{}, err
        }
        o, t, sysID := cur.Ordering, cur.TieBreaker, cur.GameSystemID
        if v, ok := set["ordering"].(string); ok {
                o = v
        }
        if v, ok := set["game_system_id"].(string); ok {
                sysID = v
        }
        if v, ok := set["tie_breaker"].(string); ok {
                t = v
        }
        st := stateOf(cur)
        s.recompute(&st, effectiveOrdering(o, sysID), t)
        return s.repo.Write(ctx, owner, actor, id, rev, set, st)
}

func stateOf(cur data.Encounter) data.EncounterState {
        return data.EncounterState{
                Status: cur.Status, Round: cur.Round, CurrentCombatantID: cur.CurrentCombatantID,
                TurnQueue: slices.Clone(cur.TurnQueue), Combatants: slices.Clone(cur.Combatants),
        }
}

// editQueue is edit followed by a turn-queue recompute under the encounter's ordering.
func (s *Encounters) editQueue(ctx context.Context, owner, actor, id string, rev int64, change func(*data.EncounterState, data.Encounter) error) (data.Encounter, error) {
        return s.edit(ctx, owner, actor, id, rev, func(st *data.EncounterState, cur data.Encounter) error {
                if err := change(st, cur); err != nil {
                        return err
                }
                s.recompute(st, effectiveOrdering(cur.Ordering, cur.GameSystemID), cur.TieBreaker)
                return nil
        })
}

// edit reads the encounter, applies change to its state, and writes the result compare-and-set on
// rev. A missed read is classified 403/404 like the write paths.
// change also receives the encounter as read, for its settings.
func (s *Encounters) edit(ctx context.Context, owner, actor, id string, rev int64, change func(*data.EncounterState, data.Encounter) error) (data.Encounter, error) {
        cur, err := s.repo.Load(ctx, owner, id)
        if err != nil {
                return data.Encounter{}, err

        st := stateOf(cur)
        if err := change(&st, cur); err != nil {
                return data.Encounter{}, err
        }
        return s.repo.SetState(ctx, owner, actor, id, rev, st)
}

import (

        "slices"
        "strings"

        "github.com/sweetrpg/initiative-api/data"
)

// Pure turn-order logic: no I/O, no clock. Randomness comes in as a permutation function.

type entry struct {
        c       data.Combatant
        listIdx int
        prevIdx int // position in the previous queue, -1 if it was not queued
        rank    int // random rank among newly queued combatants
}

func kindGroup(ordering, kind string) int {
        if ordering != data.OrderPCVsAdversary {
                return 0
        }
        switch kind {
        case data.KindPC:
                return 0
        case data.KindAdversary:
                return 1
        }
        return 2
}

// initiativeCmp orders by initiative, unset last. lowToHigh ascends; everything else descends.
func initiativeCmp(ordering string, a, b *float64) int {
        switch {
        case a == nil && b == nil:
                return 0
        case a == nil:
                return 1
        case b == nil:
                return -1
        case ordering == data.OrderLowToHigh:
                return cmp.Compare(*a, *b)
        }
        return cmp.Compare(*b, *a)
}

func tieCmp(tie string, prevLen int, a, b entry) int {
        switch tie {
        case data.TieAlphabetical:
                if c := strings.Compare(strings.ToLower(a.c.Name), strings.ToLower(b.c.Name)); c != 0 {
                        return c
                }
                return strings.Compare(a.c.ID, b.c.ID)
        case data.TieRandom:
                // Combatants already queued keep their relative order, so a recompute never reshuffles
                // ties mid-combat. Newly queued ones fall in behind them in random order.
                return cmp.Compare(randKey(a, prevLen), randKey(b, prevLen))
        }
        return cmp.Compare(a.listIdx, b.listIdx) // data.TieQuery: encounter list order
}

func randKey(e entry, prevLen int) int {
        if e.prevIdx >= 0 {
                return e.prevIdx
        }
        return prevLen + e.rank
}

// buildQueue returns the IDs of the non-removed combatants in turn order. prev is the existing
// queue. perm(n) returns a random permutation of 0..n-1.
//
// playerManaged keeps prev's order, dropping departed combatants and appending new ones in list
// order. The other orderings sort by kind group (pcVsAdversary only: pc, adversary, object), then
// initiative, then the tie breaker.
func buildQueue(ordering, tie string, cs []data.Combatant, prev []string, perm func(int) []int) []string {
        prevIdx := make(map[string]int, len(prev))
        for i, id := range prev {
                prevIdx[id] = i
        }
        var es []entry
        for i, c := range cs {
                if c.Removed {
                        continue
                }
                pi, ok := prevIdx[c.ID]
                if !ok {
                        pi = -1
                }
                es = append(es, entry{c: c, listIdx: i, prevIdx: pi})
        }
        if ordering == data.OrderPlayerManaged {
                return manualQueue(es)
        }
        assignRanks(es, perm)
        slices.SortStableFunc(es, func(a, b entry) int {
                return cmp.Or(
                        cmp.Compare(kindGroup(ordering, a.c.Kind), kindGroup(ordering, b.c.Kind)),
                        initiativeCmp(ordering, a.c.Initiative, b.c.Initiative),
                        tieCmp(tie, len(prev), a, b),
                        cmp.Compare(a.listIdx, b.listIdx),
                )
        })
        return ids(es)
}

func assignRanks(es []entry, perm func(int) []int) {
        var fresh []int
        for i, e := range es {
                if e.prevIdx < 0 {
                        fresh = append(fresh, i)
                }
        }
        if len(fresh) == 0 {
                return
        }
        for r, p := range perm(len(fresh)) {
                es[fresh[r]].rank = p
        }
}

func manualQueue(es []entry) []string {
        slices.SortStableFunc(es, func(a, b entry) int {
                switch {
                case a.prevIdx >= 0 && b.prevIdx >= 0:
                        return cmp.Compare(a.prevIdx, b.prevIdx)
                case a.prevIdx >= 0:
                        return -1
                case b.prevIdx >= 0:
                        return 1
                }
                return cmp.Compare(a.listIdx, b.listIdx)
        })
        return ids(es)
}

func ids(es []entry) []string {
        out := make([]string, 0, len(es))
        for _, e := range es {
                out = append(out, e.c.ID)
        }
        return out
}

// expireConditions applies the start-of-turn rule to c: each timed condition loses one round and
// is removed at zero. Open-ended conditions are untouched. It builds a new slice, so shared
// backing arrays are never written.
func expireConditions(c *data.Combatant) {
        next := make([]data.Condition, 0, len(c.Conditions))
        for _, cond := range c.Conditions {
                if cond.RoundsRemaining == nil {
                        next = append(next, cond)
                        continue
                }
                if left := *cond.RoundsRemaining - 1; left > 0 {
                        next = append(next, data.Condition{Name: cond.Name, RoundsRemaining: &left})
                }
        }
        c.Conditions = next
}

// nextTurn returns the queue position after pos and whether it wrapped past the end. pos -1 (no
// current combatant) yields position 0 without wrapping.
func nextTurn(pos, n int) (idx int, wrapped bool) {
        if pos+1 >= n {
                return 0, pos >= 0
        }
        return pos + 1, false
}

// prevTurn returns the queue position before pos and whether it wrapped past the start.
func prevTurn(pos, n int) (idx int, wrapped bool) {
        if pos <= 0 {
                return n - 1, true
        }
        return pos - 1, false
}

// KeyGameSystemsExtra is the flag that unlocks every game system other than No System.
const KeyGameSystemsExtra = "initiative.game_systems_extra"

// EntitlementRequiredError is the typed denial for a flag the caller does not hold.
type EntitlementRequiredError struct{ Key string }

func (e *EntitlementRequiredError) Error() string {
        return fmt.Sprintf("domain: entitlement %s is required", e.Key)
}

type checkerKey struct{}

// WithChecker attaches the calling user's entitlement checker to ctx, so rules deep in the domain
// (a system switch inside a settings update) can check without every signature carrying it. The
// server sets it per request, bound to that user's own bearer (PADR-0037).
func WithChecker(ctx context.Context, c Checker) context.Context {
        return context.WithValue(ctx, checkerKey{}, c)
}

// requireSystem rejects use of a gated game system by a caller without the flag. No System is
// free. Creates and switches fail closed: with no checker in ctx or entitlements unreachable the
// call is refused.
func requireSystem(ctx context.Context, systemID string) error {
        if systemID == "" || systemID == data.DefaultSystemID {
                return nil
        }
        chk, ok := ctx.Value(checkerKey{}).(Checker)
        if !ok {
                return entitlements.ErrUnavailable
        }
        res, err := chk.Check(ctx, KeyGameSystemsExtra, entitlements.FailClosed)
        if err != nil {
                return err
        }
        if !res.Flag {
                return &EntitlementRequiredError{Key: KeyGameSystemsExtra}
        }
        return nil
}

// Locks keeps encounters that use a gated game system in step with the owner's flag: locked while
// it is missing, restored unchanged when it returns.
type Locks struct {
        repo *data.Encounters
}

// Locks returns the lock reconciler for these encounters.
func (s *Encounters) Locks() *Locks { return &Locks{repo: s.repo} }

// Reconcile reads the flag and locks or unlocks owner's encounters to match. When entitlements
// cannot be read it changes nothing, so an outage never locks anyone out.
func (l *Locks) Reconcile(ctx context.Context, owner string, chk Checker) error {
        res, err := chk.Check(ctx, KeyGameSystemsExtra, entitlements.FailOpen)
        if err != nil {
                return err
        }
        if res.Degraded {
                return nil
        }
        var n int
        if res.Flag {
                n, err = l.repo.UnlockAll(ctx, owner)
        } else {
                n, err = l.repo.LockGated(ctx, owner)
        }
        if n > 0 {
                slog.InfoContext(ctx, "game-system locks changed", "locked", !res.Flag, "count", n)
        }
        return err
}

// Entitlement keys for the stored-item limits.
const (
        KeySavedRosters    = "initiative.saved_rosters"
        KeySavedEncounters = "initiative.saved_encounters"
        KeySessionsPerGM   = "initiative.sessions_per_gm"
        // KeyMembersPerSession is recorded on each session; 7.12 enforces it.
        KeyMembersPerSession = "initiative.members_per_session"

// Limited is a repository whose items can go inactive when the owner's limit falls.
type Limited interface {
        ItemStates(ctx context.Context, owner string) ([]data.ItemState, error)
        SetInactive(ctx context.Context, owner, id string, inactive bool) error
        SetKeepRanks(ctx context.Context, owner string, ids []string) error
}

// Kind is one limited item type: its URL name, its entitlement key, and its items.
type Kind struct {
        Name string
        Key  string
        Repo Limited
}

// Limits keeps only as many of an owner's items active as the entitled limit allows (spec
// initiative-entitlement-gating). It never deletes or archives anything.
type Limits struct {
        kinds []Kind
}

// NewLimits builds the enforcer for kinds.
func NewLimits(kinds ...Kind) *Limits { return &Limits{kinds: kinds} }

// plan returns the IDs to activate and to deactivate so that exactly the first limit items in
// preference order are active: the owner's choice (by rank) first, then oldest first. states must
// arrive oldest first.
func plan(states []data.ItemState, limit int) (activate, deactivate []string) {
        ordered := slices.Clone(states)
        slices.SortStableFunc(ordered, func(a, b data.ItemState) int {
                switch {
                case a.KeepRank > 0 && b.KeepRank > 0:
                        return a.KeepRank - b.KeepRank
                case a.KeepRank > 0:
                        return -1
                case b.KeepRank > 0:
                        return 1
                }
                return 0
        })
        for i, st := range ordered {
                switch {
                case i < limit && st.Inactive:
                        activate = append(activate, st.ID)
                case i >= limit && !st.Inactive:
                        deactivate = append(deactivate, st.ID)
                }
        }
        return activate, deactivate
}

func (l *Limits) kind(name string) (Kind, bool) {
        for _, k := range l.kinds {
                if k.Name == name {
                        return k, true
                }
        }
        return Kind{}, false
}

func (l *Limits) apply(ctx context.Context, owner string, k Kind, limit int) error {
        states, err := k.Repo.ItemStates(ctx, owner)
        if err != nil {
                return err
        }
        activate, deactivate := plan(states, limit)
        for _, id := range deactivate {
                if err := k.Repo.SetInactive(ctx, owner, id, true); err != nil {
                        return err
                }
        }
        for _, id := range activate {
                if err := k.Repo.SetInactive(ctx, owner, id, false); err != nil {
                        return err
                }
        }
        return nil
}

// Reconcile brings every kind in line with owner's current limits. When entitlements cannot be
// read it leaves that kind as it is, so an outage never deactivates anything. It tries every
// kind and returns the first error.
func (l *Limits) Reconcile(ctx context.Context, owner string, chk Checker) error {
        var first error
        for _, k := range l.kinds {
                res, err := chk.Check(ctx, k.Key, entitlements.FailOpen)
                if err == nil && res.Degraded {
                        continue
                }
                if err == nil {
                        err = l.apply(ctx, owner, k, res.Limit)
                }
                if err != nil {
                        slog.WarnContext(ctx, "limit reconcile failed", "kind", k.Name, "error", err)
                        if first == nil {
                                first = err
                        }
                }
        }
        return first
}

// Choose records which items of a kind the owner wants active, in order, then applies the limit.
// ids must be distinct items the owner holds, at most the limit; if fewer than the limit are
// chosen, the remaining places go to the oldest other items. Returns the active IDs afterward.
func (l *Limits) Choose(ctx context.Context, owner, kind string, ids []string, chk Checker) ([]string, error) {
        k, ok := l.kind(kind)
        if !ok {
                return nil, data.ErrNotFound
        }
        res, err := chk.Check(ctx, k.Key, entitlements.FailClosed)
        if err != nil {
                return nil, err
        }
        states, err := k.Repo.ItemStates(ctx, owner)
        if err != nil {
                return nil, err
        }
        if len(ids) > res.Limit {
                return nil, invalid("at most %d items can be active", res.Limit)
        }
        held := make(map[string]bool, len(states))
        for _, st := range states {
                held[st.ID] = true
        }
        seen := map[string]bool{}
        for _, id := range ids {
                if !held[id] || seen[id] {
                        return nil, invalid("ids must be distinct items you own")
                }
                seen[id] = true
        }
        if err := k.Repo.SetKeepRanks(ctx, owner, ids); err != nil {
                return nil, err
        }
        if err := l.apply(ctx, owner, k, res.Limit); err != nil {
                return nil, err
        }
        after, err := k.Repo.ItemStates(ctx, owner)
        if err != nil {
                return nil, err
        }
        active := []string{}
        for _, st := range after {
                if !st.Inactive {
                        active = append(active, st.ID)
                }
        }
        return active, nil
}

// normalize trims and validates a character, copying its maps so the stored value does not alias
// the caller's. The ID is left for the caller to assign.
func normalize(c data.Character) (data.Character, error) {
        c.Name = strings.TrimSpace(c.Name)
        if c.Name == "" {
                return c, invalid("character name is required")
        }
        if !slices.Contains(kinds, c.Kind) {
                return c, invalid("character kind %q", c.Kind)
        }
        c.Health, c.ExternalRefs = maps.Clone(c.Health), maps.Clone(c.ExternalRefs)
        return c, nil
}

func (s *Rosters) prepare(chars []data.Character) ([]data.Character, error) {
        out := make([]data.Character, 0, len(chars))
        for _, c := range chars {
                n, err := normalize(c)
                if err != nil {
                        return nil, err
                }
                n.ID = s.newID()
                out = append(out, n)
        }
        return out, nil
}

// editCharacters applies edit to a copy of the roster's characters and writes the result
// compare-and-set on rev.
func (s *Rosters) editCharacters(ctx context.Context, owner, actor, id string, rev int64, edit func([]data.Character) ([]data.Character, error)) (data.Roster, error) {
        cur, err := s.rosters.Get(ctx, owner, id)
        if errors.Is(err, data.ErrNotFound) {
                // The read is only needed to build the new character list; classify a miss as 403 or 404
                // like the write paths do.
                if cerr := s.rosters.Missing(ctx, owner, id); cerr != nil {
                        return data.Roster{}, cerr
                }
        }
        if err != nil {
                return data.Roster{}, err
        }
        next, err := edit(slices.Clone(cur.Characters))
        if err != nil {
                return data.Roster{}, err
        }
        return s.rosters.Update(ctx, owner, actor, id, rev, map[string]any{"characters": next})
}

func indexOf(chars []data.Character, id string) int {
        return slices.IndexFunc(chars, func(c data.Character) bool { return c.ID == id })
}

// AddCharacter appends c to the roster with a new ID.

        if err != nil {
                return data.Roster{}, err
        }
        n.ID = s.newID()
        return s.editCharacters(ctx, owner, actor, id, rev, func(cs []data.Character) ([]data.Character, error) {
                return append(cs, n), nil
        })
}

// UpdateCharacter replaces the roster character with ID charID, keeping its ID.
func (s *Rosters) UpdateCharacter(ctx context.Context, owner, actor, id string, rev int64, charID string, c data.Character) (data.Roster, error) {
        n, err := normalize(c)
        if err != nil {
                return data.Roster{}, err
        }
        n.ID = charID
        return s.editCharacters(ctx, owner, actor, id, rev, func(cs []data.Character) ([]data.Character, error) {
                i := indexOf(cs, charID)
                if i < 0 {
                        return nil, data.ErrNotFound
                }
                cs[i] = n
                return cs, nil
        })
}

// RemoveCharacter deletes the roster character with ID charID.
func (s *Rosters) RemoveCharacter(ctx context.Context, owner, actor, id string, rev int64, charID string) (data.Roster, error) {
        return s.editCharacters(ctx, owner, actor, id, rev, func(cs []data.Character) ([]data.Character, error) {
                i := indexOf(cs, charID)
                if i < 0 {
                        return nil, data.ErrNotFound
                }
                return slices.Delete(cs, i, i+1), nil
        })
}

// SessionAuthority is auth-api's session surface (PADR-0040): minting a session ID, ending a
// session, and writing membership tuples. *relations.Client implements it. Every call forwards the
// caller's own bearer (PADR-0037).
type SessionAuthority interface {
        Mint(ctx context.Context, token string) (string, error)
        EndSession(ctx context.Context, token, sessionID string) error
        WriteTuples(ctx context.Context, token, sessionID string, writes, deletes []relations.Tuple) error
}

// ErrSyncPending means a membership change was saved but its OpenFGA tuple could not be written.
// The member is denied until the tuple exists, which is the safe direction; repeating the request
// finishes the write, and the reconciler repairs it within one interval otherwise.
var ErrSyncPending = errors.New("domain: membership saved, tuple sync pending")

type authorityKey struct{}

type authority struct {
        svc   SessionAuthority
        token string
}

// WithAuthority puts auth-api's session surface and the caller's bearer on ctx. Without it every
// session change fails closed.
func WithAuthority(ctx context.Context, svc SessionAuthority, token string) context.Context {
        return context.WithValue(ctx, authorityKey{}, authority{svc, token})
}

func authorityOf(ctx context.Context) (authority, error) {
        a, ok := ctx.Value(authorityKey{}).(authority)
        if !ok || a.svc == nil {
                return authority{}, relations.ErrUnavailable
        }
        return a, nil
}

// memberRoles are the relations the membership endpoint writes; gm is never one.
var memberRoles = []string{data.RoleCoGM, data.RolePlayer, data.RoleObserver}

func tupleOf(user, role string) relations.Tuple {
        return relations.Tuple{User: "user:" + user, Relation: role}
}

// syncGrant writes user's role tuple after the Mongo record exists. It deletes the user's other
// non-gm relations in the same atomic write, so a role change or a retry cannot leave two roles;
// a non-GM actor leaves co_gm tuples alone because only the GM changes those (PADR-0040 section 2).
// A refusal from auth-api is returned as such; an outage is ErrSyncPending. A SetRole retry by a
// co_gm when Mongo already holds the new role cannot know the old relation, so it leaves a stale
// co_gm tuple for the reconciler to remove (accepted: only the GM changes co_gm members).
func (s *Sessions) syncGrant(ctx context.Context, actorRole, sessionID, user, role string) error {
        a, err := authorityOf(ctx)
        if err != nil {
                return fmt.Errorf("%w: %w", ErrSyncPending, err)
        }
        var deletes []relations.Tuple
        for _, r := range memberRoles {
                if r != role && (r != data.RoleCoGM || actorRole == data.RoleGM) {
                        deletes = append(deletes, tupleOf(user, r))
                }
        }
        err = a.svc.WriteTuples(ctx, a.token, sessionID, []relations.Tuple{tupleOf(user, role)}, deletes)
        switch {
        case err == nil:
                return nil
        case errors.Is(err, relations.ErrUnauthorized), errors.Is(err, relations.ErrDenied),
                errors.Is(err, relations.ErrLimited), errors.Is(err, relations.ErrConflict),
                errors.Is(err, relations.ErrRejected):
                // auth-api answered with a refusal the client should see as such (401, 403, 429, 409
                // session_full). Mongo is already written, so the member is denied until the tuple exists,
                // which is the safe direction; the reconciler repairs it. Only an outage is "pending".
                return err
        }
        slog.WarnContext(ctx, "membership tuple write failed; the reconciler will repair it", "session", sessionID, "user", user, "error", err)
        return fmt.Errorf("%w: %w", ErrSyncPending, err)
}

// syncRevoke deletes user's role tuple after the Mongo record is gone. The change has already taken
// effect (the member lost access when the record went), so a failure is logged and left to the
// reconciler instead of failing the request. The GM deletes all three member relations in one
// request, so a stale co_gm tuple left by a demotion whose write failed cannot outlive the removal;
// anyone else deletes only the member's own role tuple, which is all their auth-api rights cover.
func (s *Sessions) syncRevoke(ctx context.Context, actorRole, sessionID, user, role string) {
        if !slices.Contains(memberRoles, role) {
                return
        }
        deletes := []relations.Tuple{tupleOf(user, role)}
        if actorRole == data.RoleGM {
                deletes = deletes[:0]
                for _, r := range memberRoles {
                        deletes = append(deletes, tupleOf(user, r))
                }
        }
        a, err := authorityOf(ctx)
        if err == nil {
                err = a.svc.WriteTuples(ctx, a.token, sessionID, nil, deletes)
        }
        if err != nil {
                slog.WarnContext(ctx, "membership tuple delete failed; the reconciler will remove it", "session", sessionID, "user", user, "error", err)
        }
}

// endSession tells auth-api to drop every tuple on the session and close its mint. The Mongo
// records are already gone, so a failure is logged and left to garbage collection.
func (s *Sessions) endSession(ctx context.Context, sessionID string) {
        a, err := authorityOf(ctx)
        if err == nil {
                err = a.svc.EndSession(ctx, a.token, sessionID)
        }
        if err != nil {
                slog.WarnContext(ctx, "ending the session at auth-api failed; garbage collection will close it", "session", sessionID, "error", err)
        }
}

// canManageRole reports whether actorRole may assign or remove role. Only the GM changes a co_gm
// (PADR-0040): a co_gm calling auth-api directly could otherwise strip other co-GMs.
func canManageRole(actorRole, role string) bool {
        return role != data.RoleCoGM || actorRole == data.RoleGM
}

// NewSessions builds the service on db.

// Create makes actor the GM of a new session.
// Sessions beyond the GM's entitled count are refused with *EntitlementError; inactive sessions
// count, so a GM over a reduced limit cannot create.
func (s *Sessions) Create(ctx context.Context, actor, name string) (SessionView, error) {
        name = strings.TrimSpace(name)
        if name == "" || len([]rune(name)) > maxSessionName {
                return SessionView{}, invalid("session name is required and at most %d characters", maxSessionName)
        }
        chk, ok := ctx.Value(checkerKey{}).(Checker)
        if !ok {
                return SessionView{}, entitlements.ErrUnavailable
        }
        res, err := chk.Check(ctx, KeySessionsPerGM, entitlements.FailClosed)
        if err != nil {
                return SessionView{}, err
        }
        owned, err := s.repo.Count(ctx, actor)
        if err != nil {
                return SessionView{}, err
        }
        if owned >= int64(res.Limit) {
                return SessionView{}, &EntitlementError{Key: KeySessionsPerGM, Limit: res.Limit}
        }
        auth, err := authorityOf(ctx)
        if err != nil {
                return SessionView{}, err
        }
        id, err := auth.svc.Mint(ctx, auth.token)
        if err != nil {
                return SessionView{}, err
        }
        sess := data.Session{Name: name}
        if err := s.repo.Open(ctx, actor, id, &sess); err != nil {
                s.endSession(ctx, id) // frees the mint slot; garbage collection covers a failure here
                return SessionView{}, err
        }
        return SessionView{Session: sess, Role: data.RoleGM}, nil
}

// actionRelation is the OpenFGA relation on the session each action needs. The role rules in
// Allowed still apply on top: OpenFGA says whether the caller may touch the session at all.
var actionRelation = map[Action]string{
        ActView: "can_view", ActLeave: "can_view",
        ActInvite: "can_edit", ActBlock: "can_edit", ActManageMembers: "can_edit",
        ActDecideRequest: "can_approve",
        ActDelete:        "gm",
}

// as authorizes actor for action a on session id with auth-api (OpenFGA), then returns the
// session, actor's membership, and the decision the data layer needs. A denial is ErrForbidden and
// reads nothing; an unavailable decision is relations.ErrUnavailable (503), never an allow.
func (s *Sessions) as(ctx context.Context, actor, id string, a Action) (data.Session, data.Member, data.Decision, error) {
        d, err := data.Authorize(ctx, actor, id, actionRelation[a])
        if err != nil {
                return data.Session{}, data.Member{}, d, err
        }
        sess, err := s.repo.Live(ctx, d)
        if err != nil {
                return sess, data.Member{}, d, err
        }
        m, err := s.repo.Member(ctx, d, actor)
        if errors.Is(err, data.ErrNotFound) || (err == nil && !Allowed(m.Role, a)) {
                return data.Session{}, data.Member{}, d, data.ErrForbidden
        }
        if err != nil {
                return data.Session{}, data.Member{}, d, err
        }
        if m.Role == data.RoleGM {
                sess = s.refreshOwnerLimits(ctx, d, sess)
        }
        if sess.InactiveAt != nil && a != ActView && a != ActDelete && a != ActLeave {
                return sess, m, d, data.ErrInactive
        }
        return sess, m, d, nil
}

// refreshOwnerLimits re-reads the GM's entitlements with the GM's own token (the checker in ctx)
// and records them on the session. A failed, degraded, or stale (cached past its TTL) read keeps the old record, is logged, and
// never fails the request.
func (s *Sessions) refreshOwnerLimits(ctx context.Context, d data.Decision, sess data.Session) data.Session {
        chk, ok := ctx.Value(checkerKey{}).(Checker)
        if !ok {
                return sess
        }
        limits := make(map[string]int, len(sessionKeys))
        for _, key := range sessionKeys {
                res, err := chk.Check(ctx, key, entitlements.FailOpen)
                if err != nil || res.Degraded || res.Stale {
                        slog.WarnContext(ctx, "session owner entitlements not refreshed", "session", sess.ID, "key", key, "error", err)
                        return sess
                }
                limits[key] = res.Limit
        }
        fresh := sess.OwnerLimitsAt != nil && s.now().Sub(*sess.OwnerLimitsAt) < ownerLimitsRefreshEvery
        if fresh && maps.Equal(limits, sess.OwnerLimits) {
                return sess
        }
        if err := s.repo.RecordOwnerLimits(ctx, d, limits); err != nil {
                slog.WarnContext(ctx, "session owner entitlements not recorded", "session", sess.ID, "error", err)
                return sess
        }
        now := s.now()
        sess.OwnerLimits, sess.OwnerLimitsAt = limits, &now
        return sess
}

// Items exposes the sessions as limited items, so the limit reconciler can deactivate those beyond

// Get returns the session with actor's role. Only members may read it.
func (s *Sessions) Get(ctx context.Context, actor, id string) (SessionView, error) {
        sess, m, _, err := s.as(ctx, actor, id, ActView)
        return SessionView{Session: sess, Role: m.Role}, err

// List returns the sessions actor belongs to.
func (s *Sessions) List(ctx context.Context, actor string, q data.Query) (data.Page[SessionView], error) {
        page, err := s.repo.SessionIDsOf(ctx, actor, q)
        if err != nil {
                return data.Page[SessionView]{}, err
        }
        out := data.Page[SessionView]{NextCursor: page.NextCursor, Items: []SessionView{}}
        for _, m := range page.Items {
                d, err := data.Authorize(ctx, actor, m.SessionID, actionRelation[ActView])
                if errors.Is(err, data.ErrForbidden) || errors.Is(err, data.ErrNotFound) {
                        continue // membership record the tuples don't back (yet), or malformed
                }
                if err != nil {
                        return data.Page[SessionView]{}, err
                }
                sess, err := s.repo.Live(ctx, d)
                if errors.Is(err, data.ErrNotFound) {
                        continue // deleted while we listed
                }
                if err != nil {
                        return data.Page[SessionView]{}, err
                }
                out.Items = append(out.Items, SessionView{Session: sess, Role: m.Role})
        }
        return out, nil
}

// Delete soft-deletes the session and ends every membership. Only the GM may; the owner-scoped
// delete enforces it a second time.
func (s *Sessions) Delete(ctx context.Context, actor, id string, rev int64) error {
        _, _, d, err := s.as(ctx, actor, id, ActDelete)
        if err != nil {
                return err
        }
        if err := s.repo.SoftDelete(ctx, d, rev); err != nil {
                return err
        }
        // End the session at auth-api before dropping the member records: once the session record is
        // soft-deleted a retry 404s at Live(), so a skipped end would leak the mint and tuples until
        // garbage collection.
        s.endSession(ctx, d.SessionID())
        return s.repo.DropSession(ctx, d)
}

func hashCode(code string) string {
        sum := sha256.Sum256([]byte(strings.ToUpper(strings.TrimSpace(code))))
        return hex.EncodeToString(sum[:])
}

func newCode() (string, error) {
        b := make([]byte, 10) // 80 bits
        if _, err := rand.Read(b); err != nil {
                return "", err
        }
        return base32.StdEncoding.WithPadding(base32.NoPadding).EncodeToString(b), nil
}

// CreateInvite makes a join code for the session. The code is returned once; only its hash is kept.
// ttl 0 means seven days.
func (s *Sessions) CreateInvite(ctx context.Context, actor, id string, ttl time.Duration) (data.Invite, string, error) {
        _, _, d, err := s.as(ctx, actor, id, ActInvite)
        if err != nil {
                return data.Invite{}, "", err
        }
        if ttl == 0 {
                ttl = defaultInviteTTL
        }
        if ttl < minInviteTTL || ttl > maxInviteTTL {
                return data.Invite{}, "", invalid("invite lifetime must be between %s and %s", minInviteTTL, maxInviteTTL)
        }
        code, err := newCode()
        if err != nil {
                return data.Invite{}, "", err
        }
        in, err := s.repo.AddInvite(ctx, d, hashCode(code), s.now().Add(ttl))
        return in, code, err
}

// Join turns a presented code into a pending request. A pending request grants nothing. A code
// that is unknown, expired, revoked, or from a blocked user fails the same way and creates nothing.
func (s *Sessions) Join(ctx context.Context, actor, code string) (data.JoinRequest, error) {
        t, err := s.repo.ResolveInvite(ctx, actor, hashCode(code), s.now())
        if errors.Is(err, data.ErrNotFound) {
                return data.JoinRequest{}, ErrInviteInvalid
        }
        if err != nil {
                return data.JoinRequest{}, err
        }
        st, err := s.repo.ForJoin(ctx, t)
        if err != nil {
                return data.JoinRequest{}, err
        }
        if !st.Exists || !st.Active || st.Blocked {
                return data.JoinRequest{}, ErrInviteInvalid
        }
        if st.Member {
                return data.JoinRequest{}, data.ErrConflict
        }
        return s.repo.RequestJoin(ctx, t)
}

        if pendingOnly {

}

// Approve makes the requester a member with role. The request settles first, so two approvals
// cannot both succeed; a block placed after the request still stops it.
func (s *Sessions) Approve(ctx context.Context, actor, id, requestID, role string) (data.Member, error) {
        if !slices.Contains(assignable, role) {
                return data.Member{}, invalid("role must be one of %s", strings.Join(assignable, ", "))
        }
        _, am, d, err := s.as(ctx, actor, id, ActDecideRequest)
        if err != nil {
                return data.Member{}, err
        }
        if !canManageRole(am.Role, role) {
                return data.Member{}, data.ErrForbidden
        }
        jr, err := s.repo.Decide(ctx, d, requestID, data.RequestApproved, role, "")
        if errors.Is(err, data.ErrConflict) && jr.Status == data.RequestApproved && jr.Role == role {
                return s.finishApproval(ctx, d, am.Role, jr)
        }
        if err != nil {
                return data.Member{}, err
        }
        if blocked, err := s.repo.Blocked(ctx, d, jr.UserID); err != nil || blocked {
                if err == nil {
                        err = data.ErrForbidden
                }
                return data.Member{}, err
        }
        m, err := s.repo.AddMember(ctx, d, jr.UserID, role)
        if err != nil {
                return data.Member{}, err
        }
        return m, s.syncGrant(ctx, am.Role, d.SessionID(), m.UserID, role)
}

// finishApproval is the retry of an approval that settled but whose tuple write failed: the member
// record exists, so only the tuple is written again. Any other state is the original conflict.
func (s *Sessions) finishApproval(ctx context.Context, d data.Decision, actorRole string, jr data.JoinRequest) (data.Member, error) {
        m, err := s.repo.Member(ctx, d, jr.UserID)
        if err != nil || m.Role != jr.Role {
                return data.Member{}, data.ErrConflict
        }
        return m, s.syncGrant(ctx, actorRole, d.SessionID(), m.UserID, m.Role)
}

// Reject settles a pending request with a reason; the requester stays out.
func (s *Sessions) Reject(ctx context.Context, actor, id, requestID, reason string) (data.JoinRequest, error) {
        reason = strings.TrimSpace(reason)
        if reason == "" {
                return data.JoinRequest{}, invalid("a reason is required to reject a request")
        }
        _, _, d, err := s.as(ctx, actor, id, ActDecideRequest)
        if err != nil {
                return data.JoinRequest{}, err
        }
        return s.repo.Decide(ctx, d, requestID, data.RequestRejected, "", reason)
}

// SetRole changes a member's role. The gm role cannot be given or taken.
func (s *Sessions) SetRole(ctx context.Context, actor, id, user, role string) (data.Member, error) {
        if !slices.Contains(assignable, role) {
                return data.Member{}, invalid("role must be one of %s", strings.Join(assignable, ", "))
        }
        _, am, d, err := s.as(ctx, actor, id, ActManageMembers)
        if err != nil {
                return data.Member{}, err
        }
        t, err := s.target(ctx, d, user)
        if err != nil {
                return data.Member{}, err
        }
        if !canManageRole(am.Role, role) || !canManageRole(am.Role, t.Role) {
                return data.Member{}, data.ErrForbidden
        }
        m, err := s.repo.SetRole(ctx, d, user, role)
        if err != nil {
                return data.Member{}, err
        }
        return m, s.syncGrant(ctx, am.Role, d.SessionID(), user, role)
}

// Remove ends a member's access.
func (s *Sessions) Remove(ctx context.Context, actor, id, user string) error {
        _, am, d, err := s.as(ctx, actor, id, ActManageMembers)
        if err != nil {
                return err
        }
        t, err := s.target(ctx, d, user)
        if err != nil {
                return err
        }
        if !canManageRole(am.Role, t.Role) {
                return data.ErrForbidden
        }
        if err := s.repo.RemoveMember(ctx, d, user); err != nil {
                return err
        }
        s.syncRevoke(ctx, am.Role, d.SessionID(), user, t.Role)
        return nil
}

// Leave removes actor from the session. The GM cannot leave; they delete the session.
func (s *Sessions) Leave(ctx context.Context, actor, id string) error {
        _, m, d, err := s.as(ctx, actor, id, ActView)
        if err != nil {
                return err
        }
        if !Allowed(m.Role, ActLeave) {
                return invalid("the GM cannot leave; delete the session instead")
        }
        if err := s.repo.RemoveMember(ctx, d, actor); err != nil {
                return err
        }
        s.syncRevoke(ctx, m.Role, d.SessionID(), actor, m.Role)
        return nil
}

// Block keeps user out of the session, removing them and any pending request.
func (s *Sessions) Block(ctx context.Context, actor, id, user string) error {
        _, am, d, err := s.as(ctx, actor, id, ActBlock)
        if err != nil {
                return err
        }
        if user == actor {
                return invalid("you cannot block yourself")
        }
        t, err := s.target(ctx, d, user)
        if err != nil && !errors.Is(err, data.ErrNotFound) {
                return err
        }
        if err == nil && !canManageRole(am.Role, t.Role) {
                return data.ErrForbidden
        }
        if err := s.repo.AddBlock(ctx, d, user); err != nil {
                return err
        }
        if err := s.repo.DropRequest(ctx, d, user); err != nil {
                return err
        }
        switch err := s.repo.RemoveMember(ctx, d, user); {
        case err == nil:
                s.syncRevoke(ctx, am.Role, d.SessionID(), user, t.Role)
        case !errors.Is(err, data.ErrNotFound):
                return err
        }
        return nil
}

// Unblock lets user join again.
func (s *Sessions) Unblock(ctx context.Context, actor, id, user string) error {
        _, _, d, err := s.as(ctx, actor, id, ActBlock)
        if err != nil {
                return err
        }
        return s.repo.RemoveBlock(ctx, d, user)
}

}

// KeyThemeSlots is the entitlement limit on custom theme slots (docs/entitlements-check-contract.md).
const KeyThemeSlots = "initiative.theme_slots"

// PackagedPrefix starts the ID of every packaged theme. Custom theme IDs never carry it.
const PackagedPrefix = "pkg-"

const slotRetries = 3

var (
        // ErrReadOnly means the caller tried to change or delete a packaged theme.
        ErrReadOnly = errors.New("domain: packaged theme is read-only")

        hexColor = regexp.MustCompile(`^#[0-9A-Fa-f]{6}([0-9A-Fa-f]{2})?$`)
)

//go:embed packaged_themes.json
var packagedJSON []byte

// EntitlementError is the typed denial for a stored-item limit the caller has reached.
type EntitlementError struct {
        Key   string
        Limit int
}

func (e *EntitlementError) Error() string {
        return fmt.Sprintf("domain: entitlement limit reached for %s (%d)", e.Key, e.Limit)
}

// Checker answers entitlement checks for the calling user; the server binds the user and token.
type Checker interface {
        Check(ctx context.Context, key string, policy entitlements.Policy) (entitlements.Result, error)
}

// ThemeInput is the full field set of a custom theme.
type ThemeInput struct {
        Name, Style, MainColor, Accent1Color, Accent2Color, Background1Color, Background2Color string
        PrimaryFontName, SecondaryFontName, BackgroundImage                                    string
}

// ThemePatch changes the fields that are non-nil.
type ThemePatch struct {
        Name, Style, MainColor, Accent1Color, Accent2Color, Background1Color, Background2Color *string
        PrimaryFontName, SecondaryFontName, BackgroundImage                                    *string
}

// Themes implements custom theme rules: slot enforcement and read-only packaged themes.
type Themes struct {
        themes   *data.Owned[data.Theme, *data.Theme]
        packaged []data.Theme
}

// NewThemes builds the theme service on db.
func NewThemes(db store.Database) (*Themes, error) {
        var pk []data.Theme
        if err := json.Unmarshal(packagedJSON, &pk); err != nil {
                return nil, fmt.Errorf("packaged themes: %w", err)
        }
        return &Themes{themes: data.NewOwned[data.Theme](db, data.ThemeSpec), packaged: pk}, nil
}

// Packaged returns the read-only packaged themes.
func (s *Themes) Packaged() []data.Theme { return append([]data.Theme(nil), s.packaged...) }

func isPackaged(id string) bool { return strings.HasPrefix(id, PackagedPrefix) }

// Get returns a packaged theme by ID or owner's custom theme.
func (s *Themes) Get(ctx context.Context, owner, id string) (data.Theme, error) {
        if isPackaged(id) {
                for _, t := range s.packaged {
                        if t.ID == id {
                                return t, nil
                        }
                }
                return data.Theme{}, data.ErrNotFound
        }
        return s.themes.Get(ctx, owner, id)
}

// List returns a page of owner's custom themes in slot order.
func (s *Themes) List(ctx context.Context, owner string, q data.Query) (data.Page[data.Theme], error) {
        if len(q.Sort) == 0 {
                q.Sort = []data.SortKey{{Field: "slot"}}
        }
        return s.themes.List(ctx, owner, q)
}

// Create stores a custom theme in the lowest free slot below the entitled limit. At or over the
// limit it returns *EntitlementError; if entitlements-api is unreachable, entitlements.ErrUnavailable.
func (s *Themes) Create(ctx context.Context, owner, actor string, in ThemeInput, chk Checker) (data.Theme, error) {
        t, err := buildTheme(in)
        if err != nil {
                return data.Theme{}, err
        }
        res, err := chk.Check(ctx, KeyThemeSlots, entitlements.FailClosed)
        if err != nil {
                return data.Theme{}, err
        }
        // Inactive themes count toward the total owned, so a user over a reduced limit cannot create.
        owned, err := s.themes.Count(ctx, owner, nil)
        if err != nil {
                return data.Theme{}, err
        }
        if owned >= int64(res.Limit) {
                return data.Theme{}, &EntitlementError{Key: KeyThemeSlots, Limit: res.Limit}
        }
        for range slotRetries {
                slot, err := s.freeSlot(ctx, owner, res.Limit)
                if err != nil {
                        return data.Theme{}, err
                }
                t.Slot = &slot
                err = s.themes.Create(ctx, owner, actor, &t)
                if !errors.Is(err, data.ErrConflict) {
                        return t, err
                }
        }
        return data.Theme{}, data.ErrConflict
}

// freeSlot returns the lowest unused slot in [0, limit), or *EntitlementError when all are used.
func (s *Themes) freeSlot(ctx context.Context, owner string, limit int) (int, error) {
        page, err := s.themes.List(ctx, owner, data.Query{
                Sort: []data.SortKey{{Field: "slot"}}, Fields: []string{"slot"}, Limit: limit,
        })
        if err != nil {
                return 0, err
        }
        used := map[int]bool{}
        for _, t := range page.Items {
                if t.Slot != nil {
                        used[*t.Slot] = true
                }
        }
        for i := range limit {
                if !used[i] {
                        return i, nil
                }
        }
        return 0, &EntitlementError{Key: KeyThemeSlots, Limit: limit}
}

// Update changes owner's custom theme, compare-and-set on rev. Packaged themes return ErrReadOnly.
func (s *Themes) Update(ctx context.Context, owner, actor, id string, rev int64, p ThemePatch) (data.Theme, error) {
        if isPackaged(id) {
                return data.Theme{}, ErrReadOnly
        }
        set, err := patchSet(p)
        if err != nil {
                return data.Theme{}, err
        }
        return s.themes.Update(ctx, owner, actor, id, rev, set)
}

// Delete soft-deletes owner's custom theme, freeing its slot. Packaged themes return ErrReadOnly.
func (s *Themes) Delete(ctx context.Context, owner, actor, id string, rev int64) error {
        if isPackaged(id) {
                return ErrReadOnly
        }
        return s.themes.SoftDelete(ctx, owner, actor, id, rev)
}

// Items is the theme repository as limit enforcement sees it.
func (s *Themes) Items() Limited { return s.themes }

// Count returns how many custom themes owner holds.
func (s *Themes) Count(ctx context.Context, owner string) (int64, error) {
        return s.themes.Count(ctx, owner, nil)
}

func buildTheme(in ThemeInput) (data.Theme, error) {
        set, err := patchSet(ThemePatch{
                Name: &in.Name, Style: &in.Style, MainColor: &in.MainColor, Accent1Color: &in.Accent1Color,
                Accent2Color: &in.Accent2Color, Background1Color: &in.Background1Color,
                Background2Color: &in.Background2Color, PrimaryFontName: &in.PrimaryFontName,
                SecondaryFontName: &in.SecondaryFontName, BackgroundImage: &in.BackgroundImage,
        })
        if err != nil {
                return data.Theme{}, err
        }
        str := func(k string) string { v, _ := set[k].(string); return v }
        return data.Theme{
                Name: str("name"), Style: str("style"), MainColor: str("main_color"),
                Accent1Color: str("accent1_color"), Accent2Color: str("accent2_color"),
                Background1Color: str("background1_color"), Background2Color: str("background2_color"),
                PrimaryFontName: str("primary_font_name"), SecondaryFontName: str("secondary_font_name"),
                BackgroundImage: str("background_image"),
        }, nil
}

type themeField struct {
        key  string
        val  *string
        kind int // 0 required text, 1 color, 2 optional https URL
}

const (
        fText = iota
        fColor
        fImage
)

// patchSet validates the non-nil fields of p and returns them keyed by stored field name.
func patchSet(p ThemePatch) (map[string]any, error) {
        fields := []themeField{
                {"name", p.Name, fText}, {"style", p.Style, fText},
                {"main_color", p.MainColor, fColor}, {"accent1_color", p.Accent1Color, fColor},
                {"accent2_color", p.Accent2Color, fColor}, {"background1_color", p.Background1Color, fColor},
                {"background2_color", p.Background2Color, fColor},
                {"primary_font_name", p.PrimaryFontName, fText}, {"secondary_font_name", p.SecondaryFontName, fText},
                {"background_image", p.BackgroundImage, fImage},
        }
        set := map[string]any{}
        for _, f := range fields {
                if f.val == nil {
                        continue
                }
                v := strings.TrimSpace(*f.val)
                if err := checkThemeField(f, v); err != nil {
                        return nil, err
                }
                set[f.key] = v
        }
        return set, nil
}

func checkThemeField(f themeField, v string) error {
        switch f.kind {
        case fText:
                if v == "" || len(v) > 100 {
                        return invalid("theme %s is required and at most 100 characters", f.key)
                }
        case fColor:
                if !hexColor.MatchString(v) {
                        return invalid("theme %s must be #RRGGBB or #RRGGBBAA", f.key)
                }
        case fImage:
                if v == "" {
                        return nil
                }
                if u, err := url.Parse(v); err != nil || u.Scheme != "https" || u.Host == "" || len(v) > 2048 {
                        return invalid("theme %s must be an https URL", f.key)
                }
        }
        return nil
}

// CreateEncounterFromRoster stores enc as owner's new encounter with one combatant per roster
// character, copying name, kind, notes, default health, and external refs. The source roster is
// not modified, and enc's own combatants are replaced.
func (s *Rosters) CreateEncounterFromRoster(ctx context.Context, owner, actor, rosterID string, enc data.Encounter) (data.Encounter, error) {
        roster, err := s.rosters.Get(ctx, owner, rosterID)
        if err != nil {
                return data.Encounter{}, err
        }
        enc.Combatants = make([]data.Combatant, 0, len(roster.Characters))
        for _, c := range roster.Characters {
                enc.Combatants = append(enc.Combatants, data.Combatant{
                        ID: s.newID(), Name: c.Name, Kind: c.Kind, Notes: c.Notes,
                        Health: maps.Clone(c.Health), ExternalRefs: maps.Clone(c.ExternalRefs),
                        Conditions: []data.Condition{}, SourceCharacterID: c.ID,
                })
        }
        if err := s.encounters.Create(ctx, owner, actor, &enc); err != nil {
                return data.Encounter{}, err
        }
        return enc, nil
}

// SaveAsRoster stores a new roster holding one character per combatant of owner's encounter that
// is not removed. Current health becomes the character's defaults. The encounter is not modified.
// An empty name falls back to the encounter's name.
func (s *Rosters) SaveAsRoster(ctx context.Context, owner, actor, encounterID, name string) (data.Roster, error) {
        enc, err := s.encounters.Get(ctx, owner, encounterID)
        if err != nil {
                return data.Roster{}, err
        }
        if strings.TrimSpace(name) == "" {
                name = enc.Name
        }
        chars := make([]data.Character, 0, len(enc.Combatants))
        for _, c := range enc.Combatants {
                if c.Removed {
                        continue
                }
                chars = append(chars, data.Character{
                        Name: c.Name, Kind: c.Kind, Notes: c.Notes, Health: c.Health, ExternalRefs: c.ExternalRefs,
                })
        }
        return s.Create(ctx, owner, actor, name, false, chars)
}

// Start moves a draft encounter to active at round 1, builds the turn queue, and makes the first
// combatant current (its conditions tick as its turn begins).
func (s *Encounters) Start(ctx context.Context, owner, actor, id string, rev int64) (data.Encounter, error) {
        return s.editQueue(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                if st.Status != data.StatusDraft {
                        return fmt.Errorf("%w: cannot start from %s", ErrTransition, st.Status)
                }
                st.Status, st.Round = data.StatusActive, 1
                return nil
        })
}

// End moves an active encounter to ended.
func (s *Encounters) End(ctx context.Context, owner, actor, id string, rev int64) (data.Encounter, error) {
        return s.edit(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                if st.Status != data.StatusActive {
                        return fmt.Errorf("%w: cannot end from %s", ErrTransition, st.Status)
                }
                st.Status = data.StatusEnded
                return nil
        })
}

// Reset returns the encounter to draft at round 0 with no current combatant and an empty turn queue, clears every
// combatant's conditions, and sets health values back to the game system's defaults. Combatants,
// their initiative, and removed marks are kept.
func (s *Encounters) Reset(ctx context.Context, owner, actor, id string, rev int64) (data.Encounter, error) {
        cur, err := s.repo.Load(ctx, owner, id)
        if err != nil {
                return data.Encounter{}, err
        }
        sys, err := s.systems.GetOrDefault(ctx, cur.GameSystemID)
        if errors.Is(err, data.ErrNotFound) {
                return data.Encounter{}, invalid("the encounter's game system has no definition")
        }
        if err != nil {
                return data.Encounter{}, err
        }
        defaults := make(map[string]int, len(sys.Fields))
        for _, f := range sys.Fields {
                defaults[f.Key] = f.Def
        }
        return s.edit(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                st.Status, st.Round, st.CurrentCombatantID, st.TurnQueue = data.StatusDraft, 0, "", nil
                for i := range st.Combatants {
                        c := &st.Combatants[i]
                        c.Conditions = []data.Condition{}
                        c.Health = maps.Clone(c.Health)
                        for k := range c.Health {
                                if d, ok := defaults[k]; ok {
                                        c.Health[k] = d
                                }
                        }
                }
                return nil
        })
}

// effectiveOrdering is the ordering the engine applies. No System orders by initiative value only,
// whatever the stored ordering says: pcVsAdversary grouping needs a system that defines it, and
// none of the current definitions do.
func effectiveOrdering(ordering, gameSystemID string) string {
        if ordering == data.OrderPCVsAdversary && gameSystemID == data.DefaultSystemID {
                return data.OrderHighToLow
        }
        return ordering
}

// recompute rebuilds the turn queue under ordering and tie. In an active encounter it also keeps
// a current combatant: if the current one left the queue, the combatant now at its old position
// takes the turn (wrapping to the front and starting the next round if it was last), and an empty
// current picks the first.
func (s *Encounters) recompute(st *data.EncounterState, ordering, tie string) {
        prev := st.TurnQueue
        st.TurnQueue = buildQueue(ordering, tie, st.Combatants, prev, s.perm)
        if st.Status != data.StatusActive || slices.Contains(st.TurnQueue, st.CurrentCombatantID) {
                return
        }
        if len(st.TurnQueue) == 0 {
                st.CurrentCombatantID = ""
                return
        }
        old := slices.Index(prev, st.CurrentCombatantID)
        pos := max(old, 0)
        if old >= len(st.TurnQueue) {
                st.Round++ // the removed combatant was last: the turn wraps to the front, a new round
        }
        st.CurrentCombatantID = st.TurnQueue[pos%len(st.TurnQueue)]
        beginTurn(st)
}

// beginTurn ticks the conditions of the current combatant.
func beginTurn(st *data.EncounterState) {
        i := slices.IndexFunc(st.Combatants, func(c data.Combatant) bool { return c.ID == st.CurrentCombatantID })
        if i >= 0 {
                expireConditions(&st.Combatants[i])
        }
}

func queuedTurn(st *data.EncounterState) (pos int, err error) {
        if st.Status != data.StatusActive {
                return 0, fmt.Errorf("%w: encounter is %s", ErrTransition, st.Status)
        }
        if len(st.TurnQueue) == 0 {
                return 0, invalid("the turn queue is empty")
        }
        return slices.Index(st.TurnQueue, st.CurrentCombatantID), nil
}

// Advance passes the turn to the next combatant in the queue. Past the last entry it wraps to the
// first and starts the next round. The new current combatant's timed conditions lose one round,
// and expire at zero.
func (s *Encounters) Advance(ctx context.Context, owner, actor, id string, rev int64) (data.Encounter, error) {
        return s.edit(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                pos, err := queuedTurn(st)
                if err != nil {
                        return err
                }
                idx, wrapped := nextTurn(pos, len(st.TurnQueue))
                if wrapped {
                        st.Round++
                }
                st.CurrentCombatantID = st.TurnQueue[idx]
                beginTurn(st)
                return nil
        })
}

// StepBack returns the turn to the previous combatant. Stepping back past the first entry goes to
// the last and the previous round; it is rejected on the first turn of round 1. Conditions that
// already ticked are not restored.
func (s *Encounters) StepBack(ctx context.Context, owner, actor, id string, rev int64) (data.Encounter, error) {
        return s.edit(ctx, owner, actor, id, rev, func(st *data.EncounterState, _ data.Encounter) error {
                pos, err := queuedTurn(st)
                if err != nil {
                        return err
                }
                idx, wrapped := prevTurn(pos, len(st.TurnQueue))
                if wrapped {
                        if st.Round <= 1 {
                                return fmt.Errorf("%w: already at the first turn", ErrTransition)
                        }
                        st.Round--
                }
                st.CurrentCombatantID = st.TurnQueue[idx]
                return nil
        })
}

// Package entitlements is the client for entitlements-api's check routes
// (docs/entitlements-check-contract.md in the platform repo, PADR-0034).
//
// The caller's own bearer token is forwarded on every call (PADR-0037, no service tokens), so a
// check only answers for the user holding that token. Results are cached per (user, key) for the
// TTL the service returns. When the service cannot be reached, FailClosed callers (creates and
// expansions) get ErrUnavailable and FailOpen callers (reads and edits) get the last known value
// (at most maxStale past its TTL) or a Degraded result they must not gate on.
package entitlements

import (
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "net/url"
        "sync"
        "time"
)

const (
        // maxTTL bounds a service-supplied TTL so a bad value cannot pin a stale entitlement.
        maxTTL = 5 * time.Minute
        // maxEntries bounds the cache; past it, expired entries go first, then everything.
        maxEntries = 10000
        // maxStale bounds how long FailOpen may serve an expired value during an outage; past it the
        // result is Degraded and the caller skips gating.
        maxStale    = time.Hour
        callTimeout = 3 * time.Second
        maxBody     = 64 << 10
)

// Policy says what a check does when entitlements-api cannot be reached.
type Policy int

const (
        // FailClosed returns ErrUnavailable. Use it for creates and expansions.
        FailClosed Policy = iota
        // FailOpen returns the last known value, or a Degraded result. Use it for reads and edits.
        FailOpen
)

var (
        // ErrUnavailable means entitlements-api could not be reached or answered with a server error.
        ErrUnavailable = errors.New("entitlements: service unavailable")
        // ErrUnauthorized means the service rejected the user's token; the caller returns 401.
        ErrUnauthorized = errors.New("entitlements: token rejected")
        // ErrUnknownKey means the key is not in the catalog: a programming error, never defaulted.
        ErrUnknownKey = errors.New("entitlements: key not registered")
)

// Result is one checked key. Flag is set for type "flag", Limit for type "limit".
type Result struct {
        Key   string
        Type  string
        Flag  bool
        Limit int
        // Degraded is true when no value is known because the service was down under FailOpen.
        // Flag and Limit are meaningless then; the caller skips gating.
        Degraded bool
        // Stale is true when the value is an expired cached one served because the service was down.
        // It is usable for gating, but a caller that records the value must not treat it as a fresh read.
        Stale bool
}

type entry struct {
        res     Result
        expires time.Time
}

// Client calls entitlements-api. The zero value is not usable; use New.
type Client struct {
        base string
        http *http.Client
        now  func() time.Time

        mu    sync.Mutex
        cache map[string]entry
}

// New returns a Client for the entitlements-api base URL.
func New(baseURL string) *Client {
        return &Client{
                base:  baseURL,
                http:  &http.Client{Timeout: callTimeout},
                now:   time.Now,
                cache: map[string]entry{},
        }
}

// Check returns the user's value for key. user is the verified token subject (the cache key);
// token is that user's bearer token, forwarded to the service.
func (c *Client) Check(ctx context.Context, user, token, key string, policy Policy) (Result, error) {
        ck := user + "\x00" + key
        if res, ok := c.fresh(ck); ok {
                return res, nil
        }

        res, ttl, err := c.fetch(ctx, token, key)
        switch {
        case err == nil:
                c.store(ck, res, ttl)
                return res, nil
        case errors.Is(err, ErrUnavailable) && policy == FailOpen:
                if res, ok := c.stale(ck); ok {
                        res.Stale = true
                        return res, nil
                }
                return Result{Key: key, Degraded: true}, nil
        default:
                if errors.Is(err, ErrUnknownKey) {
                        slog.ErrorContext(ctx, "entitlement key not registered", "key", key)
                }
                return Result{}, err
        }
}

func (c *Client) fetch(ctx context.Context, token, key string) (Result, time.Duration, error) {
        var body checkBody
        if err := c.get(ctx, token, "/check?key="+url.QueryEscape(key), &body); err != nil {
                return Result{}, 0, err
        }
        return body.result(key)
}

// Prefix returns the user's values for every key under prefix in one call and caches each. It
// has no stale fallback: any failure is returned, for callers that only display the values.
func (c *Client) Prefix(ctx context.Context, user, token, prefix string) ([]Result, error) {
        var body struct {
                Results []checkBody `json:"results"`
        }
        if err := c.get(ctx, token, "/check/batch?prefix="+url.QueryEscape(prefix), &body); err != nil {
                return nil, err
        }
        out := make([]Result, 0, len(body.Results))
        for _, b := range body.Results {
                res, ttl, err := b.result(b.Key)
                if err != nil {
                        return nil, err
                }
                c.store(user+"\x00"+b.Key, res, ttl)
                out = append(out, res)
        }
        return out, nil
}

// get calls path with the user's token and decodes a 200 body into out.
func (c *Client) get(ctx context.Context, token, path string, out any) error {
        req, err := http.NewRequestWithContext(ctx, http.MethodGet, c.base+path, nil)
        if err != nil {
                return fmt.Errorf("entitlements: build request: %w", err)
        }
        req.Header.Set("Authorization", "Bearer "+token)
        req.Header.Set("Accept", "application/json")

        resp, err := c.http.Do(req)
        if err != nil {
                slog.WarnContext(ctx, "entitlements call failed", "path", path, "error", err)
                return ErrUnavailable
        }
        defer func() { _ = resp.Body.Close() }()

        switch resp.StatusCode {
        case http.StatusOK:
                if err := json.NewDecoder(io.LimitReader(resp.Body, maxBody)).Decode(out); err != nil {
                        return ErrUnavailable
                }
                return nil
        case http.StatusUnauthorized:
                return ErrUnauthorized
        case http.StatusNotFound:
                return ErrUnknownKey
        default:
                slog.WarnContext(ctx, "entitlements call rejected", "path", path, "status", resp.StatusCode)
                return ErrUnavailable
        }
}

type checkBody struct {
        Key   string `json:"key"`
        Type  string `json:"type"`
        Value any    `json:"value"`
        TTL   int    `json:"ttl_seconds"`
}

func (b checkBody) result(key string) (Result, time.Duration, error) {
        res := Result{Key: key, Type: b.Type}
        switch v := b.Value.(type) {
        case bool:
                res.Flag = v
        case float64:
                if v < 0 || v != float64(int(v)) {
                        return Result{}, 0, ErrUnavailable
                }
                res.Limit = int(v)
        default:
                return Result{}, 0, ErrUnavailable
        }
        return res, min(time.Duration(b.TTL)*time.Second, maxTTL), nil
}

func (c *Client) fresh(ck string) (Result, bool) {
        c.mu.Lock()
        defer c.mu.Unlock()
        e, ok := c.cache[ck]
        if !ok || !c.now().Before(e.expires) {
                return Result{}, false
        }
        return e.res, true
}

func (c *Client) stale(ck string) (Result, bool) {
        c.mu.Lock()
        defer c.mu.Unlock()
        e, ok := c.cache[ck]
        return e.res, ok && c.now().Before(e.expires.Add(maxStale))
}

func (c *Client) store(ck string, res Result, ttl time.Duration) {
        if ttl <= 0 {
                return
        }
        c.mu.Lock()
        defer c.mu.Unlock()
        if len(c.cache) >= maxEntries {
                now := c.now()
                for k, e := range c.cache {
                        if !now.Before(e.expires) {
                                delete(c.cache, k)
                        }
                }
                if len(c.cache) >= maxEntries {
                        c.cache = map[string]entry{}
                }
        }
        c.cache[ck] = entry{res: res, expires: c.now().Add(ttl)}
}

// Package events consumes auth-api's change events from the platform bus (PADR-0039): a deny or
// a role removal ends the subject's integration credentials. The event carries no token and the
// consumer holds no service credential (PADR-0037).
package events

import (
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "log/slog"
        "os"
        "time"

        "github.com/nats-io/nats.go"
        "github.com/nats-io/nats.go/jetstream"
        "github.com/sweetrpg/initiative-api/data"
)

const (
        // StreamName is auth-api's JetStream stream, subject auth.events.>.
        StreamName     = "AUTH_EVENTS"
        FilterSubject  = "auth.events.>"
        defaultDurable = "initiative-api-credential-revocation"
        // Change kinds the event's action carries.
        kindDeny        = "deny"
        kindRoleRemoved = "role_removed"
)

// event is the platform envelope; entity_id is the Auth0 subject and action the change kind.
// Unknown fields are ignored so the envelope can grow.
type event struct {
        EventID  string `json:"event_id"`
        EntityID string `json:"entity_id"`
        Action   string `json:"action"`
        Data     struct {
                Service string `json:"service"`
        } `json:"data"`
}

// ServiceID is the name auth-api's deny entries use for this service. A deny of any other service
// says nothing about initiative credentials.
const ServiceID = "initiative-api"

// Revoker ends a subject's credentials. domain.Credentials implements it.
type Revoker interface {
        RevokeOwner(ctx context.Context, owner, reason string) (int, error)
}

// Gate is the on/off switch for integration credentials. domain.Credentials implements it.
type Gate interface {
        Open(ctx context.Context) (bool, error)
        Close(ctx context.Context) error
}

// handle applies one message. A nil error means ack it; an unparseable or irrelevant message is
// acked (redelivery cannot fix it), a failed revoke is returned so the bus redelivers.
func handle(ctx context.Context, rev Revoker, body []byte) error {
        var ev event
        if err := json.Unmarshal(body, &ev); err != nil {
                slog.Error("dropping unparseable auth event", "error", err)
                return nil
        }
        reason := map[string]string{kindDeny: data.ReasonOwnerDenied, kindRoleRemoved: data.ReasonOwnerRoleRemoved}[ev.Action]
        if reason == "" || ev.EntityID == "" || (ev.Action == kindDeny && ev.Data.Service != ServiceID) {
                return nil
        }
        n, err := rev.RevokeOwner(ctx, ev.EntityID, reason)
        if err != nil {
                return fmt.Errorf("revoke credentials for event %s: %w", ev.EventID, err)
        }
        slog.Info("revoked integration credentials", "event_id", ev.EventID, "reason", reason, "count", n)
        return nil
}

// behindRetention reports whether a consumer that last acknowledged ackFloor can no longer replay
// the gap because the stream's oldest retained message is firstSeq. A consumer that never existed
// has nothing to miss: credentials are not issued before it is bound.
func behindRetention(exists bool, ackFloor, firstSeq uint64) bool {
        return exists && firstSeq > 1 && ackFloor+1 < firstSeq
}

// Consumer runs the durable pull consumer on AUTH_EVENTS.
type Consumer struct {
        rev  Revoker
        gate Gate
        // Stream and Filter default to AUTH_EVENTS and auth.events.>; tests point them at a prefixed stream.
        Stream, Filter string
        conn           *nats.Conn
        cc             jetstream.ConsumeContext
}

// New builds a consumer. The gate must start closed.
func New(rev Revoker, gate Gate) *Consumer {
        return &Consumer{rev: rev, gate: gate, Stream: StreamName, Filter: FilterSubject}
}

// Start connects, binds the durable consumer, and begins revoking. The gate opens only after the
// consumer has drained what was pending, and stays closed when the consumer fell behind the
// stream's retention (until an administrator's reconcile reopens it).
func (c *Consumer) Start(ctx context.Context, natsURL, durable string) error {
        if durable == "" {
                durable = defaultDurable
        }
        opts := []nats.Option{nats.Name(durable)}
        if creds := os.Getenv("NATS_CREDS"); creds != "" {
                opts = append(opts, nats.UserCredentials(creds))
        } else if user := os.Getenv("NATS_USER"); user != "" {
                opts = append(opts, nats.UserInfo(user, os.Getenv("NATS_PASSWORD")))
        }
        conn, err := nats.Connect(natsURL, opts...)
        if err != nil {
                return fmt.Errorf("connect to NATS: %w", err)
        }
        c.conn = conn
        js, err := jetstream.New(conn)
        if err != nil {
                return fmt.Errorf("create JetStream context: %w", err)
        }
        stream, err := js.Stream(ctx, c.Stream)
        if err != nil {
                return fmt.Errorf("stream %s: %w", c.Stream, err)
        }
        sinfo, err := stream.Info(ctx)
        if err != nil {
                return fmt.Errorf("stream info: %w", err)
        }
        prior, err := stream.Consumer(ctx, durable)
        exists := err == nil
        if err != nil && !errors.Is(err, jetstream.ErrConsumerNotFound) {
                return fmt.Errorf("look up consumer %q: %w", durable, err)
        }
        if exists {
                pinfo, ierr := prior.Info(ctx)
                if ierr != nil {
                        return fmt.Errorf("consumer info: %w", ierr)
                }
                if behindRetention(true, pinfo.AckFloor.Stream, sinfo.State.FirstSeq) {
                        slog.Error("auth-event consumer is behind stream retention; integration credentials stay closed until reconcile")
                        if err := c.gate.Close(ctx); err != nil {
                                return fmt.Errorf("close credential gate: %w", err)
                        }
                }
        }
        cons, err := stream.CreateOrUpdateConsumer(ctx, jetstream.ConsumerConfig{
                Durable: durable, FilterSubject: c.Filter, AckPolicy: jetstream.AckExplicitPolicy,
        })
        if err != nil {
                return fmt.Errorf("bind durable consumer %q: %w", durable, err)
        }
        cc, err := cons.Consume(func(msg jetstream.Msg) { c.process(ctx, msg) })
        if err != nil {
                return fmt.Errorf("start consuming: %w", err)
        }
        c.cc = cc
        go c.openWhenCaughtUp(ctx, cons)
        return nil
}

func (c *Consumer) process(ctx context.Context, msg jetstream.Msg) {
        if err := handle(ctx, c.rev, msg.Data()); err != nil {
                slog.Error("auth event handling failed, will retry", "error", err)
                _ = msg.Nak()
                return
        }
        _ = msg.Ack()
}

// openWhenCaughtUp polls until nothing is pending or awaiting ack, then opens the gate.
func (c *Consumer) openWhenCaughtUp(ctx context.Context, cons jetstream.Consumer) {
        t := time.NewTicker(time.Second)
        defer t.Stop()
        for {
                select {
                case <-ctx.Done():
                        return
                case <-t.C:
                }
                info, err := cons.Info(ctx)
                if err != nil || info.NumPending > 0 || info.NumAckPending > 0 {
                        continue
                }
                open, err := c.gate.Open(ctx)
                if err != nil {
                        slog.Error("open credential gate", "error", err)
                        continue
                }
                slog.Info("integration credentials caught up", "open", open)
                return
        }
}

// Stop halts consumption and closes the connection.
func (c *Consumer) Stop() {
        if c.cc != nil {
                c.cc.Stop()
        }
        if c.conn != nil {
                c.conn.Close()
        }
}

// AuthAPI calls auth-api's internal tuple endpoints with the reconciler's own bearer

type AuthAPI struct {

        Token string
        HTTP  *http.Client

// NewAuthAPI returns a client for base using token.
func NewAuthAPI(base, token string) *AuthAPI {
        return &AuthAPI{Base: base, Token: token, HTTP: &http.Client{Timeout: callTimeout}}
}

// Read returns every tuple on object, following continuation tokens.
func (a *AuthAPI) Read(ctx context.Context, object string) ([]Tuple, error) {
        var all []Tuple
        token := ""
        for {
                q := url.Values{"object": {object}, "page_size": {fmt.Sprint(readPage)}}
                if token != "" {
                        q.Set("continuation_token", token)
                }
                var page struct {
                        Tuples []Tuple `json:"tuples"`
                        Next   string  `json:"continuation_token"`
                }
                if err := a.do(ctx, http.MethodGet, "/internal/openfga/tuples?"+q.Encode(), nil, &page); err != nil {
                        return nil, err
                }
                all = append(all, page.Tuples...)
                if page.Next == "" {
                        return all, nil
                }
                token = page.Next
        }
}

// Write applies one batch (at most 100 tuples in total) atomically.
func (a *AuthAPI) Write(ctx context.Context, writes, deletes []Tuple) error {
        body, err := json.Marshal(map[string][]Tuple{"writes": writes, "deletes": deletes})
        if err != nil {
                return err
        }
        return a.do(ctx, http.MethodPost, "/internal/openfga/tuples", body, nil)
}

func (a *AuthAPI) do(ctx context.Context, method, path string, body []byte, out any) error {
        req, err := http.NewRequestWithContext(ctx, method, a.Base+path, bytes.NewReader(body))
        if err != nil {
                return err
        }
        req.Header.Set("Authorization", "Bearer "+a.Token)
        req.Header.Set("Content-Type", "application/json")
        resp, err := a.HTTP.Do(req)
        if err != nil {
                return fmt.Errorf("auth-api %s: %w", path, err)
        }
        defer func() { _ = resp.Body.Close() }()
        switch {
        case resp.StatusCode == http.StatusUnauthorized || resp.StatusCode == http.StatusForbidden:
                return ErrTokenRejected
        case resp.StatusCode != http.StatusOK && resp.StatusCode != http.StatusNoContent:
                return fmt.Errorf("auth-api %s: status %d", path, resp.StatusCode)
        case out == nil:
                return nil
        }
        return json.NewDecoder(io.LimitReader(resp.Body, maxBody)).Decode(out)
}

// ListMints returns one page of open mints at least minAge old (by auth-api's clock) after the

func (a *AuthAPI) ListMints(ctx context.Context, minAge time.Duration, after string) ([]Mint, string, error) {
        q := url.Values{"min_age": {minAge.String()}, "limit": {"200"}}

                Mints []Mint `json:"mints"`
                Next  string `json:"next"`
        }
        if err := a.do(ctx, http.MethodGet, "/internal/openfga/session-mints?"+q.Encode(), nil, &page); err != nil {
                return nil, "", err
        }
        return page.Mints, page.Next, nil
}

// CloseMint deletes every tuple on the minted session and closes its mint.

        return a.do(ctx, http.MethodPost, "/internal/openfga/session-mints/"+url.PathEscape(id)+"/close", nil, nil)
}

// Package reconcile rebuilds the OpenFGA session tuples from the membership collection, which is
// the source of truth (PADR-0035). It runs as the initiative-reconciler user with the
// openfga-writer role and an operator-minted bearer (PADR-0038). Every pass is idempotent: it
// reads the tuples on each session, then that session's current memberships, diffs them, and
// applies the difference, so a failed pass is finished by the next one. Reading tuples first means
// a grant or revoke that lands between the two reads is, at worst, corrected by the next pass
// (two narrow interleavings remain, accepted by PADR-0040 section 4). After the tuple pass it
// collects session mints that never became sessions (PADR-0040 section 1).
package reconcile

import (
        "context"
        "errors"
        "fmt"
        "log/slog"
        "regexp"
        "sort"
        "strings"
        "time"

        "github.com/sweetrpg/initiative-api/data"
)

const (
        scanPage   = 500
        maxBatch   = 100 // auth-api's tuple write limit
        userPrefix = "user:"
        sessionObj = "session:"
        // mintMinAge keeps garbage collection away from sessions still being created; auth-api
        // compares it against its own clock.
        mintMinAge = 15 * time.Minute
        // maxMintPages bounds the garbage-collection paging loop so a cursor that never ends cannot
        // hold the job until its deadline.
        maxMintPages = 1000
        // defaultGCReserve is the end of the pass's time budget kept for garbage collection, so a slow
        // tuple pass cannot starve it.
        defaultGCReserve = 60 * time.Second
        // maxGMLessPerPass is the most GM-less sessions one pass will remove. More than that looks like
        // a wrong database or an empty members collection, which would otherwise wipe every session
        // older than mintMinAge and its tuples.
        maxGMLessPerPass = 10
)

// mintIDPattern is what auth-api mints. Anything else matches no session, which would read as
// "missing" and close a mint that is not ours to judge.
var mintIDPattern = regexp.MustCompile(`^[0-9a-f]{24}$`)

// ErrTokenRejected means auth-api refused the reconciler's bearer (expired, revoked, or missing
// the openfga-writer role). The pass stops at once; an operator must mint a new token.
var ErrTokenRejected = errors.New("reconcile: reconciler token rejected")

// Tuple is one relationship, as auth-api's tuple endpoints carry it.
type Tuple struct {
        User     string `json:"user"`
        Relation string `json:"relation"`
        Object   string `json:"object"`
}

// Members is the membership source.
type Members interface {
        // ScanMembers pages through every membership record, deleted ones included. The pass uses it
        // only to learn which sessions exist.
        ScanMembers(ctx context.Context, q data.Query) (data.Page[data.Member], error)
        // SessionMembers returns one session's membership records, deleted ones included, read fresh.
        SessionMembers(ctx context.Context, sessionID string) ([]data.Member, error)
}

// Mint is an open session mint auth-api lists for garbage collection.
type Mint struct {
        ID string `json:"id"`
}

// Mints is auth-api's mint surface (openfga-writer): open mints at least minAge old, paged by
// the cursor each page returns, and a close that deletes every tuple on the session.
type Mints interface {
        ListMints(ctx context.Context, minAge time.Duration, after string) (mints []Mint, next string, err error)
        CloseMint(ctx context.Context, id string) error
}

// Sessions looks sessions up by ID in the sessions collection, soft-deleted ones included.
type Sessions interface {
        // LookupSessions maps each found ID to whether it is soft-deleted. A missing ID is absent.
        LookupSessions(ctx context.Context, ids []string) (map[string]bool, error)
        // SessionsWithGM returns the IDs among ids that have a live GM member record.
        SessionsWithGM(ctx context.Context, ids []string) (map[string]bool, error)
        // RemoveGMLess soft-deletes a live session that has no GM, and its membership records.
        RemoveGMLess(ctx context.Context, sessionID string) error
}

// Tuples is auth-api's internal tuple surface.
type Tuples interface {
        // Read returns every tuple on object.
        Read(ctx context.Context, object string) ([]Tuple, error)
        Write(ctx context.Context, writes, deletes []Tuple) error
}

// Reconciler diffs memberships against tuples.
type Reconciler struct {
        Members Members
        Tuples  Tuples
        // Mints and Sessions enable the mint garbage collection; both nil skips it.
        Mints    Mints
        Sessions Sessions
        // GCReserve is how much of ctx's deadline the tuple pass leaves for garbage collection; zero
        // means 60 seconds. It has no effect when ctx has no deadline.
        GCReserve time.Duration
}

// Result counts one pass.
type Result struct {
        Sessions, Written, Deleted, Failed int
        // MintsClosed counts mints garbage collection closed; MintsSkipped counts those it left for
        // the next pass because the sessions lookup failed.
        MintsClosed, MintsSkipped int
        // GMLessRemoved counts live sessions with no GM that garbage collection soft-deleted.
        GMLessRemoved int
        // GMLessCapped is set when a pass reached maxGMLessPerPass and left further GM-less sessions.
        GMLessCapped bool
}

var roleRelations = map[string]bool{data.RoleGM: true, data.RoleCoGM: true, data.RolePlayer: true, data.RoleObserver: true}

// sessionObjects lists the session objects that have any membership record, live or deleted, so a
// session whose members are all gone is still visited and its tuples removed.
func (r Reconciler) sessionObjects(ctx context.Context) ([]string, error) {
        seen := map[string]bool{}
        cursor := ""
        for {
                page, err := r.Members.ScanMembers(ctx, data.Query{Limit: scanPage, Cursor: cursor})
                if err != nil {
                        return nil, fmt.Errorf("scan members: %w", err)
                }
                for _, m := range page.Items {
                        seen[m.SessionID] = true
                }
                if page.NextCursor == "" {
                        break
                }
                cursor = page.NextCursor
        }
        ids := make([]string, 0, len(seen))
        for id := range seen {
                ids = append(ids, id)
        }
        sort.Strings(ids)
        return ids, nil
}

// wanted is the tuple set one session's live memberships call for.
func wanted(sessionID string, members []data.Member) map[Tuple]bool {
        obj := sessionObj + sessionID
        out := map[Tuple]bool{}
        for _, m := range members {
                if m.DeletedAt != nil || !roleRelations[m.Role] {
                        continue
                }
                out[Tuple{User: userPrefix + m.UserID, Relation: m.Role, Object: obj}] = true
        }
        return out
}

// Run does one pass. It returns ErrTokenRejected if auth-api refuses the token; other failures
// are logged per session and counted, and returned together after every session was tried.
func (r Reconciler) Run(ctx context.Context) (Result, error) {
        tupleCtx, cancel := r.tuplePassContext(ctx)
        defer cancel()
        ids, err := r.sessionObjects(tupleCtx)
        if err != nil {
                return Result{}, err
        }
        res := Result{Sessions: len(ids)}
        for i, id := range ids {
                if tupleCtx.Err() != nil {
                        // The tuple pass's budget is spent: one line and one failure per session left, so the
                        // job fails and garbage collection (on the full context) still runs.
                        res.Failed += len(ids) - i
                        slog.ErrorContext(ctx, "reconcile: tuple pass ran out of time", "done", i, "left", len(ids)-i)
                        break
                }
                w, d, err := r.one(tupleCtx, id)
                res.Written += w
                res.Deleted += d
                if errors.Is(err, ErrTokenRejected) {
                        return res, err
                }
                if err != nil {
                        res.Failed++
                        slog.ErrorContext(ctx, "reconcile session failed", "session", id, "error", err)
                }
        }
        // Garbage collection runs on the whole context, not the tuple pass's shorter one.
        if err := r.collect(ctx, &res); err != nil {
                return res, err
        }
        if res.Failed > 0 {
                return res, fmt.Errorf("reconcile: %d failures across %d sessions", res.Failed, res.Sessions)
        }
        return res, nil
}

// tuplePassContext is ctx with its deadline moved earlier by the GC reserve, so the tuple pass
// cannot use the budget garbage collection needs.
func (r Reconciler) tuplePassContext(ctx context.Context) (context.Context, context.CancelFunc) {
        deadline, ok := ctx.Deadline()
        if !ok {
                return context.WithCancel(ctx)
        }
        reserve := r.GCReserve
        if reserve <= 0 {
                reserve = defaultGCReserve
        }
        return context.WithDeadline(ctx, deadline.Add(-reserve))
}

// collect closes mints that never became live sessions: for each open mint at least mintMinAge
// old, one sessions lookup that includes soft-deleted records. A mint is closed only when that
// lookup succeeds and returns no record or a deleted one. A failed lookup skips the whole batch
// until the next pass. It pages to the end, because live sessions stay open and would otherwise
// fill the first page and hide a leaked mint.
func (r Reconciler) collect(ctx context.Context, res *Result) error {
        if r.Mints == nil || r.Sessions == nil {
                return nil
        }
        after := ""
        for page := 0; ; page++ {
                if page >= maxMintPages {
                        res.Failed++
                        slog.ErrorContext(ctx, "reconcile: mint paging did not end", "pages", page)
                        return nil
                }
                mints, next, err := r.Mints.ListMints(ctx, mintMinAge, after)
                if err != nil {
                        if errors.Is(err, ErrTokenRejected) {
                                return err
                        }
                        res.Failed++
                        slog.ErrorContext(ctx, "reconcile: listing session mints failed", "error", err)
                        return nil
                }
                if err := r.closeOrphans(ctx, mints, res); err != nil {
                        return err
                }
                if next == "" {
                        return nil
                }
                if next == after {
                        res.Failed++
                        slog.ErrorContext(ctx, "reconcile: auth-api returned the same mint cursor twice", "cursor", next)
                        return nil
                }
                after = next
        }
}

func (r Reconciler) closeOrphans(ctx context.Context, mints []Mint, res *Result) error {
        if len(mints) == 0 {
                return nil
        }
        ids := make([]string, 0, len(mints))
        for _, m := range mints {
                if !mintIDPattern.MatchString(m.ID) {
                        res.MintsSkipped++
                        slog.ErrorContext(ctx, "reconcile: ignoring a mint with a malformed id", "mint", m.ID)
                        continue
                }
                ids = append(ids, m.ID)
        }
        if len(ids) == 0 {
                return nil
        }
        found, err := r.Sessions.LookupSessions(ctx, ids)
        if err != nil {
                // Counted as a failure too: a Mongo outage that persists must fail the job so the stall
                // alert fires, not exit 0 with the mints left open.
                res.MintsSkipped += len(ids)
                res.Failed++
                slog.ErrorContext(ctx, "reconcile: sessions lookup failed; leaving this batch of mints for the next pass", "mints", len(ids), "error", err)
                return nil
        }
        gmless := r.gmLess(ctx, ids, found, res)
        for _, id := range ids {
                if deleted, exists := found[id]; exists && !deleted {
                        if !gmless[id] {
                                continue // a live session with its GM
                        }
                        if res.GMLessRemoved >= maxGMLessPerPass {
                                if !res.GMLessCapped {
                                        res.GMLessCapped = true
                                        res.Failed++
                                        slog.ErrorContext(ctx, "reconcile: reached the cap of sessions with no GM removed in one pass; leaving the rest", "cap", maxGMLessPerPass)
                                }
                                continue
                        }
                        err := r.Sessions.RemoveGMLess(ctx, id)
                        if errors.Is(err, data.ErrConflict) {
                                slog.InfoContext(ctx, "reconcile: session has a GM or is too new to remove; leaving it", "session", id)
                                continue
                        }
                        if err != nil {
                                res.Failed++
                                slog.ErrorContext(ctx, "reconcile: removing a session with no GM failed", "session", id, "error", err)
                                continue
                        }
                        res.GMLessRemoved++
                        slog.InfoContext(ctx, "reconcile: removed a live session that has no GM", "session", id)
                }
                if err := r.Mints.CloseMint(ctx, id); err != nil {
                        if errors.Is(err, ErrTokenRejected) {
                                return err
                        }
                        res.Failed++
                        slog.ErrorContext(ctx, "reconcile: closing a mint failed", "mint", id, "error", err)
                        continue
                }
                res.MintsClosed++
                slog.InfoContext(ctx, "reconcile: closed a mint with no live session", "mint", id)
        }
        return nil
}

// gmLess returns which of the live sessions among ids have no live GM record, using one query. A
// failed query counts as a failure and returns nothing, so no session is removed on a guess.
func (r Reconciler) gmLess(ctx context.Context, ids []string, found map[string]bool, res *Result) map[string]bool {
        var live []string
        for _, id := range ids {
                if deleted, exists := found[id]; exists && !deleted {
                        live = append(live, id)
                }
        }
        if len(live) == 0 {
                return nil
        }
        withGM, err := r.Sessions.SessionsWithGM(ctx, live)
        if err != nil {
                res.Failed++
                slog.ErrorContext(ctx, "reconcile: GM lookup failed; leaving live sessions alone", "sessions", len(live), "error", err)
                return nil
        }
        out := map[string]bool{}
        for _, id := range live {
                if !withGM[id] {
                        out[id] = true
                }
        }
        return out
}

// one reconciles a single session and returns how many tuples it wrote and deleted. It reads the
// tuples first and the session's members second.
func (r Reconciler) one(ctx context.Context, sessionID string) (written, deleted int, err error) {
        obj := sessionObj + sessionID
        have, err := r.Tuples.Read(ctx, obj)
        if err != nil {
                return 0, 0, err
        }
        members, err := r.Members.SessionMembers(ctx, sessionID)
        if err != nil {
                return 0, 0, fmt.Errorf("read members: %w", err)
        }
        want := wanted(sessionID, members)
        haveSet := map[Tuple]bool{}
        var dels []Tuple
        for _, t := range have {
                if !roleRelations[t.Relation] || !strings.HasPrefix(t.User, userPrefix) {
                        continue // not a tuple this job owns
                }
                haveSet[t] = true
                if !want[t] {
                        dels = append(dels, t)
                }
        }
        var adds []Tuple
        for t := range want {
                if !haveSet[t] {
                        adds = append(adds, t)
                }
        }
        sortTuples(adds)
        sortTuples(dels)
        for _, t := range adds {
                slog.InfoContext(ctx, "reconcile: writing missing tuple", "user", t.User, "relation", t.Relation, "object", t.Object)
        }
        for _, t := range dels {
                slog.InfoContext(ctx, "reconcile: deleting extra tuple", "user", t.User, "relation", t.Relation, "object", t.Object)
        }
        // Deletes go first so a role change (old tuple out, new in) never leaves two roles at once.
        for len(dels) > 0 || len(adds) > 0 {
                var bd, ba []Tuple
                bd, dels = take(dels, maxBatch)
                ba, adds = take(adds, maxBatch-len(bd))
                if err := r.Tuples.Write(ctx, ba, bd); err != nil {
                        return written, deleted, err
                }
                written += len(ba)
                deleted += len(bd)
        }
        return written, deleted, nil
}

func take(ts []Tuple, n int) (head, rest []Tuple) {
        if n > len(ts) {
                n = len(ts)
        }
        return ts[:n], ts[n:]
}

func sortTuples(ts []Tuple) {
        sort.Slice(ts, func(i, j int) bool {
                a, b := ts[i], ts[j]
                return a.Object+a.Relation+a.User < b.Object+b.Relation+b.User
        })
}

// Package relations is the client for auth-api's POST /authz/check-relation (PADR-0035).
//
// The caller's own bearer token is forwarded on every call (PADR-0037, no service tokens), so a
// check only answers for the user holding that token. Only positive decisions are cached, for a
// few seconds, so a removal takes effect within positiveTTL and a new grant at once. Any failure
// to get a decision is an error: session data fails closed (503), never open.
package relations

import (
        "bytes"
        "context"
        "encoding/json"
        "errors"
        "fmt"
        "io"
        "log/slog"
        "net/http"
        "sync"
        "time"
)

const (
        // positiveTTL bounds how long a removal can go unnoticed on a cached allow.
        positiveTTL = 5 * time.Second
        // maxEntries bounds the cache; past it, expired entries go first, then everything.
        maxEntries  = 10000
        callTimeout = 3 * time.Second
        maxBody     = 4 << 10
)

var (
        // ErrUnavailable means auth-api or OpenFGA could not give a decision. Callers answer 503.
        ErrUnavailable = errors.New("relations: authorization unavailable")
        // ErrUnauthorized means auth-api rejected the user's token. Callers answer 401.
        ErrUnauthorized = errors.New("relations: token rejected")
        // ErrDenied means auth-api refused the change: the caller lacks the relation, or is denied
        // the service. Callers answer 403.
        ErrDenied = errors.New("relations: not permitted")
        // ErrLimited means auth-api's mint or tuple limits stopped the call. Callers answer 429.
        ErrLimited = errors.New("relations: rate limited")
        // ErrConflict means auth-api refused the change as conflicting (the session's tuple cap).
        // Callers answer 409.
        ErrConflict = errors.New("relations: conflict")
        // ErrRejected means auth-api answered 400: this service sent a malformed request, so retrying
        // the same call never helps. Callers answer 500.
        ErrRejected = errors.New("relations: request rejected as malformed")
)

// Client calls auth-api. The zero value is not usable; use New.
type Client struct {
        base string
        http *http.Client
        now  func() time.Time

        mu    sync.Mutex
        cache map[string]time.Time // key -> expiry of a cached allow
}

// New returns a Client for the auth-api base URL.
func New(baseURL string) *Client {
        return &Client{base: baseURL, http: &http.Client{Timeout: callTimeout}, now: time.Now, cache: map[string]time.Time{}}
}

// Check reports whether the holder of token has relation on object (for example
// "session:abc", "can_view"). user is the verified caller ID and only keys the cache; auth-api
// decides from the token. A deny is (false, nil).
func (c *Client) Check(ctx context.Context, user, token, object, relation string) (bool, error) {
        if user == "" || token == "" || object == "" || relation == "" {
                return false, fmt.Errorf("relations: user, token, object and relation are required")
        }
        key := user + "\x00" + object + "\x00" + relation
        if c.allowed(key) {
                return true, nil
        }
        ok, err := c.fetch(ctx, token, object, relation)
        if err != nil {
                return false, err
        }
        if ok {
                c.remember(key)
        }
        return ok, nil
}

func (c *Client) fetch(ctx context.Context, token, object, relation string) (bool, error) {
        payload, err := json.Marshal(map[string]string{"object": object, "relation": relation})
        if err != nil {
                return false, fmt.Errorf("relations: encode request: %w", err)
        }
        req, err := http.NewRequestWithContext(ctx, http.MethodPost, c.base+"/authz/check-relation", bytes.NewReader(payload))
        if err != nil {
                return false, fmt.Errorf("relations: build request: %w", err)
        }
        req.Header.Set("Authorization", "Bearer "+token)
        req.Header.Set("Content-Type", "application/json")
        req.Header.Set("Accept", "application/json")

        resp, err := c.http.Do(req)
        if err != nil {
                slog.WarnContext(ctx, "check-relation call failed", "error", err)
                return false, ErrUnavailable
        }
        defer func() { _ = resp.Body.Close() }()

        switch resp.StatusCode {
        case http.StatusOK:
                var body struct {
                        Allowed bool `json:"allowed"`
                }
                if err := json.NewDecoder(io.LimitReader(resp.Body, maxBody)).Decode(&body); err != nil {
                        return false, ErrUnavailable
                }
                return body.Allowed, nil
        case http.StatusUnauthorized:
                return false, ErrUnauthorized
        default:
                slog.WarnContext(ctx, "check-relation rejected", "status", resp.StatusCode)
                return false, ErrUnavailable
        }
}

func (c *Client) allowed(key string) bool {
        c.mu.Lock()
        defer c.mu.Unlock()
        exp, ok := c.cache[key]
        return ok && c.now().Before(exp)
}

func (c *Client) remember(key string) {
        c.mu.Lock()
        defer c.mu.Unlock()
        if len(c.cache) >= maxEntries {
                now := c.now()
                for k, exp := range c.cache {
                        if !now.Before(exp) {
                                delete(c.cache, k)
                        }
                }
                if len(c.cache) >= maxEntries {
                        c.cache = map[string]time.Time{}
                }
        }
        c.cache[key] = c.now().Add(positiveTTL)
}

// Tuple is one membership relation, written for the session object the call names.
type Tuple struct {
        User     string `json:"user"` // "user:<canonical id>"
        Relation string `json:"relation"`
}

// Mint asks auth-api for a new session ID and writes the caller's gm tuple on it. The ID comes
// from auth-api, never from the client (PADR-0040).
func (c *Client) Mint(ctx context.Context, token string) (string, error) {
        var out struct {
                SessionID string `json:"session_id"`
        }
        if err := c.call(ctx, http.MethodPost, "/authz/sessions", token, nil, http.StatusCreated, &out); err != nil {
                return "", err
        }
        if out.SessionID == "" {
                return "", ErrUnavailable
        }
        return out.SessionID, nil
}

// EndSession deletes every tuple on the session and closes its mint. Only the GM may.
func (c *Client) EndSession(ctx context.Context, token, sessionID string) error {
        return c.call(ctx, http.MethodDelete, "/authz/sessions/"+sessionID, token, nil, http.StatusNoContent, nil)
}

// WriteTuples applies membership tuple changes on one session in a single atomic write. Writing a
// tuple that exists and deleting one that does not are no-ops, so a retry is safe.
func (c *Client) WriteTuples(ctx context.Context, token, sessionID string, writes, deletes []Tuple) error {
        body := struct {
                SessionID string  `json:"session_id"`
                Writes    []Tuple `json:"writes"`
                Deletes   []Tuple `json:"deletes"`
        }{sessionID, writes, deletes}
        if body.Writes == nil {
                body.Writes = []Tuple{}
        }
        if body.Deletes == nil {
                body.Deletes = []Tuple{}
        }
        return c.call(ctx, http.MethodPost, "/authz/session-tuples", token, body, http.StatusOK, nil)
}

// call sends one authenticated request and maps the answer: 401 is ErrUnauthorized, 403 ErrDenied,
// 429 ErrLimited, anything else but want (or an unreadable body) ErrUnavailable.
func (c *Client) call(ctx context.Context, method, path, token string, body any, want int, out any) error {
        if token == "" {
                return ErrUnauthorized
        }
        var rdr io.Reader
        if body != nil {
                payload, err := json.Marshal(body)
                if err != nil {
                        return fmt.Errorf("relations: encode request: %w", err)
                }
                rdr = bytes.NewReader(payload)
        }
        req, err := http.NewRequestWithContext(ctx, method, c.base+path, rdr)
        if err != nil {
                return fmt.Errorf("relations: build request: %w", err)
        }
        req.Header.Set("Authorization", "Bearer "+token)
        req.Header.Set("Content-Type", "application/json")
        req.Header.Set("Accept", "application/json")
        resp, err := c.http.Do(req)
        if err != nil {
                slog.WarnContext(ctx, "auth-api call failed", "path", path, "error", err)
                return ErrUnavailable
        }
        defer func() { _ = resp.Body.Close() }()
        switch resp.StatusCode {
        case want:
                if out == nil {
                        return nil
                }
                if err := json.NewDecoder(io.LimitReader(resp.Body, maxBody)).Decode(out); err != nil {
                        return ErrUnavailable
                }
                return nil
        case http.StatusUnauthorized:
                return ErrUnauthorized
        case http.StatusForbidden:
                return ErrDenied
        case http.StatusTooManyRequests:
                return ErrLimited
        case http.StatusConflict:
                return ErrConflict
        case http.StatusBadRequest:
                slog.ErrorContext(ctx, "auth-api rejected a session call as malformed", "path", path)
                return ErrRejected
        }
        slog.WarnContext(ctx, "auth-api rejected a session call", "path", path, "status", resp.StatusCode)
        return ErrUnavailable
}

// Package seed loads the versioned game-system definitions checked into this repo. Loading is
// idempotent: unchanged definitions are skipped and keep their identifiers.
package seed

import (
        "context"
        "embed"
        "encoding/json"
        "fmt"

        "github.com/sweetrpg/initiative-api/data"
)

//go:embed systems/families.json systems/systems.json
var files embed.FS

// Upserter is the part of data.SystemRepo the loader needs.
type Upserter interface {
        Upsert(ctx context.Context, sys data.GameSystem) (data.Outcome, error)
}

// Result counts what a load did.
type Result struct{ Created, Updated, Unchanged int }

type systemFile struct {
        ID      string              `json:"id"`
        Name    string              `json:"name"`
        Family  string              `json:"family"`
        Aliases []string            `json:"aliases"`
        Gated   bool                `json:"gated"`
        Fields  []data.TrackerField `json:"fields"` // optional override of the family's fields
}

// Definitions parses the seed files and expands each system's tracker fields from its family
// (a system may carry its own fields instead). Every result passes data.ValidateSystem.
func Definitions() ([]data.GameSystem, error) {
        var families map[string][]data.TrackerField
        if err := readJSON("systems/families.json", &families); err != nil {
                return nil, err
        }
        var systems []systemFile
        if err := readJSON("systems/systems.json", &systems); err != nil {
                return nil, err
        }
        return expand(families, systems)
}

// expand builds and validates the systems, and checks identifiers across the whole set: IDs are
// unique, and no alias repeats another alias or equals any system's ID (an alias lookup must
// never be ambiguous with an ID lookup).
func expand(families map[string][]data.TrackerField, systems []systemFile) ([]data.GameSystem, error) {
        out := make([]data.GameSystem, 0, len(systems))
        for _, s := range systems {
                fields := s.Fields
                if len(fields) == 0 {
                        var ok bool
                        if fields, ok = families[s.Family]; !ok {
                                return nil, fmt.Errorf("seed: system %q uses unknown family %q", s.ID, s.Family)
                        }
                }
                aliases := s.Aliases
                if aliases == nil {
                        aliases = []string{}
                }
                sys := data.GameSystem{ID: s.ID, Name: s.Name, Family: s.Family, Aliases: aliases, Gated: s.Gated, Fields: fields}
                if err := data.ValidateSystem(sys); err != nil {
                        return nil, err
                }
                out = append(out, sys)
        }
        return out, checkIdentifiers(out)
}

func checkIdentifiers(systems []data.GameSystem) error {
        owner := map[string]string{} // identifier -> system that claims it
        for _, s := range systems {
                if prev, dup := owner[s.ID]; dup {
                        return fmt.Errorf("seed: id %q used by both %q and %q", s.ID, prev, s.ID)
                }
                owner[s.ID] = s.ID
        }
        for _, s := range systems {
                for _, a := range s.Aliases {
                        if prev, dup := owner[a]; dup {
                                return fmt.Errorf("seed: alias %q of %q collides with %q", a, s.ID, prev)
                        }
                        owner[a] = s.ID
                }
        }
        return nil
}

func readJSON(name string, v any) error {
        b, err := files.ReadFile(name)
        if err != nil {
                return err
        }
        if err := json.Unmarshal(b, v); err != nil {
                return fmt.Errorf("seed: parse %s: %w", name, err)
        }
        return nil
}

// Load upserts every seeded definition and reports the outcomes. It stops at the first error.
func Load(ctx context.Context, repo Upserter) (Result, error) {
        defs, err := Definitions()
        if err != nil {
                return Result{}, err
        }
        var r Result
        for _, d := range defs {
                o, err := repo.Upsert(ctx, d)
                if err != nil {
                        return r, fmt.Errorf("seed: %s: %w", d.ID, err)
                }
                switch o {
                case data.Created:
                        r.Created++
                case data.Updated:
                        r.Updated++
                default:
                        r.Unchanged++
                }
        }
        return r, nil
}

const correlationKey = "correlation_id"

// Correlation reuses the caller's X-Correlation-ID or mints one, and echoes it on the response
// so a request can be followed across services and log lines.
func Correlation() gin.HandlerFunc {
        return func(c *gin.Context) {
                id := c.GetHeader(constants.CorrelationHeader)
                if id == "" || len(id) > 128 {
                        id = newID()
                }
                c.Set(correlationKey, id)
                c.Header(constants.CorrelationHeader, id)
                sloggin.AddCustomAttributes(c, slog.String("correlation_id", id))
                c.Next()
        }
}

// CorrelationID returns the request's correlation ID, or "" outside the Correlation middleware.
func CorrelationID(c *gin.Context) string {
        return c.GetString(correlationKey)
}

func newID() string {
        b := make([]byte, 16)
        _, _ = rand.Read(b)
        return hex.EncodeToString(b)
}

const (
        credentialContextKey = "initiative.credential"
        // Keys authz-client.go uses for the caller's identity; authz.Viewer and authz.Roles read them,
        // so a credential principal looks like its owner to every handler. TestCredentialPrincipal
        // fails if the library changes them.
        authzViewerKey = "authz.viewer"
        authzRolesKey  = "authz.roles"
)

// authenticate lets an integration secret ("ic_..." bearer) stand in for the user token. A secret
// resolves to its owner on every request, so a revoke applies to the next one. The principal gets
// the user role only and its scopes gate the method: read for GET/HEAD/OPTIONS, write otherwise.
// Anything else goes to userAuth (auth-api) unchanged.
func authenticate(creds *domain.Credentials, userAuth gin.HandlerFunc) gin.HandlerFunc {
        return func(c *gin.Context) {
                secret, _ := strings.CutPrefix(c.GetHeader("Authorization"), "Bearer ")
                if creds == nil || !strings.HasPrefix(secret, domain.SecretPrefix) {
                        userAuth(c)
                        return
                }
                cred, ok := resolvedCredential(c)
                var err error
                if !ok {
                        cred, err = creds.Authenticate(c.Request.Context(), secret)
                }
                if errors.Is(err, data.ErrNotFound) {
                        c.AbortWithStatusJSON(http.StatusUnauthorized, ErrorBody{Code: "unauthorized", Message: "Missing or invalid credential.", CorrelationID: CorrelationID(c)})
                        return
                }
                if err != nil {
                        writeError(c, err, nil)
                        return
                }
                need := data.ScopeWrite
                switch c.Request.Method {
                case http.MethodGet, http.MethodHead, http.MethodOptions:
                        need = data.ScopeRead
                }
                if !slices.Contains(cred.Scopes, need) {
                        c.AbortWithStatusJSON(http.StatusForbidden, ErrorBody{Code: "scope_missing", Message: "This credential lacks the " + need + " scope.", CorrelationID: CorrelationID(c)})
                        return
                }
                c.Set(authzViewerKey, cred.OwnerUserID)
                c.Set(authzRolesKey, []string{authz.RoleUser})
                c.Set(credentialContextKey, cred.ID)
                c.Next()
        }
}

// userOnly rejects integration credentials: account-level routes need the user's own bearer
// (they forward it to entitlements-api, PADR-0037), and a credential must not manage credentials.
func userOnly(c *gin.Context) {
        if _, ok := c.Get(credentialContextKey); ok {
                c.AbortWithStatusJSON(http.StatusForbidden, ErrorBody{Code: "user_token_required", Message: "This request needs your own sign-in, not an integration credential.", CorrelationID: CorrelationID(c)})
                return
        }
        c.Next()
}

// setupAdminCredentialHandlers registers the administrator's reopen. The group takes no integration

func setupAdminCredentialHandlers(g *gin.RouterGroup, creds *domain.Credentials) {
        g.POST("/admin/credentials/reopen", func(c *gin.Context) {
                actor := authz.Viewer(c)
                if err := creds.Reopen(c.Request.Context(), actor); err != nil {
                        writeError(c, err, nil)
                        return
                }
                slog.WarnContext(c.Request.Context(), "audit: integration credentials reopened", "audit", true, "actor", actor, "correlation_id", CorrelationID(c))
                c.JSON(http.StatusOK, gin.H{"open": true})
        })
        g.GET("/admin/credentials", func(c *gin.Context) { adminListCredentials(c, creds) })
        g.GET("/admin/credentials/deletions", func(c *gin.Context) { adminDeletions(c, creds) })
        g.DELETE("/admin/credentials/:id", func(c *gin.Context) { adminDeleteCredential(c, creds) })
}

// pageQuery reads ?cursor and ?limit; it answers 400 and returns false on a bad limit.
func pageQuery(c *gin.Context) (data.Query, bool) {
        q := data.Query{Cursor: c.Query("cursor")}
        if v := c.Query("limit"); v != "" {
                n, err := strconv.Atoi(v)
                if err != nil || n < 1 {
                        badRequest(c, "limit must be a positive integer")
                        return q, false
                }
                q.Limit = n
        }
        return q, true
}

// adminListCredentials lists any owner's credentials; ?owner= narrows it. Secrets and hashes are
// never in the model's JSON.
func adminListCredentials(c *gin.Context, creds *domain.Credentials) {
        q, ok := pageQuery(c)
        if !ok {
                return
        }
        page, err := creds.ListAll(c.Request.Context(), c.Query("owner"), q)
        if err != nil {
                writeError(c, err, nil)
                return
        }
        c.JSON(http.StatusOK, gin.H{"items": page.Items, "next_cursor": page.NextCursor})
}

// adminDeleteCredential deletes a credential permanently. The reason is required; the audit entry
// is stored before the credential goes and is logged as well.
func adminDeleteCredential(c *gin.Context, creds *domain.Credentials) {
        var b struct {
                Reason string `json:"reason"`
        }
        if err := c.ShouldBindJSON(&b); err != nil || strings.TrimSpace(b.Reason) == "" {
                c.AbortWithStatusJSON(http.StatusUnprocessableEntity, ErrorBody{Code: "reason_required", Message: "A reason is required to delete a credential.", CorrelationID: CorrelationID(c)})
                return
        }
        actor := authz.Viewer(c)
        entry, err := creds.DeletePermanently(c.Request.Context(), actor, c.Param("id"), b.Reason)
        if errors.Is(err, data.ErrNotFound) || errors.Is(err, data.ErrInvalid) {
                writeError(c, err, nil)
                return
        }
        if err != nil {
                slog.ErrorContext(c.Request.Context(), "credential deletion did not complete", "actor", actor, "credential_id", c.Param("id"), "error", err, "correlation_id", CorrelationID(c))
                c.AbortWithStatusJSON(http.StatusInternalServerError, ErrorBody{Code: "deletion_incomplete", Message: "The deletion did not complete and may be partly done. Retry the same request.", CorrelationID: CorrelationID(c)})
                return
        }
        slog.WarnContext(c.Request.Context(), "audit: integration credential deleted", "audit", true, "actor", actor,
                "credential_id", entry.CredentialID, "owner", entry.OwnerUserID, "label", entry.Label, "reason", entry.Reason, "correlation_id", CorrelationID(c))
        c.JSON(http.StatusOK, entry)
}

        h := entitlementHandlers{svc: s, usage: map[string]func(context.Context, string) (int64, error){
                "initiative.theme_slots":      s.Themes.Count,
                "initiative.saved_rosters":    s.Rosters.Count,
                "initiative.saved_encounters": s.Encounters.Count,
        }}
        g.GET("/entitlements", h.list)
}

// list shows the caller's Initiative entitlements with usage so clients can render state and
// upgrade prompts. Clients never use it to decide anything; enforcement stays in the handlers
// that create things.
func (h entitlementHandlers) list(c *gin.Context) {
        owner := authz.Viewer(c)
        results, err := h.svc.Entitlements.Prefix(c.Request.Context(), owner, authz.Token(c), entitlementPrefix)
        if err != nil {
                writeError(c, err, nil)
                return
        }
        items := make([]EntitlementItem, 0, len(results))
        for _, r := range results {
                it := EntitlementItem{Key: r.Key, Type: r.Type}
                if r.Type == "flag" {
                        it.Value = r.Flag
                } else {
                        it.Value = r.Limit
                }
                if count, ok := h.usage[r.Key]; ok {
                        n, err := count(c.Request.Context(), owner)
                        if err != nil {
                                writeError(c, err, nil)
                                return
                        }
                        it.Usage = &n
                }
                items = append(items, it)
        }
        c.JSON(http.StatusOK, gin.H{"items": items})
}

// ErrorBody is the typed error response: a stable code, a safe message, the correlation ID, and
// for entitlement denials the key and limit. Current carries the stored record on a 409.
type ErrorBody struct {
        Code          string `json:"code"`
        Message       string `json:"message"`
        CorrelationID string `json:"correlation_id,omitempty"`
        Key           string `json:"key,omitempty"`
        Limit         *int   `json:"limit,omitempty"`
        Current       any    `json:"current,omitempty"`
}

// writeError maps err to a status and typed body. current is returned with revision conflicts.
// Unrecognized errors are logged and reported as an opaque 500.
func writeError(c *gin.Context, err error, current any) {
        body, status := classifyError(err)
        body.CorrelationID = CorrelationID(c)
        if status == http.StatusConflict && body.Code == "revision_conflict" {
                body.Current = current
        }
        if status >= http.StatusInternalServerError && status != http.StatusServiceUnavailable {
                slog.ErrorContext(c.Request.Context(), "request failed", "error", err, "correlation_id", body.CorrelationID)
        }
        if body.Code == "membership_sync_pending" || body.Code == "rate_limited" {
                c.Header("Retry-After", "5")
        }
        c.AbortWithStatusJSON(status, body)
}

func classifyError(err error) (ErrorBody, int) {
        var ee *domain.EntitlementError
        var er *domain.EntitlementRequiredError
        switch {
        case errors.As(err, &ee):
                return ErrorBody{Code: "entitlement_limit_reached", Message: "Your plan's limit for this item has been reached.", Key: ee.Key, Limit: &ee.Limit}, http.StatusForbidden
        case errors.Is(err, domain.ErrReadOnly):
                return ErrorBody{Code: "packaged_theme_read_only", Message: "Packaged themes cannot be changed. Create a custom theme from it instead."}, http.StatusForbidden
        case errors.Is(err, domain.ErrCredentialsClosed):
                return ErrorBody{Code: "credentials_unavailable", Message: "Integration credentials are temporarily unavailable."}, http.StatusServiceUnavailable
        case errors.Is(err, domain.ErrOwnerLimitsUnknown):
                return ErrorBody{Code: "owner_entitlements_pending", Message: "The session owner has not connected yet. Try again after they do."}, http.StatusServiceUnavailable
        case errors.Is(err, domain.ErrSyncPending):
                return ErrorBody{Code: "membership_sync_pending", Message: "Your change was saved, but access is still being updated. Repeat the request in a moment."}, http.StatusServiceUnavailable
        case errors.Is(err, relations.ErrDenied):
                return ErrorBody{Code: "forbidden", Message: "You may not do that."}, http.StatusForbidden
        case errors.Is(err, relations.ErrRejected):
                // A malformed request is our bug and retrying cannot help; not 503. It falls to the opaque
                // 500 and writeError logs it.
                return ErrorBody{Code: "internal", Message: "Something went wrong."}, http.StatusInternalServerError
        case errors.Is(err, relations.ErrConflict):
                // The member record is already saved, and the reconciler writes tuples from Mongo through
                // the internal path, which the 200-tuple cap does not cover, so a refused member is let in
                // on the next pass. Accepted: the members_per_session entitlement caps a session at 20
                // members, and only a direct caller of auth-api can reach the tuple cap.
                return ErrorBody{Code: "session_full", Message: "The session has reached its member limit."}, http.StatusConflict
        case errors.Is(err, relations.ErrLimited):
                return ErrorBody{Code: "rate_limited", Message: "Too many requests. Try again shortly."}, http.StatusTooManyRequests
        case errors.Is(err, relations.ErrUnavailable):
                return ErrorBody{Code: "authorization_unavailable", Message: "Session access cannot be checked right now. Try again shortly."}, http.StatusServiceUnavailable
        case errors.Is(err, relations.ErrUnauthorized):
                return ErrorBody{Code: "unauthorized", Message: "Your session is not valid."}, http.StatusUnauthorized
        case errors.Is(err, entitlements.ErrUnavailable):
                return ErrorBody{Code: "entitlements_unavailable", Message: "Entitlements are temporarily unavailable. Try again shortly."}, http.StatusServiceUnavailable
        case errors.Is(err, entitlements.ErrUnauthorized):
                return ErrorBody{Code: "unauthorized", Message: "Your session is not valid."}, http.StatusUnauthorized
        case errors.Is(err, data.ErrNotFound):
                return ErrorBody{Code: "not_found", Message: "Not found."}, http.StatusNotFound
        case errors.Is(err, data.ErrLocked), errors.As(err, &er):
                return ErrorBody{Code: "entitlement_required", Message: "This needs an entitlement your account does not currently hold.", Key: domain.KeyGameSystemsExtra}, http.StatusForbidden
        case errors.Is(err, data.ErrInactive):
                return ErrorBody{Code: "item_inactive", Message: "This item is over your plan's limit. You can still view or delete it; choose which items stay active to use it again."}, http.StatusForbidden
        case errors.Is(err, data.ErrForbidden):
                return ErrorBody{Code: "forbidden", Message: "You do not own this record."}, http.StatusForbidden
        case errors.Is(err, data.ErrRevisionConflict):
                return ErrorBody{Code: "revision_conflict", Message: "The record changed since you loaded it."}, http.StatusConflict
        case errors.Is(err, data.ErrConflict):
                return ErrorBody{Code: "conflict", Message: "The request conflicts with an existing record."}, http.StatusConflict
        case errors.Is(err, data.ErrInvalid):
                return ErrorBody{Code: "invalid", Message: strings.TrimPrefix(err.Error(), data.ErrInvalid.Error()+": ")}, http.StatusUnprocessableEntity
        }
        return ErrorBody{Code: "internal", Message: "Something went wrong."}, http.StatusInternalServerError
}

        c.AbortWithStatusJSON(http.StatusBadRequest, ErrorBody{Code: "bad_request", Message: msg, CorrelationID: CorrelationID(c)})
}

// ProtectedGroup returns a route group whose every route requires a bearer token that auth-api
// verifies (signature, expiry, iss, aud) for this service and whose user holds one of roles
// (default: RoleUser). Missing or invalid tokens get 401, a missing role gets 403. Routes added
// to the group inherit this; handlers never repeat the check. The caller's own token is the only
// credential, per PADR-0037 (no service tokens), and a client application identity alone is a 401.
func ProtectedGroup(r *gin.Engine, client *authz.Client, creds *domain.Credentials, roles ...string) *gin.RouterGroup {
        if len(roles) == 0 {
                roles = []string{authz.RoleUser}
        }
        return r.Group("", typedErrors(), authenticate(creds, authz.RequireAnyRole(client, constants.ServiceName, roles...)))
}

// SetupHandlers registers every route group on r.
func SetupHandlers(r *gin.Engine, s Services) {
        setupStatusHandlers(r)

        g := ProtectedGroup(r, s.Authz, s.Credentials)
        g.Use(withChecker(s.Entitlements), withGate(s.Relations))
        if s.Limits != nil || s.Locks != nil {
                g.Use(reconcileLimits(s))
        }
        u := g.Group("", userOnly)
        setupThemeHandlers(u, s.Themes, s.Entitlements)
        setupEntitlementHandlers(u, s)
        setupCredentialHandlers(u, s)
        setupActiveItemHandlers(u, s)
        if s.Sessions != nil {
                setupSessionHandlers(u, s.Sessions)
        }
        setupRefHandlers(g, s.Encounters)
        if s.Credentials != nil {
                setupAdminCredentialHandlers(ProtectedGroup(r, s.Authz, nil, authz.RoleAdmin), s.Credentials)
        }
}

// withGate gives the data layer the relation checker, and the domain auth-api's session surface,
// both with the caller's own bearer token (PADR-0037), so session data can be authorized and
// membership tuples written per request. A nil client leaves ctx without a gate, which
// makes session data fail closed.
func withGate(c *relations.Client) gin.HandlerFunc {
        return func(ctx *gin.Context) {
                if c != nil {
                        token := authz.Token(ctx)
                        rc := data.WithGate(ctx.Request.Context(), c, token)
                        ctx.Request = ctx.Request.WithContext(domain.WithAuthority(rc, c, token))
                }
                ctx.Next()
        }
}

// reconcileEvery bounds how often one owner's limits are re-read on ordinary requests. The
// entitlements client caches each key for its own TTL, so a changed limit applies within the
// larger of the two.
var reconcileEvery = 30 * time.Second

// reconcileLimits applies the owner's current limits (inactive items beyond a reduced limit,
// reactivation on a raise) on requests carrying the owner's own bearer: that bearer is the only
// way to read their entitlements (PADR-0037), so integration credentials are skipped. A delete or
// a revoke that succeeded reconciles at once, because it freed a place. Failures never fail the
// request.
func reconcileLimits(s Services) gin.HandlerFunc {
        var last sync.Map // owner -> time of the last reconcile
        run := func(c *gin.Context, owner string) {
                last.Store(owner, time.Now())
                chk := checkerFor(c, s.Entitlements)
                if s.Limits != nil {
                        if err := s.Limits.Reconcile(c.Request.Context(), owner, chk); err != nil {
                                slog.WarnContext(c.Request.Context(), "limit reconcile failed", "error", err)
                        }
                }
                if s.Locks != nil {
                        if err := s.Locks.Reconcile(c.Request.Context(), owner, chk); err != nil {
                                slog.WarnContext(c.Request.Context(), "game-system lock reconcile failed", "error", err)
                        }
                }
        }
        return func(c *gin.Context) {
                // The handler (and the authenticate middleware before it) set the viewer, so read it first.
                owner := authz.Viewer(c)
                _, viaCredential := c.Get(credentialContextKey)
                if owner == "" || viaCredential {
                        c.Next()
                        return
                }
                // Before the handler, so the response already reflects the current limits.
                if prev, ok := last.Load(owner); !ok || time.Since(prev.(time.Time)) >= reconcileEvery {
                        run(c, owner)
                }
                c.Next()
                // A delete or revoke freed a place; an inactive item takes it now.
                if c.Writer.Status() < http.StatusMultipleChoices &&
                        (c.Request.Method == http.MethodDelete || strings.HasSuffix(c.FullPath(), "/revoke")) {
                        run(c, owner)
                }
        }
}

// withChecker puts the caller's entitlement checker on the request context for domain rules that
// check a flag mid-operation, such as choosing a gated game system.
func withChecker(ent *entitlements.Client) gin.HandlerFunc {
        return func(c *gin.Context) {
                c.Request = c.Request.WithContext(domain.WithChecker(c.Request.Context(), checkerFor(c, ent)))
                c.Next()
        }
}

// resolvedCredentialKey holds the credential the limiter step already looked up, so authenticate
// does not query it twice.
const resolvedCredentialKey = "ratelimit.credential"

// RateLimit returns the per-client limiter (PADR-0036) preceded by a step that sets its client key.
// The shared limiter keys on X-API-Key, else the IP, and both are caller-controlled, so that step
// never lets the caller's header through:
//   - a bearer that resolves to a live integration credential is keyed by the credential's ID,
//     so credentials are distinct clients;
//   - everything else, including a made-up "ic_" secret, is keyed by IP.
func RateLimit(pool *redis.Pool, opts ratelimit.Options, creds *domain.Credentials) []gin.HandlerFunc {
        return []gin.HandlerFunc{typedErrors(), clientKey(creds), ratelimit.Middleware(pool, opts)}
}

func clientKey(creds *domain.Credentials) gin.HandlerFunc {
        return func(c *gin.Context) {
                c.Request.Header.Del("X-API-Key")
                secret, _ := strings.CutPrefix(c.GetHeader("Authorization"), "Bearer ")
                if creds != nil && strings.HasPrefix(secret, domain.SecretPrefix) {
                        // Any failure (unknown, revoked, store error) leaves the IP key; authenticate then
                        // rejects or reports it.
                        if cred, err := creds.Authenticate(c.Request.Context(), secret); err == nil {
                                c.Set(resolvedCredentialKey, cred)
                                c.Request.Header.Set("X-API-Key", "cred:"+cred.ID)
                        }
                }
                c.Next()
        }
}

func resolvedCredential(c *gin.Context) (data.IntegrationCredential, bool) {
        v, ok := c.Get(resolvedCredentialKey)
        if !ok {
                return data.IntegrationCredential{}, false
        }
        cred, ok := v.(data.IntegrationCredential)
        return cred, ok
}

// typedErrors rewrites the {"error", "message"} bodies that shared middleware writes (authz-client.go
// for 401/403/503, api-core's rate limiter for 429/503) into this API's typed {"code", "message",
// "correlation_id"} shape, so a client reads one error format. Bodies that already have a code pass
// through, and the library's error string becomes the code (the token one, invalid_token, is
// unauthorized like the credential 401). It wraps the writer once per request.
func typedErrors() gin.HandlerFunc {
        return func(c *gin.Context) {
                if _, done := c.Writer.(*typedWriter); !done {
                        c.Writer = &typedWriter{ResponseWriter: c.Writer, c: c}
                }
                c.Next()
        }
}

type typedWriter struct {
        gin.ResponseWriter
        c *gin.Context
}

func (w *typedWriter) Write(b []byte) (int, error) {
        switch w.Status() {
        case http.StatusUnauthorized, http.StatusForbidden, http.StatusTooManyRequests, http.StatusServiceUnavailable:
                var old struct {
                        Error   string  `json:"error"`
                        Message string  `json:"message"`
                        Code    *string `json:"code"`
                }
                if json.Unmarshal(bytes.TrimSpace(b), &old) == nil && old.Error != "" && old.Code == nil {
                        code := old.Error
                        if code == "invalid_token" {
                                code = "unauthorized"
                        }
                        out, err := json.Marshal(ErrorBody{Code: code, Message: old.Message, CorrelationID: CorrelationID(w.c)})
                        if err != nil {
                                return 0, err
                        }
                        // The rewritten body has a different length than the one the caller sized.
                        w.Header().Del("Content-Length")
                        if _, err := w.ResponseWriter.Write(out); err != nil {
                                return 0, err
                        }
                        return len(b), nil
                }
        }
        return w.ResponseWriter.Write(b)
}

// DB is an in-memory store.Database.
type DB struct {
        mu   sync.Mutex
        cols map[string]*Collection
}

// NewDB returns an empty DB.
func NewDB() *DB { return &DB{cols: map[string]*Collection{}} }

// Collection returns (creating on first use) the named collection.
func (d *DB) Collection(name string) store.Collection { return d.Fake(name) }

// Fake returns the concrete fake collection, for assertions on its contents.
func (d *DB) Fake(name string) *Collection {
        d.mu.Lock()
        defer d.mu.Unlock()
        if c, ok := d.cols[name]; ok {
                return c
        }
        c := &Collection{}
        d.cols[name] = c
        return c
}

// Collection is one in-memory collection.
type Collection struct {
        mu      sync.Mutex
        docs    []bson.M
        Indexes []store.IndexSpec
        // FindCalls records each Find's options, so tests can prove limits and sorts reach the query.
        FindCalls []store.FindOptions
}

// Docs returns a copy of the stored documents.
func (c *Collection) Docs() []bson.M {
        c.mu.Lock()
        defer c.mu.Unlock()
        return append([]bson.M(nil), c.docs...)
}

func toM(v any) (bson.M, error) {
        b, err := bson.Marshal(v)
        if err != nil {
                return nil, err
        }
        var m bson.M
        return m, bson.Unmarshal(b, &m)
}

func decode(m bson.M, out any) error {
        b, err := bson.Marshal(m)
        if err != nil {
                return err
        }
        return bson.Unmarshal(b, out)
}

func (c *Collection) InsertOne(_ context.Context, doc any) error {
        m, err := toM(doc)
        if err != nil {
                return err
        }
        c.mu.Lock()
        defer c.mu.Unlock()
        if c.violatesUnique(m, -1) {
                return store.ErrDuplicate
        }
        c.docs = append(c.docs, m)
        return nil
}

func (c *Collection) FindOne(_ context.Context, filter bson.D, out any) error {
        c.mu.Lock()
        defer c.mu.Unlock()
        for _, d := range c.docs {
                if ok, err := match(d, filter); err != nil {
                        return err
                } else if ok {
                        return decode(d, out)
                }
        }
        return store.ErrNotFound
}

func (c *Collection) Find(_ context.Context, filter bson.D, opts store.FindOptions, out any) error {
        c.mu.Lock()
        defer c.mu.Unlock()
        c.FindCalls = append(c.FindCalls, opts)
        var hits []bson.M
        for _, d := range c.docs {
                ok, err := match(d, filter)
                if err != nil {
                        return err
                }
                if ok {
                        hits = append(hits, d)
                }
        }
        sort.SliceStable(hits, func(i, j int) bool { return less(hits[i], hits[j], opts.Sort) })
        if opts.Limit > 0 && int64(len(hits)) > opts.Limit {
                hits = hits[:opts.Limit]
        }
        res := make(bson.A, 0, len(hits))
        for _, h := range hits {
                res = append(res, project(h, opts.Projection))
        }
        return decode(bson.M{"r": res}, &wrap{out})
}

// wrap decodes {"r": [...]} straight into a slice pointer via reflection.
type wrap struct{ out any }

func (w *wrap) UnmarshalBSON(b []byte) error {
        var raw bson.Raw = b
        arr, err := raw.LookupErr("r")
        if err != nil {
                return err
        }
        rv := reflect.ValueOf(w.out).Elem()
        rv.Set(reflect.MakeSlice(rv.Type(), 0, 0))
        vals, err := arr.Array().Values()
        if err != nil {
                return err
        }
        for _, v := range vals {
                item := reflect.New(rv.Type().Elem())
                if err := v.Unmarshal(item.Interface()); err != nil {
                        return err
                }
                rv.Set(reflect.Append(rv, item.Elem()))
        }
        return nil
}

func project(d bson.M, proj bson.D) bson.M {
        if len(proj) == 0 {
                return d
        }
        out := bson.M{"_id": d["_id"]}
        for _, e := range proj {
                if v, ok := d[e.Key]; ok {
                        out[e.Key] = v
                }
        }
        return out
}

func (c *Collection) FindOneAndUpdate(_ context.Context, filter, update bson.D, out any) error {
        c.mu.Lock()
        defer c.mu.Unlock()
        for i, d := range c.docs {
                ok, err := match(d, filter)
                if err != nil {
                        return err
                }
                if !ok {
                        continue
                }
                next := cloneM(d)
                if err := applyUpdate(next, update); err != nil {
                        return err
                }
                if c.violatesUnique(next, i) {
                        return store.ErrDuplicate
                }
                c.docs[i] = next
                return decode(next, out)
        }
        return store.ErrNotFound
}

func (c *Collection) DeleteOne(_ context.Context, filter bson.D) error {
        c.mu.Lock()
        defer c.mu.Unlock()
        for i, d := range c.docs {
                ok, err := match(d, filter)
                if err != nil {
                        return err
                }
                if ok {
                        c.docs = append(c.docs[:i], c.docs[i+1:]...)
                        return nil
                }
        }
        return store.ErrNotFound
}

func (c *Collection) CountDocuments(_ context.Context, filter bson.D) (int64, error) {
        c.mu.Lock()
        defer c.mu.Unlock()
        var n int64
        for _, d := range c.docs {
                ok, err := match(d, filter)
                if err != nil {
                        return 0, err
                }
                if ok {
                        n++
                }
        }
        return n, nil
}

func (c *Collection) EnsureIndexes(_ context.Context, specs []store.IndexSpec) error {
        c.mu.Lock()
        defer c.mu.Unlock()
        c.Indexes = append([]store.IndexSpec(nil), specs...)
        return nil
}

// violatesUnique checks _id and every unique index (honoring partial filters) against m,
// ignoring the document at index skip.
func (c *Collection) violatesUnique(m bson.M, skip int) bool {
        for i, d := range c.docs {
                if i == skip {
                        continue
                }
                if reflect.DeepEqual(d["_id"], m["_id"]) {
                        return true
                }
                for _, ix := range c.Indexes {
                        if ix.Unique && sameKey(ix, d, m) {
                                return true
                        }
                }
        }
        return false
}

func sameKey(ix store.IndexSpec, a, b bson.M) bool {
        for _, d := range []bson.M{a, b} {
                if len(ix.Partial) > 0 {
                        if ok, _ := match(d, ix.Partial); !ok {
                                return false
                        }
                }
        }
        for _, k := range ix.Keys {
                av, bv := a[k.Key], b[k.Key]
                if arr, ok := av.(bson.A); ok { // multikey: any shared element collides
                        return anyShared(arr, bv)
                }
                if !reflect.DeepEqual(norm(av), norm(bv)) {
                        return false
                }
        }
        return true
}

func anyShared(a bson.A, b any) bool {
        barr, ok := b.(bson.A)
        if !ok {
                return false
        }
        for _, x := range a {
                for _, y := range barr {
                        if reflect.DeepEqual(norm(x), norm(y)) {
                                return true
                        }
                }
        }
        return false
}

func cloneM(m bson.M) bson.M {
        out := make(bson.M, len(m))
        for k, v := range m {
                out[k] = v
        }
        return out
}

func applyUpdate(d bson.M, update bson.D) error {
        for _, op := range update {
                fields, ok := op.Value.(bson.D)
                if !ok {
                        return fmt.Errorf("fake: update operand for %s must be bson.D", op.Key)
                }
                for _, f := range fields {
                        switch op.Key {
                        case "$set":
                                m, err := toM(bson.D{{Key: "v", Value: f.Value}})
                                if err != nil {
                                        return err
                                }
                                d[f.Key] = m["v"]
                        case "$unset":
                                delete(d, f.Key)
                        case "$inc":
                                cur, _ := toFloat(norm(d[f.Key]))
                                inc, _ := toFloat(norm(f.Value))
                                d[f.Key] = int64(cur + inc)
                        default:
                                return fmt.Errorf("fake: unsupported update operator %s", op.Key)
                        }
                }
        }
        return nil
}

// norm maps stored and filter values to comparable canonical forms.
func norm(v any) any {
        switch x := v.(type) {
        case time.Time:
                return int64(primitive.NewDateTimeFromTime(x))
        case primitive.DateTime:
                return int64(x)
        case int:
                return int64(x)
        case int32:
                return int64(x)
        case float64:
                if x == float64(int64(x)) {
                        return int64(x)
                }
        case *time.Time:
                if x == nil {
                        return nil
                }
                return norm(*x)
        }
        return v
}

func toFloat(v any) (float64, bool) {
        switch x := v.(type) {
        case int64:
                return float64(x), true
        case float64:
                return x, true
        }
        return 0, false
}

func cmp(a, b any) (int, bool) {
        a, b = norm(a), norm(b)
        if af, ok := toFloat(a); ok {
                if bf, ok := toFloat(b); ok {
                        switch {
                        case af < bf:
                                return -1, true
                        case af > bf:
                                return 1, true
                        }
                        return 0, true
                }
        }
        as, aok := a.(string)
        bs, bok := b.(string)
        if aok && bok {
                switch {
                case as < bs:
                        return -1, true
                case as > bs:
                        return 1, true
                }
                return 0, true
        }
        return 0, false
}

func less(a, b bson.M, sortD bson.D) bool {
        for _, k := range sortD {
                c, _ := cmp(a[k.Key], b[k.Key])
                if c == 0 {
                        continue
                }
                if dir, _ := toFloat(norm(k.Value)); dir < 0 {
                        return c > 0
                }
                return c < 0
        }
        return false
}

func match(d bson.M, filter bson.D) (bool, error) {
        for _, e := range filter {
                ok, err := matchElem(d, e)
                if err != nil || !ok {
                        return false, err
                }
        }
        return true, nil
}

func matchElem(d bson.M, e bson.E) (bool, error) {
        switch e.Key {
        case "$and", "$or":
                clauses, ok := e.Value.(bson.A)
                if !ok {
                        return false, fmt.Errorf("fake: %s needs bson.A", e.Key)
                }
                if len(clauses) == 0 {
                        return false, fmt.Errorf("fake: %s argument must be a non-empty array", e.Key)
                }
                for _, cl := range clauses {
                        sub, ok := cl.(bson.D)
                        if !ok {
                                return false, fmt.Errorf("fake: %s clause must be bson.D", e.Key)
                        }
                        m, err := match(d, sub)
                        if err != nil {
                                return false, err
                        }
                        if m == (e.Key == "$or") {
                                return m, nil
                        }
                }
                return e.Key == "$and", nil
        }
        if ops, ok := e.Value.(bson.D); ok && len(ops) > 0 && len(ops[0].Key) > 0 && ops[0].Key[0] == '$' {
                return matchOps(d[e.Key], ops)
        }
        return eq(d[e.Key], e.Value), nil
}

func eq(have, want any) bool {
        if want == nil {
                return have == nil
        }
        if arr, ok := have.(bson.A); ok { // equality against an array field matches any element
                for _, x := range arr {
                        if reflect.DeepEqual(norm(x), norm(want)) {
                                return true
                        }
                }
        }
        return reflect.DeepEqual(norm(have), norm(want))
}

func matchOps(have any, ops bson.D) (bool, error) {
        var pattern, opts string
        for _, o := range ops {
                switch o.Key {
                case "$options":
                        opts, _ = o.Value.(string)
                case "$regex":
                        pattern, _ = o.Value.(string)
                }
        }
        for _, o := range ops {
                ok, err := matchOp(have, o, pattern, opts)
                if err != nil || !ok {
                        return false, err
                }
        }
        return true, nil
}

func matchOp(have any, o bson.E, pattern, opts string) (bool, error) {
        switch o.Key {
        case "$options":
                return true, nil
        case "$ne":
                return !eq(have, o.Value), nil
        case "$in":
                vals, ok := o.Value.(bson.A)
                if !ok {
                        return false, fmt.Errorf("fake: $in needs bson.A")
                }
                for _, v := range vals {
                        if eq(have, v) {
                                return true, nil
                        }
                }
                return false, nil
        case "$regex":
                s, ok := have.(string)
                if !ok {
                        return false, nil
                }
                if opts == "i" {
                        pattern = "(?i)" + pattern
                }
                re, err := regexp.Compile(pattern)
                if err != nil {
                        return false, err
                }
                return re.MatchString(s), nil
        case "$gt", "$gte", "$lt", "$lte":
                c, ok := cmp(have, o.Value)
                if !ok {
                        return false, nil
                }
                return (o.Key == "$gt" && c > 0) || (o.Key == "$gte" && c >= 0) ||
                        (o.Key == "$lt" && c < 0) || (o.Key == "$lte" && c <= 0), nil
        case "$exists":
                want, _ := o.Value.(bool)
                return (have != nil) == want, nil
        }
        return false, fmt.Errorf("fake: unsupported operator %s", o.Key)
}

package store

import (
        "context"
        "errors"

        "go.mongodb.org/mongo-driver/bson"
        "go.mongodb.org/mongo-driver/mongo"
        "go.mongodb.org/mongo-driver/mongo/options"
)

// MongoDatabase adapts a *mongo.Database to Database.
type MongoDatabase struct{ DB *mongo.Database }

// Collection returns the named collection.
func (m MongoDatabase) Collection(name string) Collection {
        return mongoCollection{c: m.DB.Collection(name)}
}

type mongoCollection struct{ c *mongo.Collection }

func mapErr(err error) error {
        switch {
        case err == nil:
                return nil
        case errors.Is(err, mongo.ErrNoDocuments):
                return ErrNotFound
        case mongo.IsDuplicateKeyError(err):
                return ErrDuplicate
        }
        return err
}

func (m mongoCollection) InsertOne(ctx context.Context, doc any) error {
        _, err := m.c.InsertOne(ctx, doc)
        return mapErr(err)
}

func (m mongoCollection) FindOne(ctx context.Context, filter bson.D, out any) error {
        return mapErr(m.c.FindOne(ctx, filter).Decode(out))
}

func (m mongoCollection) Find(ctx context.Context, filter bson.D, opts FindOptions, out any) error {
        fo := options.Find()
        if len(opts.Sort) > 0 {
                fo.SetSort(opts.Sort)
        }
        if len(opts.Projection) > 0 {
                fo.SetProjection(opts.Projection)
        }
        if opts.Limit > 0 {
                fo.SetLimit(opts.Limit)
        }
        cur, err := m.c.Find(ctx, filter, fo)
        if err != nil {
                return mapErr(err)
        }
        return mapErr(cur.All(ctx, out))
}

func (m mongoCollection) FindOneAndUpdate(ctx context.Context, filter, update bson.D, out any) error {
        opts := options.FindOneAndUpdate().SetReturnDocument(options.After)
        return mapErr(m.c.FindOneAndUpdate(ctx, filter, update, opts).Decode(out))
}

func (m mongoCollection) DeleteOne(ctx context.Context, filter bson.D) error {
        res, err := m.c.DeleteOne(ctx, filter)
        if err != nil {
                return mapErr(err)
        }
        if res.DeletedCount == 0 {
                return ErrNotFound
        }
        return nil
}

func (m mongoCollection) CountDocuments(ctx context.Context, filter bson.D) (int64, error) {
        n, err := m.c.CountDocuments(ctx, filter)
        return n, mapErr(err)
}

func (m mongoCollection) EnsureIndexes(ctx context.Context, specs []IndexSpec) error {
        models := make([]mongo.IndexModel, 0, len(specs))
        for _, s := range specs {
                o := options.Index().SetName(s.Name).SetUnique(s.Unique)
                if len(s.Partial) > 0 {
                        o.SetPartialFilterExpression(s.Partial)
                }
                models = append(models, mongo.IndexModel{Keys: s.Keys, Options: o})
        }
        _, err := m.c.Indexes().CreateMany(ctx, models)
        return mapErr(err)
}
